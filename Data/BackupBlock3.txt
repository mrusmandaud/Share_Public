// ========================================================
// BACKUP / RESTORE
// ========================================================

android.content.SharedPreferences state =
SettingsActivity.this.getSharedPreferences(
    "settings_state",
    MODE_PRIVATE
);


// ========================================================
// WAIT FOR FILE PICKER RESULT
// ========================================================

if (!state.getBoolean(
    "backup_result_ready",
    false
)) {

    // ----------------------------------------------------
    // No result yet.
    // Start the appropriate Android file picker.
    // ----------------------------------------------------

    String action =
    state.getString(
        "backup_action",
        ""
    );


    if ("backup".equals(action)) {

        Intent intent =
        new Intent(
            Intent.ACTION_CREATE_DOCUMENT
        );

        intent.addCategory(
            Intent.CATEGORY_OPENABLE
        );

        intent.setType(
            "text/plain"
        );

        intent.putExtra(
            Intent.EXTRA_TITLE,
            "saverecords_backup.txt"
        );

        startActivityForResult(
            intent,
            7001
        );

        return;
    }


    if ("restore".equals(action)) {

        Intent intent =
        new Intent(
            Intent.ACTION_OPEN_DOCUMENT
        );

        intent.addCategory(
            Intent.CATEGORY_OPENABLE
        );

        intent.setType(
            "text/plain"
        );

        startActivityForResult(
            intent,
            7002
        );

        return;
    }

    return;
}


// ========================================================
// RESULT IS READY
// ========================================================

int resultCode =
state.getInt(
    "backup_result_code",
    0
);

String uriString =
state.getString(
    "backup_result_uri",
    ""
);


// ========================================================
// USER CANCELLED
// ========================================================

if (
    uriString == null
    || uriString.length() == 0
) {

    state.edit()
        .putBoolean(
            "backup_result_ready",
            false
        )
        .putInt(
            "backup_result_code",
            0
        )
        .putString(
            "backup_result_uri",
            ""
        )
        .apply();

    return;
}


android.net.Uri resultUri =
android.net.Uri.parse(
    uriString
);


// ========================================================
// BACKUP
// ========================================================

if (resultCode == 7001) {

    try {

        java.io.File sourceFile =
        new java.io.File(
            getFilesDir(),
            "saverecords.txt"
        );


        if (!sourceFile.exists()) {

            Toast.makeText(
                SettingsActivity.this,
                "No saved records found",
                Toast.LENGTH_SHORT
            ).show();

        } else {

            java.io.InputStream input =
            new java.io.FileInputStream(
                sourceFile
            );


            java.io.OutputStream output =
            getContentResolver()
            .openOutputStream(
                resultUri
            );


            if (
                input != null
                && output != null
            ) {

                byte[] buffer =
                new byte[8192];

                int length;

                while (
                    (length =
                    input.read(buffer)) != -1
                ) {

                    output.write(
                        buffer,
                        0,
                        length
                    );
                }


                output.flush();
                output.close();
                input.close();


                Toast.makeText(
                    SettingsActivity.this,
                    "Backup completed",
                    Toast.LENGTH_SHORT
                ).show();
            }
        }

    } catch (Exception e) {

        Toast.makeText(
            SettingsActivity.this,
            "Backup failed: "
            + e.getMessage(),
            Toast.LENGTH_LONG
        ).show();
    }
}


// ========================================================
// RESTORE
// ========================================================

if (resultCode == 7002) {

    try {

        java.io.InputStream input =
        getContentResolver()
        .openInputStream(
            resultUri
        );


        if (input == null) {

            throw new Exception(
                "Unable to open backup file"
            );
        }


        java.io.File destinationFile =
        new java.io.File(
            getFilesDir(),
            "saverecords.txt"
        );


        java.io.OutputStream output =
        new java.io.FileOutputStream(
            destinationFile,
            false
        );


        byte[] buffer =
        new byte[8192];

        int length;

        while (
            (length =
            input.read(buffer)) != -1
        ) {

            output.write(
                buffer,
                0,
                length
            );
        }


        output.flush();
        output.close();
        input.close();


        Toast.makeText(
            SettingsActivity.this,
            "Restore completed",
            Toast.LENGTH_SHORT
        ).show();

    } catch (Exception e) {

        Toast.makeText(
            SettingsActivity.this,
            "Restore failed: "
            + e.getMessage(),
            Toast.LENGTH_LONG
        ).show();
    }
}


// ========================================================
// CLEAR RESULT STATE
// ========================================================

state.edit()
    .putBoolean(
        "backup_result_ready",
        false
    )
    .putInt(
        "backup_result_code",
        0
    )
    .putString(
        "backup_result_uri",
        ""
    )
    .putString(
        "backup_action",
        ""
    )
    .apply();
