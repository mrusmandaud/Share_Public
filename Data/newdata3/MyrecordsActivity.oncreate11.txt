final java.util.ArrayList<String> allRecords =
        new java.util.ArrayList<String>();

try {

    java.io.FileInputStream fileInputStream =
            openFileInput("myrecords.txt");

    java.io.BufferedReader reader =
            new java.io.BufferedReader(
                    new java.io.InputStreamReader(
                            fileInputStream,
                            "UTF-8"
                    )
            );

    String line;

    while (
            (line = reader.readLine()) != null
    ) {

        line = line.trim();

        if (line.length() > 0) {

            allRecords.add(line);

        }

    }

    reader.close();
    fileInputStream.close();

} catch (Exception e) {

    allRecords.clear();

}

// ============================================================
// NATIVE SELECTION INITIALIZATION
//
// IMPORTANT:
// This is deliberately NOT:
//     SELECTION();
//
// because your compiler reports that More Block as undefined
// from this injected source file.
//
// The selection is attached directly to the native cells here.
// ============================================================

final java.util.ArrayList<android.widget.TextView> selectionCells =
        new java.util.ArrayList<android.widget.TextView>();

final java.util.ArrayList<
        java.util.ArrayList<android.widget.TextView>
> selectionRecords =
        new java.util.ArrayList<
                java.util.ArrayList<android.widget.TextView>
        >();

for (
        int i = 0;
        i < recordsContainer.getChildCount();
        i++
) {

    android.view.View recordView =
            recordsContainer.getChildAt(i);

    if (
            !(recordView instanceof
                    android.widget.LinearLayout)
    ) {
        continue;
    }

    android.widget.LinearLayout record =
            (android.widget.LinearLayout) recordView;

    java.util.ArrayList<android.widget.TextView>
            recordCells =
            new java.util.ArrayList<android.widget.TextView>();

    for (
            int j = 0;
            j < record.getChildCount();
            j++
    ) {

        android.view.View child =
                record.getChildAt(j);

        if (
                !(child instanceof
                        android.widget.LinearLayout)
        ) {
            continue;
        }

        android.widget.LinearLayout recordRow =
                (android.widget.LinearLayout) child;

        for (
                int k = 0;
                k < recordRow.getChildCount();
                k++
        ) {

            android.view.View cellView =
                    recordRow.getChildAt(k);

            if (
                    cellView instanceof
                            android.widget.TextView
            ) {

                android.widget.TextView cell =
                        (android.widget.TextView)
                                cellView;

                recordCells.add(cell);
                selectionCells.add(cell);
            }
        }
    }

    if (recordCells.size() > 0) {
        selectionRecords.add(recordCells);
    }
}


final int selectionBlue =
        android.graphics.Color.rgb(
                33,
                150,
                243
        );

final int selectionWhite =
        android.graphics.Color.WHITE;

final int selectionBlack =
        android.graphics.Color.BLACK;

final int[] selectionFirst =
        new int[] { -1 };


final Runnable clearNativeSelection =
        new Runnable() {

            @Override
            public void run() {

                for (
                        int i = 0;
                        i < selectionCells.size();
                        i++
                ) {

                    android.widget.TextView cell =
                            selectionCells.get(i);

                    cell.setBackgroundColor(
                            selectionWhite
                    );

                    cell.setTextColor(
                            selectionBlack
                    );
                }

                selectionFirst[0] = -1;
            }
        };


for (
        int i = 0;
        i < selectionCells.size();
        i++
) {

    final android.widget.TextView cell =
            selectionCells.get(i);

    final int cellIndex = i;

    cell.setClickable(true);

    cell.setOnClickListener(
            new android.view.View.OnClickListener() {

                @Override
                public void onClick(
                        android.view.View v
                ) {

                    if (
                            selectionFirst[0] == cellIndex
                    ) {

                        clearNativeSelection.run();
                        return;
                    }


                    if (
                            selectionFirst[0] == -1
                    ) {

                        clearNativeSelection.run();

                        cell.setBackgroundColor(
                                selectionBlue
                        );

                        cell.setTextColor(
                                selectionWhite
                        );

                        selectionFirst[0] =
                                cellIndex;

                        return;
                    }


                    int start =
                            selectionFirst[0];

                    int end =
                            cellIndex;

                    if (start > end) {

                        int temp =
                                start;

                        start = end;
                        end = temp;
                    }

                    clearNativeSelection.run();

                    for (
                            int x = start;
                            x <= end;
                            x++
                    ) {

                        android.widget.TextView
                                selectedCell =
                                selectionCells.get(x);

                        selectedCell.setBackgroundColor(
                                selectionBlue
                        );

                        selectedCell.setTextColor(
                                selectionWhite
                        );
                    }

                    selectionFirst[0] = -1;
                }
            }
    );
}


// ============================================================
// LONG PRESS = COMPLETE RECORD
// ============================================================

for (
        int r = 0;
        r < selectionRecords.size();
        r++
) {

    final java.util.ArrayList<
            android.widget.TextView
    > recordCells =
            selectionRecords.get(r);

    for (
            int c = 0;
            c < recordCells.size();
            c++
    ) {

        android.widget.TextView cell =
                recordCells.get(c);

        cell.setOnLongClickListener(
                new android.view.View.OnLongClickListener() {

                    @Override
                    public boolean onLongClick(
                            android.view.View v
                    ) {

                        clearNativeSelection.run();

                        for (
                                int x = 0;
                                x < recordCells.size();
                                x++
                        ) {

                            android.widget.TextView
                                    selectedCell =
                                    recordCells.get(x);

                            selectedCell.setBackgroundColor(
                                    selectionBlue
                            );

                            selectedCell.setTextColor(
                                    selectionWhite
                            );
                        }

                        selectionFirst[0] = -1;

                        return true;
                    }
                }
        );
    }
}


// ============================================================
// SEARCH
// ============================================================

searchInput.addTextChangedListener(
        new android.text.TextWatcher() {

            @Override
            public void beforeTextChanged(
                    CharSequence s,
                    int start,
                    int count,
                    int after
            ) {
            }


            @Override
            public void onTextChanged(
                    CharSequence s,
                    int start,
                    int before,
                    int count
            ) {

                String query = "";

                if (s != null) {

                    query =
                            s.toString()
                                    .trim()
                                    .toLowerCase(
                                            java.util.Locale
                                                    .getDefault()
                                    );
                }

                int visibleCount = 0;


                for (
                        int i = 0;
                        i < recordsContainer.getChildCount();
                        i++
                ) {

                    android.view.View child =
                            recordsContainer.getChildAt(i);

                    if (child.getTag() == null) {
                        continue;
                    }

                    String text =
                            child.getTag()
                                    .toString()
                                    .toLowerCase(
                                            java.util.Locale
                                                    .getDefault()
                                    );

                    if (
                            query.length() == 0 ||
                            text.contains(query)
                    ) {

                        child.setVisibility(
                                android.view.View.VISIBLE
                        );

                        visibleCount++;

                    } else {

                        child.setVisibility(
                                android.view.View.GONE
                        );
                    }
                }


                if (visibleCount == 0) {

                    emptyState.setVisibility(
                            android.view.View.VISIBLE
                    );

                    if (query.length() == 0) {

                        emptyTitle.setText(
                                "No Records Yet"
                        );

                        emptyMessage.setText(
                                "Your saved records will appear here."
                        );

                    } else {

                        emptyTitle.setText(
                                "No Matching Records"
                        );

                        emptyMessage.setText(
                                "No saved record matches your search."
                        );
                    }

                } else {

                    emptyState.setVisibility(
                            android.view.View.GONE
                    );
                }
            }


            @Override
            public void afterTextChanged(
                    android.text.Editable s
            ) {
            }

        }
