selectAllButton.setOnClickListener(

            @Override
            public void onClick(
                    android.view.View v
            ) {

                final int BLUE =
                        android.graphics.Color.rgb(
                                33,
                                150,
                                243
                        );

                final int WHITE =
                        android.graphics.Color.WHITE;


                for (
                        int i = 0;
                        i < popupRecordsContainer.getChildCount();
                        i++
                ) {

                    android.view.View recordView =
                            popupRecordsContainer
                                    .getChildAt(i);

                    if (
                            !(recordView instanceof
                                    android.widget.LinearLayout)
                    ) {
                        continue;
                    }

                    android.widget.LinearLayout record =
                            (android.widget.LinearLayout)
                                    recordView;


                    for (
                            int j = 0;
                            j < record.getChildCount();
                            j++
                    ) {

                        android.view.View rowView =
                                record.getChildAt(j);

                        if (
                                !(rowView instanceof
                                        android.widget.LinearLayout)
                        ) {
                            continue;
                        }

                        android.widget.LinearLayout row =
                                (android.widget.LinearLayout)
                                        rowView;


                        for (
                                int k = 0;
                                k < row.getChildCount();
                                k++
                        ) {

                            android.view.View cellView =
                                    row.getChildAt(k);

                            if (
                                    cellView instanceof
                                            android.widget.TextView
                            ) {

                                android.widget.TextView cell =
                                        (android.widget.TextView)
                                                cellView;

                                cell.setBackgroundColor(
                                        BLUE
                                );

                                cell.setTextColor(
                                        WHITE
                                );
                            }
                        }
                    }
                }


                popupWindow.dismiss();
            }
        }
);

// COPY
// ============================================================

copyButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(
                    android.view.View v
            ) {

                final int BLUE =
                        android.graphics.Color.rgb(
                                33,
                                150,
                                243
                        );


                StringBuilder selectedText =
                        new StringBuilder();


                for (
                        int i = 0;
                        i < popupRecordsContainer.getChildCount();
                        i++
                ) {

                    android.view.View recordView =
                            popupRecordsContainer
                                    .getChildAt(i);

                    if (
                            !(recordView instanceof
                                    android.widget.LinearLayout)
                    ) {
                        continue;
                    }

                    android.widget.LinearLayout record =
                            (android.widget.LinearLayout)
                                    recordView;


                    StringBuilder recordText =
                            new StringBuilder();


                    for (
                            int j = 0;
                            j < record.getChildCount();
                            j++
                    ) {

                        android.view.View rowView =
                                record.getChildAt(j);

                        if (
                                !(rowView instanceof
                                        android.widget.LinearLayout)
                        ) {
                            continue;
                        }

                        android.widget.LinearLayout row =
                                (android.widget.LinearLayout)
                                        rowView;


                        for (
                                int k = 0;
                                k < row.getChildCount();
                                k++
                        ) {

                            android.view.View cellView =
                                    row.getChildAt(k);

                            if (
                                    !(cellView instanceof
                                            android.widget.TextView)
                            ) {
                                continue;
                            }


                            android.widget.TextView cell =
                                    (android.widget.TextView)
                                            cellView;


                            if (
                                    cell.getBackground()
                                            instanceof
                                            android.graphics.drawable.ColorDrawable
                            ) {

                                android.graphics.drawable.ColorDrawable
                                        bg =
                                        (android.graphics.drawable.ColorDrawable)
                                                cell.getBackground();


                                if (
                                        bg.getColor() == BLUE
                                ) {

                                    if (
                                            recordText.length() > 0
                                    ) {
                                        recordText.append(
                                                " | "
                                        );
                                    }

                                    recordText.append(
                                            cell.getText()
                                                    .toString()
                                    );
                                }
                            }
                        }
                    }


                    if (
                            recordText.length() > 0
                    ) {

                        if (
                                selectedText.length() > 0
                        ) {
                            selectedText.append(
                                    "\n"
                            );
                        }

                        selectedText.append(
                                recordText.toString()
                        );
                    }
                }


                if (
                        selectedText.length() == 0
                ) {

                    android.widget.Toast.makeText(
                            MyrecordsActivity.this,
                            "No data selected",
                            android.widget.Toast.LENGTH_SHORT
                    ).show();

                } else {

                    android.content.ClipboardManager
                            clipboard =
                            (android.content.ClipboardManager)
                                    getSystemService(
                                            android.content.Context
                                                    .CLIPBOARD_SERVICE
                                    );


                    clipboard.setPrimaryClip(
                            android.content.ClipData
                                    .newPlainText(
                                            "Selected Records",
                                            selectedText.toString()
                                    )
                    );


                    android.widget.Toast.makeText(
                            MyrecordsActivity.this,
                            "Copied",
                            android.widget.Toast.LENGTH_SHORT
                    ).show();
                }


                popupWindow.dismiss();
            }
        }
);
