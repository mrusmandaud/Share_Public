setContentView(R.layout.newrecord);

final android.widget.EditText recordName =
        findViewById(
                R.id.recordName
        );

final android.widget.EditText recordPhone =
        findViewById(
                R.id.recordPhone
        );

final android.widget.EditText recordAddress =
        findViewById(
                R.id.recordAddress
        );

final android.widget.Spinner recordProduct =
        findViewById(
                R.id.recordProduct
        );

final android.widget.CheckBox recordConfirmCheckBox =
        findViewById(
                R.id.recordConfirmCheckBox
        );

final android.widget.Button saveRecordButton =
        findViewById(
                R.id.saveRecordButton
        );

final android.widget.Button clearRecordButton =
        findViewById(
                R.id.clearRecordButton
        );


final java.util.ArrayList<String> productList =
        ButtonsController.getProducts(
                NewrecordActivity.this
        );


final android.widget.ArrayAdapter<String> productAdapter =
        new android.widget.ArrayAdapter<String>(
                NewrecordActivity.this,
                android.R.layout.simple_spinner_item,
                productList
        );


productAdapter.setDropDownViewResource(
        android.R.layout.simple_spinner_dropdown_item
);


recordProduct.setAdapter(
        productAdapter
);


saveRecordButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(
                    android.view.View v
            ) {

                ButtonsController.saveRecord(
                        NewrecordActivity.this,
                        recordName,
                        recordPhone,
                        recordAddress,
                        recordProduct,
                        recordConfirmCheckBox
                );
            }
        }
);


clearRecordButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(
                    android.view.View v
            ) {

                ButtonsController.clearRecord(
                        recordName,
                        recordPhone,
                        recordAddress,
                        recordProduct,
                        recordConfirmCheckBox
                );
            }
        }
);
