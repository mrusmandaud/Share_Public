final androidx.drawerlayout.widget.DrawerLayout drawer =
        findViewById(R.id.drawerMain);

final View sidebar =
        getLayoutInflater().inflate(
                R.layout.sidebar,
                drawer,
                false
        );

androidx.drawerlayout.widget.DrawerLayout.LayoutParams p =
        new androidx.drawerlayout.widget.DrawerLayout.LayoutParams(
                (int) (300 * getResources().getDisplayMetrics().density),
                -1
        );

p.gravity = android.view.Gravity.START;

drawer.addView(sidebar, p);


// =====================================================
// MAIN PAGE → NEW RECORD
// =====================================================

findViewById(R.id.btnNewRecord).setOnClickListener(
        new View.OnClickListener() {

            @Override
            public void onClick(View v) {

                ButtonsController.newRecord(
                        MainActivity.this
                );
            }
        }
);


// =====================================================
// MAIN PAGE → MY RECORDS
// =====================================================

findViewById(R.id.btnMyRecords).setOnClickListener(
        new View.OnClickListener() {

            @Override
            public void onClick(View v) {

                ButtonsController.myRecords(
                        MainActivity.this
                );
            }
        }
);


// =====================================================
// MENU BUTTON
// =====================================================

findViewById(R.id.btnMenu).setOnClickListener(
        new View.OnClickListener() {

            @Override
            public void onClick(View v) {

                drawer.openDrawer(
                        android.view.Gravity.START
                );
            }
        }
);


// =====================================================
// SIDEBAR → NEW RECORD
// =====================================================

sidebar.findViewById(
        R.id.btnSidebarNewRecord
).setOnClickListener(
        new View.OnClickListener() {

            @Override
            public void onClick(View v) {

// =====================================================
// SIDEBAR → SETTINGS
// =====================================================

sidebar.findViewById(
        R.id.btnSidebarSettings
).setOnClickListener(
        new View.OnClickListener() {

            @Override
            public void onClick(View v) {

                drawer.closeDrawer(
                        android.view.Gravity.START
                );

                startActivity(
                        new Intent(
                                MainActivity.this,
                                SettingsActivity.class
                        )
                );
            }
        }
);
// =====================================================
// SIDEBAR → CONTACT US
// =====================================================

sidebar.findViewById(
        R.id.btnSidebarContactUs
).setOnClickListener(
        new View.OnClickListener() {

            @Override
            public void onClick(View v) {

                drawer.closeDrawer(
                        android.view.Gravity.START
                );

                final View contactView =
                        getLayoutInflater().inflate(
                                R.layout.contactus,
                                null
                        );

                final android.widget.FrameLayout root =
                        findViewById(
                                android.R.id.content
                        );

                root.addView(
                        contactView,
                        new android.widget.FrameLayout.LayoutParams(
                                -1,
                                -1
                        )
                );

                contactView.bringToFront();

                contactView.findViewById(
                        R.id.contactBack
                ).setOnClickListener(
                        new View.OnClickListener() {

                            @Override
                            public void onClick(View v) {

                                root.removeView(
                                        contactView
                                );
                            }
                        }
                );
            }
        }
);
