// ============================================================
// READ SAVED RECORDS
// ============================================================

final java.util.ArrayList<String> allRecords =
        new java.util.ArrayList<String>();


try {

    java.io.FileInputStream fileInputStream =
            openFileInput("myrecords.txt");

    java.io.BufferedReader reader =
            new java.io.BufferedReader(
                    new java.io.InputStreamReader(
                            fileInputStream,
                            "UTF-8"
                    )
            );


    String line;


    while (
            (line = reader.readLine()) != null
    ) {

        line = line.trim();


        if (line.length() > 0) {

            allRecords.add(line);

        }

    }


    reader.close();

    fileInputStream.close();


} catch (Exception e) {

    allRecords.clear();

}


// ============================================================
// TOTAL RECORD COUNT
// ============================================================

recordsCount.setText(
        String.valueOf(
                allRecords.size()
        )
);


// ============================================================
// DENSITY
// ============================================================

        getResources()

                .density;

// ============================================================

                1,

// BLACK
// 40% OPACITY
//
// 102 / 255 = 40%
//
// #66000000
// ============================================================

        android.graphics.Color.argb(
                102,
                0,
                0,
                0
        );

// CLEAR OLD RECORD ROWS
// ============================================================

            topLine

// ============================================================
// EMPTY STATE
// ============================================================

    emptyState.setVisibility(
            android.view.View.VISIBLE

} else {

            android.view.View.GONE

// ============================================================
// SEARCH
// ============================================================

searchInput.addTextChangedListener(
        new android.text.TextWatcher() {

            @Override
            public void beforeTextChanged(
                    CharSequence s,
                    int start,
                    int count,
                    int after
            ) {
            }


            @Override
            public void onTextChanged(
                    CharSequence s,
                    int start,
                    int before,
                    int count
            ) {

                String query = "";


                if (s != null) {

                    query =
                            s.toString()
                                    .trim()
                                    .toLowerCase(
                                            java.util.Locale
                                                    .getDefault()
                                    );

                }


                int visibleCount = 0;


                // ====================================================
                // CHECK EVERY RECORD
                // ====================================================

                for (
                        int i = 0;
                        i < recordsContainer.getChildCount();
                        i++
                ) {

                    android.view.View child =
                            recordsContainer.getChildAt(i);


                    // Top horizontal line has no tag.
                    if (child.getTag() == null) {

                        continue;

                    }


                    String text =
                            child.getTag()
                                    .toString()
                                    .toLowerCase(
                                            java.util.Locale
                                                    .getDefault()
                                    );


                    if (
                            query.length() == 0 ||
                            text.contains(query)
                    ) {

                        child.setVisibility(
                                android.view.View.VISIBLE
                        );


                        visibleCount++;

                    } else {

                        child.setVisibility(
                                android.view.View.GONE
                        );

                    }

                }


                // ====================================================
                // SEARCH EMPTY STATE
                // ====================================================

                if (visibleCount == 0) {

                    emptyState.setVisibility(
                            android.view.View.VISIBLE
                    );


                    if (query.length() == 0) {

                        emptyTitle.setText(
                                "No Records Yet"
                        );


                        emptyMessage.setText(
                                "Your saved records will appear here."
                        );

                    } else {

                        emptyTitle.setText(
                                "No Matching Records"
                        );


                        emptyMessage.setText(
                                "No saved record matches your search."
                        );

                    }

                } else {

                    emptyState.setVisibility(
                            android.view.View.GONE
                    );

                }

            }


            @Override
            public void afterTextChanged(
                    android.text.Editable s
            ) {
            }

        }
);
