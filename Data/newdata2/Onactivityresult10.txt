@Override
protected void onActivityResult(
        int requestCode,
        int resultCode,
        Intent data) {

    super.onActivityResult(
            requestCode,
            resultCode,
            data
    );

    android.content.SharedPreferences prefs =
            getSharedPreferences(
                    "settings_state",
                    MODE_PRIVATE
            );

    if (requestCode == 7001
            || requestCode == 7002) {

        String action =
                prefs.getString(
                        "backup_action",
                        ""
                );

        boolean correctAction =
                (requestCode == 7001
                        && action.equals("backup"))
                ||
                (requestCode == 7002
                        && action.equals("restore"));

        if (!correctAction) {

            prefs.edit()
                    .remove("backup_action")
                    .remove("backup_result_uri")
                    .remove("backup_result_code")
                    .putBoolean(
                            "backup_result_ready",
                            false
                    )
                    .apply();

            return;
        }

        if (resultCode == RESULT_OK
                && data != null
                && data.getData() != null) {

            prefs.edit()
                    .putString(
                            "backup_result_uri",
                            data.getData().toString()
                    )
                    .putBoolean(
                            "backup_result_ready",
                            true
                    )
                    .putInt(
                            "backup_result_code",
                            RESULT_OK
                    )
                    .apply();

            SettingsActivity.this.BACKUP();

        } else {

            prefs.edit()
                    .remove("backup_action")
                    .remove("backup_result_uri")
                    .remove("backup_result_code")
                    .putBoolean(
                            "backup_result_ready",
                            false
                    )
                    .apply();
        }

        return;
    }

    if (requestCode == 8001) {

        if (resultCode == RESULT_OK
                && data != null
                && data.getData() != null) {

            prefs.edit()
                    .putString(
                            "export_result_uri",
                            data.getData().toString()
                    )
                    .putBoolean(
                            "export_result_ready",
                            true
                    )
                    .apply();

            SettingsActivity.this.EXPORT();

        } else {

            prefs.edit()
                    .remove("export_result_ready")
                    .remove("export_result_uri")
                    .remove("export_format")
                    .remove("export_filename")
                    .apply();
        }

        return;
    }
}
