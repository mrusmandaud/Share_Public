android.content.SharedPreferences prefs =
        getSharedPreferences(
                "settings_state",
                MODE_PRIVATE
        );

if (_requestCode == 7001
        || _requestCode == 7002) {

    String action =
            prefs.getString(
                    "backup_action",
                    ""
            );

    boolean correctAction =
            (_requestCode == 7001
                    && action.equals("backup"))
            ||
            (_requestCode == 7002
                    && action.equals("restore"));

    if (!correctAction) {

        prefs.edit()
                .remove("backup_action")
                .remove("backup_result_uri")
                .remove("backup_result_code")
                .putBoolean(
                        "backup_result_ready",
                        false
                )
                .apply();

        return;
    }

    if (_resultCode == RESULT_OK
            && _data != null
            && _data.getData() != null) {

        prefs.edit()
                .putString(
                        "backup_result_uri",
                        _data.getData().toString()
                )
                .putBoolean(
                        "backup_result_ready",
                        true
                )
                .putInt(
                        "backup_result_code",
                        RESULT_OK
                )
                .apply();

        BACKUP();

    } else {

        prefs.edit()
                .remove("backup_action")
                .remove("backup_result_uri")
                .remove("backup_result_code")
                .putBoolean(
                        "backup_result_ready",
                        false
                )
                .apply();
    }

    return;
}

if (_requestCode == 8001) {

    if (_resultCode == RESULT_OK
            && _data != null
            && _data.getData() != null) {

        prefs.edit()
                .putString(
                        "export_result_uri",
                        _data.getData().toString()
                )
                .putBoolean(
                        "export_result_ready",
                        true
                )
                .apply();

        EXPORT();

    } else {

        prefs.edit()
                .remove("export_result_ready")
                .remove("export_result_uri")
                .remove("export_format")
                .remove("export_filename")
                .apply();
    }

    return;
}
