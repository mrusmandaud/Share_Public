android.webkit.WebView selectionWebView = null;

for (
        int i = 0;
        i < binding.recordsContainer.getChildCount();
        i++
) {

    android.view.View child =
            binding.recordsContainer.getChildAt(i);

    if (
            child instanceof android.webkit.WebView
    ) {

        selectionWebView =
                (android.webkit.WebView) child;

        break;
    }
}

if (selectionWebView == null) {
    return;
}

final android.webkit.WebView finalTableView =
        selectionWebView;

finalTableView.setWebViewClient(
        new android.webkit.WebViewClient() {

            @Override
            public void onPageFinished(
                    android.webkit.WebView view,
                    String url
            ) {

                String js =
                        "(function(){" +

                        "var cells=" +
                        "document.querySelectorAll(" +
                        "'tbody td.data');" +

                        "if(cells.length===0){return;}" +

                        "var firstIndex=-1;" +

                        "function clearBlue(){" +

                        "for(var i=0;" +
                        "i<cells.length;i++){" +

                        "cells[i].style.backgroundColor=" +
                        "'white';" +

                        "cells[i].style.color=" +
                        "'black';" +

                        "}" +

                        "}" +

                        "function blueCell(i){" +

                        "cells[i].style.backgroundColor=" +
                        "'#2196F3';" +

                        "cells[i].style.color=" +
                        "'white';" +

                        "}" +

                        "function selectAllBlue(){" +

                        "clearBlue();" +

                        "for(var i=0;" +
                        "i<cells.length;i++){" +

                        "blueCell(i);" +

                        "}" +

                        "firstIndex=0;" +

                        "}" +

                        "function deselectAllBlue(){" +

                        "clearBlue();" +

                        "firstIndex=-1;" +

                        "}" +

                        "window._selectAllCells=" +

                        "function(){" +
                        "selectAllBlue();" +
                        "};" +

                        "window._deselectAllCells=" +

                        "function(){" +
                        "deselectAllBlue();" +
                        "};" +

                        "for(var i=0;" +
                        "i<cells.length;i++){" +

                        "(function(index){" +

                        "cells[index].onclick=" +
                        "function(){" +

                        "var currentColor=" +
                        "cells[index].style.backgroundColor;" +

                        "var clickedBlue=" +

                        "(currentColor===" +
                        "'rgb(33, 150, 243)' || " +

                        "currentColor==='#2196F3');" +

                        "if(clickedBlue){" +

                        "clearBlue();" +

                        "firstIndex=-1;" +

                        "return;" +

                        "}" +

                        "if(firstIndex===-1){" +

                        "clearBlue();" +

                        "blueCell(index);" +

                        "firstIndex=index;" +

                        "return;" +

                        "}" +

                        "var start=firstIndex;" +
                        "var end=index;" +

                        "if(start>end){" +

                        "var temp=start;" +

                        "start=end;" +
                        "end=temp;" +

                        "}" +

                        "clearBlue();" +

                        "for(" +
                        "var x=start;" +
                        "x<=end;" +
                        "x++" +
                        "){" +

                        "blueCell(x);" +

                        "}" +

                        "firstIndex=start;" +

                        "};" +

                        "})(i);" +

                        "}" +

                        "window._selectionReady=true;" +

                        "})()";

                view.evaluateJavascript(
                        js,
                        null
                );
            }
        }
);

finalTableView.reload();
