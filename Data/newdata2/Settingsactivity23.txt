findViewById(R.id.settingsBackupCard).setOnClickListener(
    new View.OnClickListener() {
        @Override
        public void onClick(View v) {

            getSharedPreferences(
                "settings_state",
                MODE_PRIVATE
            )
            .edit()
            .putString(
                "backup_action",
                "backup"
            )
            .putBoolean(
                "backup_result_ready",
                false
            )
            .remove(
                "backup_result_uri"
            )
            .remove(
                "backup_result_code"
            )
            .apply();

            SettingsActivity.this.BACKUP();
        }
    }
);
