// ==========================================================
// SELECTION
// SINGLE CELL + RANGE SELECTION
// ==========================================================

android.webkit.WebView selectionWebView = null;

for (
        int i = 0;
        i < binding.recordsContainer.getChildCount();
        i++
) {

    android.view.View child =
            binding.recordsContainer.getChildAt(i);

    if (
            child instanceof android.webkit.WebView
    ) {

        selectionWebView =
                (android.webkit.WebView) child;

        break;
    }
}


if (selectionWebView == null) {
    return;
}


final android.webkit.WebView finalTableView =
        selectionWebView;


// ==========================================================
// SELECTION WEBVIEW CLIENT
// ==========================================================

finalTableView.setWebViewClient(
        new android.webkit.WebViewClient() {

            @Override
            public void onPageFinished(
                    android.webkit.WebView view,
                    String url
            ) {

                String js =
                        "(function(){" +

                        "var cells=" +
                        "document.querySelectorAll(" +
                        "'tbody td.data');" +

                        "if(cells.length===0){return;}" +

                        "var firstIndex=-1;" +


                        // ==================================
                        // CLEAR ALL
                        // ==================================

                        "function clearBlue(){" +

                        "for(var i=0;" +
                        "i<cells.length;i++){" +

                        "cells[i].style.backgroundColor='white';" +

                        "cells[i].style.color='black';" +

                        "}" +

                        "}" +


                        // ==================================
                        // BLUE ONE CELL
                        // ==================================

                        "function blueCell(i){" +

                        "cells[i].style.backgroundColor=" +
                        "'#2196F3';" +

                        "cells[i].style.color='white';" +

                        "}" +


                        // ==================================
                        // SELECT ALL
                        // ==================================

                        "function selectAllBlue(){" +

                        "clearBlue();" +

                        "for(var i=0;" +
                        "i<cells.length;i++){" +

                        "blueCell(i);" +

                        "}" +

                        "firstIndex=0;" +

                        "}" +


                        // ==================================
                        // DESELECT ALL
                        // ==================================

                        "function deselectAllBlue(){" +

                        "clearBlue();" +

                        "firstIndex=-1;" +

                        "}" +


                        // ==================================
                        // EXTERNAL FUNCTIONS
                        // ==================================

                        "window._selectAllCells=" +
                        "function(){" +

                        "selectAllBlue();" +

                        "};" +

                        "window._deselectAllCells=" +
                        "function(){" +

                        "deselectAllBlue();" +

                        "};" +


                        // ==================================
                        // CELL TAP
                        // ==================================

                        "for(var i=0;" +
                        "i<cells.length;i++){" +

                        "(function(index){" +

                        "cells[index].onclick=" +
                        "function(){" +


                        // ==================================
                        // CLICK SELECTED CELL
                        // = DESELECT
                        // ==================================

                        "var currentColor=" +
                        "cells[index].style.backgroundColor;" +

                        "var clickedBlue=" +
                        "(currentColor===" +
                        "'rgb(33, 150, 243)' || " +

                        "currentColor==='#2196F3');" +

                        "if(clickedBlue){" +

                        "clearBlue();" +

                        "firstIndex=-1;" +

                        "return;" +

                        "}" +


                        // ==================================
                        // FIRST TAP
                        // SINGLE CELL
                        // ==================================

                        "if(firstIndex===-1){" +

                        "clearBlue();" +

                        "blueCell(index);" +

                        "firstIndex=index;" +

                        "return;" +

                        "}" +


                        // ==================================
                        // SECOND TAP
                        // RANGE
                        // ==================================

                        "var start=firstIndex;" +

                        "var end=index;" +


                        // ==================================
                        // REVERSE RANGE
                        // ==================================

                        "if(start>end){" +

                        "var temp=start;" +

                        "start=end;" +

                        "end=temp;" +

                        "}" +


                        "clearBlue();" +


                        // ==================================
                        // SELECT INCLUSIVE RANGE
                        // ==================================

                        "for(" +
                        "var x=start;" +
                        "x<=end;" +
                        "x++" +
                        "){" +

                        "blueCell(x);" +

                        "}" +


                        // ==================================
                        // KEEP RANGE START
                        // ==================================

                        "firstIndex=start;" +

                        "};" +

                        "})(i);" +

                        "}" +


                        "window._selectionReady=true;" +

                        "})()";


                view.evaluateJavascript(
                        js,
                        null
                );
            }
        }
);


// ==========================================================
// RELOAD TO INSTALL SELECTION
// ==========================================================

finalTableView.reload();
