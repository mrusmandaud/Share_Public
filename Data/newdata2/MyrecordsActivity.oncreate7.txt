/*
 * ==========================================================
 * READ RECORDS
 * ==========================================================
 */

String savedData = "";

try {

    java.io.FileInputStream fis =
            openFileInput(
                    "myrecords.txt"
            );

    java.io.ByteArrayOutputStream buffer =
            new java.io.ByteArrayOutputStream();

    byte[] readBuffer =
            new byte[8192];

    int length;

    while (
            (length =
                    fis.read(readBuffer)) != -1
    ) {

        buffer.write(
                readBuffer,
                0,
                length
        );
    }

    fis.close();

    savedData =
            buffer.toString(
                    "UTF-8"
            );

} catch (Exception e) {

    savedData = "";
}


/*
 * ==========================================================
 * COUNT
 * ==========================================================
 */

int recordCount = 0;

String[] countRecords =
        savedData.split(
                "\\r?\\n"
        );

for (String record : countRecords) {

    if (!record.trim().equals("")) {

        recordCount++;
    }
}

/*
 * ==========================================================
 * RECORD ROWS
 * ==========================================================
 */

String[] records =
        savedData.split(
                "\\r?\\n"
        );

int number = 1;

for (String record : records) {

    if (!record.trim().equals("")) {

        String[] parts =
                record.split(
                        "\\s*\\|\\s*",
                        -1
                );

        html.append("<tr>");

        html.append(
                "<td class='number'>"
        );

        html.append(
                String.valueOf(number)
        );

        html.append("</td>");

        for (
                int i = 0;
                i < 4;
                i++
        ) {

            html.append(
                    "<td class='data'>"
            );

            if (parts.length > i) {

                html.append(
                        android.text.TextUtils
                                .htmlEncode(
                                        parts[i]
                                )
                );
            }

            html.append("</td>");
        }

        html.append("</tr>");

        number++;
    }
}

html.append(
        "</tbody>"
        + "</table>"
        + "</body>"
        + "</html>"
);

android.view.View searchView =
        findViewById(
                R.id.recordsSearchInput
        );

if (
        searchView
                instanceof android.widget.EditText
) {

    final android.widget.EditText searchInput =
            (android.widget.EditText)
                    searchView;

    searchInput.addTextChangedListener(
            new android.text.TextWatcher() {

                @Override
                public void beforeTextChanged(
                        CharSequence s,
                        int start,
                        int count,
                        int after
                ) {
                }

                @Override
                public void onTextChanged(
                        CharSequence s,
                        int start,
                        int before,
                        int count
                ) {

                    String searchText =
                            s == null
                                    ? ""
                                    : s.toString();

                    final String javascriptText =
                            org.json.JSONObject.quote(
                                    searchText
                            );

                    recordsWebView.evaluateJavascript(
                            "(function(){"

                            + "var q="
                            + javascriptText
                            + ".toLowerCase().trim();"

                            + "var rows="
                            + "document.querySelectorAll("
                            + "'tbody tr'"
                            + ");"

                            + "for(var i=0;"
                            + "i<rows.length;"
                            + "i++){"

                            + "var cells="
                            + "rows[i].querySelectorAll("
                            + "'td.data'"
                            + ");"

                            + "var text='';"

                            + "for(var j=0;"
                            + "j<cells.length;"
                            + "j++){"

                            + "text+="
                            + "cells[j].textContent+' ';"

                            + "}"

                            + "text="
                            + "text.toLowerCase();"

                            + "if(q===''||"
                            + "text.indexOf(q)!==-1){"

                            + "rows[i].style.display='';"

                            + "}else{"

                            + "rows[i].style.display='none';"

                            + "}"

                            + "}"

                            + "})()",
                            null
                    );
                }

                @Override
                public void afterTextChanged(
                        android.text.Editable s
                ) {
                }
            }
    );
}
