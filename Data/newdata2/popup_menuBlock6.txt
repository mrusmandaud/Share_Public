/*
 * ==========================================================
 * POPUP MENU
 * ==========================================================
 */

final android.util.DisplayMetrics popupMetrics =
        getResources().getDisplayMetrics();

final float popupDensity =
        popupMetrics.density;

final int popupWidth =
        (int) (220 * popupDensity);

final int buttonHeight =
        (int) (50 * popupDensity);

final int horizontalPadding =
        (int) (16 * popupDensity);

final int verticalPadding =
        (int) (8 * popupDensity);

final int cornerRadius =
        (int) (12 * popupDensity);


/*
 * ==========================================================
 * POPUP ROOT
 * ==========================================================
 */

final android.widget.LinearLayout popupLayout =
        new android.widget.LinearLayout(
                MyrecordsActivity.this
        );

popupLayout.setOrientation(
        android.widget.LinearLayout.VERTICAL
);

popupLayout.setPadding(
        horizontalPadding,
        verticalPadding,
        horizontalPadding,
        verticalPadding
);


/*
 * ==========================================================
 * POPUP BACKGROUND
 * ==========================================================
 */

final android.graphics.drawable.GradientDrawable popupBackground =
        new android.graphics.drawable.GradientDrawable();

popupBackground.setColor(
        android.graphics.Color.WHITE
);

popupBackground.setCornerRadius(
        cornerRadius
);

popupBackground.setStroke(
        (int) (1 * popupDensity),
        android.graphics.Color.LTGRAY
);

popupLayout.setBackground(
        popupBackground
);


/*
 * ==========================================================
 * CREATE BUTTON
 * ==========================================================
 */

final java.util.function.Function<String, android.widget.Button>
        createPopupButton =
        new java.util.function.Function<String, android.widget.Button>() {

            @Override
            public android.widget.Button apply(
                    String text
            ) {

                android.widget.Button button =
                        new android.widget.Button(
                                MyrecordsActivity.this
                        );

                button.setText(text);

                button.setTextSize(
                        15
                );

                button.setTextColor(
                        android.graphics.Color.BLACK
                );

                button.setGravity(
                        android.view.Gravity.CENTER_VERTICAL
                );

                button.setAllCaps(false);

                button.setBackgroundColor(
                        android.graphics.Color.TRANSPARENT
                );

                button.setPadding(
                        (int) (8 * popupDensity),
                        0,
                        (int) (8 * popupDensity),
                        0
                );

                button.setLayoutParams(
                        new android.widget.LinearLayout.LayoutParams(
                                -1,
                                buttonHeight
                        )
                );

                return button;
            }
        };


/*
 * ==========================================================
 * COPY
 * ==========================================================
 */

final android.widget.Button copyButton =
        createPopupButton.apply(
                "Copy"
        );

copyButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(
                    android.view.View v
            ) {

                ButtonsController.copy(
                        MyrecordsActivity.this
                );
            }
        }
);

popupLayout.addView(
        copyButton
);

/*
 * ==========================================================
 * SELECT ALL
 * ==========================================================
 */

final android.widget.Button selectAllButton =
        createPopupButton.apply(
                "Select All"
        );

selectAllButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(
                    android.view.View v
            ) {

                android.view.View container =
                        findViewById(
                                R.id.recordsContainer
                        );

                if (
                        container
                                instanceof android.widget.LinearLayout
                ) {

                    android.widget.LinearLayout layout =
                            (android.widget.LinearLayout)
                                    container;

                    for (
                            int i = 0;
                            i < layout.getChildCount();
                            i++
                    ) {

                        android.view.View child =
                                layout.getChildAt(i);

                        if (
                                child
                                        instanceof android.webkit.WebView
                        ) {

                            android.webkit.WebView webView =
                                    (android.webkit.WebView)
                                            child;

                            webView.evaluateJavascript(
                                    "if(window._selectAllCells)"
                                    + "{window._selectAllCells();}",
                                    null
                            );

                            break;
                        }
                    }
                }
            }
        }
);

popupLayout.addView(
        selectAllButton
);

/*
 * ==========================================================
 * POPUP WINDOW
 * ==========================================================
 */

final android.widget.PopupWindow popupWindow =
        new android.widget.PopupWindow(
                popupLayout,
                popupWidth,
                android.widget.LinearLayout.LayoutParams.WRAP_CONTENT,
                true
        );

popupWindow.setBackgroundDrawable(
        new android.graphics.drawable.ColorDrawable(
                android.graphics.Color.TRANSPARENT
        )
);

popupWindow.setOutsideTouchable(
        true
);

popupWindow.setFocusable(
        true
);

popupWindow.setElevation(
        (int) (8 * popupDensity)
);


/*
 * ==========================================================
 * SHOW POPUP
 * ==========================================================
 */

android.view.View anchor =
        findViewById(
                R.id.recordsContainer
        );

if (anchor != null) {

    popupWindow.showAsDropDown(
            anchor,
            0,
            0
    );
}
