    // ============================================================
    // SAVE RECORD
    // ============================================================

    public static boolean saveRecord(
            Activity activity,
            EditText nameView,
            EditText phoneView,
            EditText addressView,
            Spinner productSpinner,
            CheckBox confirm
    ) {

        if (activity == null) {
            return false;
        }


        // --------------------------------------------------------
        // CONFIRM CHECKBOX
        // --------------------------------------------------------

        if (
                confirm == null
                ||
                !confirm.isChecked()
        ) {

            Toast.makeText(
                    activity,
                    "Please confirm and agree to save",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        // --------------------------------------------------------
        // NAME
        // --------------------------------------------------------

        String name = "";

        if (
                nameView != null
                &&
                nameView.getText() != null
        ) {

            name =
                    nameView
                            .getText()
                            .toString()
                            .trim();
        }


        // --------------------------------------------------------
        // PHONE
        // --------------------------------------------------------

        String phone = "";

        if (
                phoneView != null
                &&
                phoneView.getText() != null
        ) {

            phone =
                    phoneView
                            .getText()
                            .toString()
                            .trim();
        }


        // --------------------------------------------------------
        // ADDRESS
        // --------------------------------------------------------

        String address = "";

        if (
                addressView != null
                &&
                addressView.getText() != null
        ) {

            address =
                    addressView
                            .getText()
                            .toString()
                            .trim();
        }


        // --------------------------------------------------------
        // PRODUCT
        // --------------------------------------------------------

        String product = "";

        if (
                productSpinner != null
                &&
                productSpinner.getSelectedItem() != null
        ) {

            product =
                    productSpinner
                            .getSelectedItem()
                            .toString()
                            .trim();
        }


        // --------------------------------------------------------
        // VALIDATION
        // --------------------------------------------------------

        if (name.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Name",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (phone.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Phone",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (address.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Address",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (
                product.length() == 0
                ||
                product.equalsIgnoreCase(
                        "Select Product"
                )
        ) {

            Toast.makeText(
                    activity,
                    "Please select Product",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        // --------------------------------------------------------
        // RECORD FORMAT
        // --------------------------------------------------------

        String record =
                name
                        + " | "
                        + phone
                        + " | "
                        + address
                        + " | "
                        + product
                        + "\n";


        FileOutputStream output = null;


        try {

            output =
                    activity.openFileOutput(
                            RECORDS_FILE,
                            Context.MODE_APPEND
                    );

            output.write(
                    record.getBytes("UTF-8")
            );

            output.flush();


            Toast.makeText(
                    activity,
                    "Record saved successfully",
                    Toast.LENGTH_SHORT
            ).show();


            clearRecord(
                    nameView,
                    phoneView,
                    addressView,
                    productSpinner,
                    confirm
            );


            return true;


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to save record",
                    Toast.LENGTH_SHORT
            ).show();

            return false;


        } finally {

            if (output != null) {

                try {
                    output.close();
                } catch (Exception ignored) {
                }
            }
        }
    }


    // ============================================================
    // CLEAR RECORD
    // ============================================================

    public static void clearRecord(
            EditText nameView,
            EditText phoneView,
            EditText addressView,
            Spinner productSpinner,
            CheckBox confirm
    ) {

        if (nameView != null) {
            nameView.setText("");
        }

        if (phoneView != null) {
            phoneView.setText("");
        }

        if (addressView != null) {
            addressView.setText("");
        }

        if (
                productSpinner != null
                &&
                productSpinner.getAdapter() != null
                &&
                productSpinner.getAdapter().getCount() > 0
        ) {

            productSpinner.setSelection(0);
        }

        if (confirm != null) {
            confirm.setChecked(false);
        }
    }


    // ============================================================
    // GET RECORDS
    // ============================================================

    public static ArrayList<String> getRecords(
            Activity activity
    ) {

        ArrayList<String> records =
                new ArrayList<String>();

        if (activity == null) {
            return records;
        }


        BufferedReader reader = null;


        try {

            FileInputStream input =
                    activity.openFileInput(
                            RECORDS_FILE
                    );

            reader =
                    new BufferedReader(
                            new InputStreamReader(
                                    input,
                                    "UTF-8"
                            )
                    );


            String line;


            while (
                    (line = reader.readLine()) != null
            ) {

                line =
                        line.trim();

                if (line.length() > 0) {

                    records.add(line);
                }
            }


        } catch (Exception e) {

            // File does not exist or cannot be read.
            // Return empty list.


        } finally {

            if (reader != null) {

                try {
                    reader.close();
                } catch (Exception ignored) {
                }
            }
        }


        return records;
    }


    // ============================================================
    // SELECTION STATE
    //
    // records:
    //     Each item represents ONE RECORD.
    //
    // Each record contains:
    //     0 = Name
    //     1 = Phone
    //     2 = Address
    //     3 = Product
    //
    // The record-number cell is NOT included.
    // ============================================================

    private static class SelectionState {

        final ArrayList<ArrayList<TextView>>
                records;

        final ArrayList<TextView>
                cells;

        final ArrayList<Integer>
                cellRecordIndexes;


        int firstIndex = -1;

        int lastIndex = -1;

        int firstRecord = -1;

        int lastRecord = -1;


        SelectionState(
                ArrayList<ArrayList<TextView>>
                        records
        ) {

            this.records =
                    records;

            this.cells =
                    new ArrayList<TextView>();

            this.cellRecordIndexes =
                    new ArrayList<Integer>();


            if (records == null) {
                return;
            }


            // ----------------------------------------------------
            // Keep the record grouping.
            // Also create a flat lookup list only for mapping a
            // tapped TextView back to its record.
            // ----------------------------------------------------

            for (
                    int r = 0;
                    r < records.size();
                    r++
            ) {

                ArrayList<TextView> record =
                        records.get(r);

                if (record == null) {
                    continue;
                }


                for (
                        int c = 0;
                        c < record.size();
                        c++
                ) {

                    TextView cell =
                            record.get(c);

                    if (cell != null) {

                        cells.add(cell);

                        cellRecordIndexes.add(r);
                    }
                }
            }
        }
    }


    // ============================================================
    // GET SELECTION STATE
    // ============================================================

    private static SelectionState getSelectionState(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }

        return SELECTION_STATES.get(
                activity
        );
    }


    // ============================================================
    // FIND RECORD INDEX FOR A CELL
    // ============================================================

    private static int getRecordIndexForCell(
            SelectionState state,
            int cellIndex
    ) {

        if (state == null) {
            return -1;
        }

        if (
                cellIndex < 0
                ||
                cellIndex >=
                        state.cellRecordIndexes.size()
        ) {
            return -1;
        }

        return state.cellRecordIndexes.get(
                cellIndex
        );
    }


    // ============================================================
    // FIND CELL POSITION INSIDE RECORD
    // ============================================================

    private static int getCellPositionInRecord(
            SelectionState state,
            int recordIndex,
            TextView target
    ) {

        if (
                state == null
                ||
                target == null
                ||
                recordIndex < 0
                ||
                recordIndex >= state.records.size()
        ) {
            return -1;
        }


        ArrayList<TextView> record =
                state.records.get(
                        recordIndex
                );

        if (record == null) {
            return -1;
        }


        for (
                int i = 0;
                i < record.size();
                i++
        ) {

            if (record.get(i) == target) {
                return i;
            }
        }


        return -1;
    }


    // ============================================================
    // SET CELL SELECTED
    // ============================================================

    private static void setCellSelected(
            TextView cell,
            boolean selected
    ) {

        if (cell == null) {
            return;
        }


        if (selected) {

            cell.setBackgroundColor(
                    SELECTED_COLOR
            );

            cell.setTextColor(
                    WHITE
            );

        } else {

            cell.setBackgroundColor(
                    WHITE
            );

            cell.setTextColor(
                    BLACK
            );
        }
    }


    // ============================================================
    // SELECT COMPLETE RECORD
    // ============================================================

    private static void selectRecord(
            ArrayList<TextView> record
    ) {

        if (record == null) {
            return;
        }


        for (
                int i = 0;
                i < record.size();
                i++
        ) {

            setCellSelected(
                    record.get(i),
                    true
            );
        }
    }


    // ============================================================
    // CLEAR ALL VISUAL SELECTION
    // ============================================================

    private static void clearSelection(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }


        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            setCellSelected(
                    state.cells.get(i),
                    false
            );
        }
    }


    // ============================================================
    // DESELECT CURRENT STATE
    // ============================================================

    private static void deselectState(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }


        clearSelection(
                state
        );


        state.firstIndex =
                -1;

        state.lastIndex =
                -1;

        state.firstRecord =
                -1;

        state.lastRecord =
                -1;
    }


    // ============================================================
    // SELECT RECORD RANGE
    //
    // If the first and second taps are in different records,
    // every complete record between them is selected.
    //
    // This fixes the old flattened-cell range problem.
    // ============================================================

    private static void selectRecordRange(
            SelectionState state,
            int startRecord,
            int endRecord
    ) {

        if (state == null) {
            return;
        }


        if (startRecord > endRecord) {

            int temp =
                    startRecord;

            startRecord =
                    endRecord;

            endRecord =
                    temp;
        }


        for (
                int r = startRecord;
                r <= endRecord;
                r++
        ) {

            if (
                    r < 0
                    ||
                    r >= state.records.size()
            ) {
                continue;
            }


            selectRecord(
                    state.records.get(r)
            );
        }
    }


    // ============================================================
    // SELECTION
    //
    // IMPORTANT:
    // The recordsContainer tag is created by
    // MyrecordsActivity.onCreate.
    //
    // Its structure is:
    //
    // record 0:
    //     Name
    //     Phone
    //     Address
    //     Product
    //
    // record 1:
    //     Name
    //     Phone
    //     Address
    //     Product
    //
    // etc.
    //
    // The record-number cell is excluded by MyrecordsActivity.
    // ============================================================

    @SuppressWarnings("unchecked")
    public static void selection(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }


        final LinearLayout
                selectionContainer =
                activity.findViewById(
                        R.id.recordsContainer
                );


        if (selectionContainer == null) {
            return;
        }


        Object selectionObject =
                selectionContainer.getTag();


        if (!(selectionObject instanceof ArrayList)) {

            SELECTION_STATES.remove(
                    activity
            );

            return;
        }


        final ArrayList<ArrayList<TextView>>
                selectionRecords =
                (ArrayList<ArrayList<TextView>>)
                        selectionObject;


        if (selectionRecords.size() == 0) {

            SELECTION_STATES.remove(
                    activity
            );

            return;
        }


        final SelectionState state =
                new SelectionState(
                        selectionRecords
                );


        if (state.cells.size() == 0) {

            SELECTION_STATES.remove(
                    activity
            );

            return;
        }


        SELECTION_STATES.put(
                activity,
                state
        );


        // --------------------------------------------------------
        // CLEAR OLD VISUAL STATE
        // --------------------------------------------------------

        clearSelection(
                state
        );


        // --------------------------------------------------------
        // TAP HANDLERS
        // --------------------------------------------------------

        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            final TextView cell =
                    state.cells.get(i);

            if (cell == null) {
                continue;
            }


            final int index =
                    i;


            cell.setClickable(
                    true
            );


            // ----------------------------------------------------
            // NORMAL TAP
            // ----------------------------------------------------

            cell.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            SelectionState currentState =
                                    getSelectionState(
                                            activity
                                    );


                            if (currentState == null) {
                                return;
                            }


                            int tappedRecord =
                                    getRecordIndexForCell(
                                            currentState,
                                            index
                                    );


                            if (tappedRecord < 0) {
                                return;
                            }


                            // ------------------------------------
                            // FIRST TAP
                            // ------------------------------------

                            if (
                                    currentState.firstIndex
                                            == -1
                            ) {

                                clearSelection(
                                        currentState
                                );


                                setCellSelected(
                                        cell,
                                        true
                                );


                                currentState.firstIndex =
                                        index;

                                currentState.lastIndex =
                                        index;

                                currentState.firstRecord =
                                        tappedRecord;

                                currentState.lastRecord =
                                        tappedRecord;


                                return;
                            }


                            // ------------------------------------
                            // SAME SINGLE CELL
                            // ------------------------------------

                            if (
                                    currentState.firstIndex
                                            ==
                                            currentState.lastIndex
                                    &&
                                    index
                                            ==
                                            currentState.firstIndex
                            ) {

                                deselectState(
                                        currentState
                                );

                                return;
                            }


                            int firstRecord =
                                    currentState.firstRecord;


                            if (firstRecord < 0) {

                                firstRecord =
                                        getRecordIndexForCell(
                                                currentState,
                                                currentState.firstIndex
                                        );
                            }


                            // ------------------------------------
                            // SAME RECORD
                            //
                            // Preserve the original single-cell
                            // range behavior inside one record.
                            // ------------------------------------

                            if (
                                    firstRecord
                                            ==
                                            tappedRecord
                            ) {

                                ArrayList<TextView>
                                        record =
                                        currentState.records
                                                .get(
                                                        tappedRecord
                                                );


                                int firstPosition =
                                        getCellPositionInRecord(
                                                currentState,
                                                tappedRecord,
                                                currentState.cells
                                                        .get(
                                                                currentState.firstIndex
                                                        )
                                        );


                                int secondPosition =
                                        getCellPositionInRecord(
                                                currentState,
                                                tappedRecord,
                                                cell
                                        );


                                if (
                                        firstPosition < 0
                                        ||
                                        secondPosition < 0
                                ) {

                                    return;
                                }


                                int start =
                                        firstPosition;

                                int end =
                                        secondPosition;


                                if (start > end) {

                                    int temp =
                                            start;

                                    start =
                                            end;

                                    end =
                                            temp;
                                }


                                clearSelection(
                                        currentState
                                );


                                for (
                                        int c = start;
                                        c <= end;
                                        c++
                                ) {

                                    if (
                                            c >= 0
                                            &&
                                            c < record.size()
                                    ) {

                                        setCellSelected(
                                                record.get(c),
                                                true
                                        );
                                    }
                                }


                                currentState.firstRecord =
                                        tappedRecord;

                                currentState.lastRecord =
                                        tappedRecord;

                                currentState.firstIndex =
                                        index;

                                currentState.lastIndex =
                                        index;


                                return;
                            }


                            // ------------------------------------
                            // DIFFERENT RECORDS
                            //
                            // Select COMPLETE records.
                            // ------------------------------------

                            int startRecord =
                                    firstRecord;

                            int endRecord =
                                    tappedRecord;


                            if (
                                    startRecord
                                            >
                                            endRecord
                            ) {

                                int temp =
                                        startRecord;

                                startRecord =
                                        endRecord;

                                endRecord =
                                        temp;
                            }


                            clearSelection(
                                    currentState
                            );


                            selectRecordRange(
                                    currentState,
                                    startRecord,
                                    endRecord
                            );


                            currentState.firstRecord =
                                    startRecord;

                            currentState.lastRecord =
                                    endRecord;


                            currentState.firstIndex =
                                    index;

                            currentState.lastIndex =
                                    index;
                        }
                    }
            );


            // ----------------------------------------------------
            // LONG PRESS → POPUP
            // ----------------------------------------------------

            cell.setOnLongClickListener(
                    new View.OnLongClickListener() {

                        @Override
                        public boolean onLongClick(
                                View v
                        ) {

                            popupMenu(
                                    activity
                            );

                            return true;
                        }
                    }
            );
        }
    }


    // ============================================================
    // SELECT ALL
    // ============================================================

    public static void selectAll(
            Activity activity
    ) {

        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return;
        }


        if (state.cells.size() == 0) {
            return;
        }


        clearSelection(
                state
        );


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            selectRecord(
                    state.records.get(r)
            );
        }


        state.firstRecord =
                0;

        state.lastRecord =
                state.records.size() - 1;


        state.firstIndex =
                0;

        state.lastIndex =
                state.cells.size() - 1;
    }


    // ============================================================
    // DESELECT ALL
    // ============================================================

    public static void deselectAll(
            Activity activity
    ) {

        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return;
        }


        deselectState(
                state
        );
    }


    // ============================================================
    // CHECK WHETHER CELL IS SELECTED
    // ============================================================

    private static boolean isSelected(
            TextView cell
    ) {

        if (cell == null) {
            return false;
        }


        if (
                cell.getBackground()
                        instanceof ColorDrawable
        ) {

            ColorDrawable background =
                    (ColorDrawable)
                            cell.getBackground();


            return background.getColor()
                    ==
                    SELECTED_COLOR;
        }


        return false;
    }


    // ============================================================
    // GET SELECTED TEXT
    //
    // IMPORTANT:
    // Selection is treated as RECORD data.
    //
    // If any cell belonging to a record is selected,
    // the COMPLETE record is returned.
    //
    // This prevents Copy/Share from producing partial records.
    // ============================================================

    private static String getSelectedText(
            Activity activity
    ) {

        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return "";
        }


        StringBuilder selectedText =
                new StringBuilder();


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            boolean recordSelected =
                    false;


            // ----------------------------------------------------
            // A record is selected when at least one of its
            // four data cells is selected.
            // ----------------------------------------------------

            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);

                if (
                        cell != null
                        &&
                        isSelected(cell)
                ) {

                    recordSelected =
                            true;

                    break;
                }
            }


            if (!recordSelected) {
                continue;
            }


            // ----------------------------------------------------
            // Add COMPLETE record.
            // ----------------------------------------------------

            StringBuilder recordText =
                    new StringBuilder();


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);


                if (cell == null) {
                    continue;
                }


                if (
                        recordText.length() > 0
                ) {

                    recordText.append(
                            " | "
                    );
                }


                recordText.append(
                        cell.getText()
                                .toString()
                );
            }


            if (
                    recordText.length() == 0
            ) {
                continue;
            }


            if (
                    selectedText.length() > 0
            ) {

                selectedText.append(
                        "\n"
                );
            }


            selectedText.append(
                    recordText.toString()
            );
        }


        return selectedText.toString();
    }


    // ============================================================
    // COPY
    // ============================================================

    public static void copy(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }


        String selectedText =
                getSelectedText(
                        activity
                );


        if (
                selectedText == null
                ||
                selectedText.length() == 0
        ) {

            Toast.makeText(
                    activity,
                    "No data selected",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        ClipboardManager clipboard =
                (ClipboardManager)
                        activity.getSystemService(
                                Context.CLIPBOARD_SERVICE
                        );


        if (clipboard == null) {

            Toast.makeText(
                    activity,
                    "Clipboard unavailable",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        clipboard.setPrimaryClip(
                ClipData.newPlainText(
                        "Selected Records",
                        selectedText
                )
        );


        Toast.makeText(
                activity,
                "Copied",
                Toast.LENGTH_SHORT
        ).show();
    }


    // ============================================================
    // SHARE
    // ============================================================

    public static void share(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }


        String selectedText =
                getSelectedText(
                        activity
                );


        if (
                selectedText == null
                ||
                selectedText.length() == 0
        ) {

            Toast.makeText(
                    activity,
                    "No data selected",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        try {

            Intent shareIntent =
                    new Intent(
                            Intent.ACTION_SEND
                    );


            shareIntent.setType(
                    "text/plain"
            );


            shareIntent.putExtra(
                    Intent.EXTRA_TEXT,
                    selectedText
            );


            activity.startActivity(
                    Intent.createChooser(
                            shareIntent,
                            "Share"
                    )
            );


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to share",
                    Toast.LENGTH_SHORT
            ).show();
        }
    }

    // ============================================================
    // DP HELPER
    // ============================================================

    private static int dp(
            Activity activity,
            int value
    ) {

        if (activity == null) {
            return value;
        }


        float density =
                activity
                        .getResources()
                        .getDisplayMetrics()
                        .density;


        return (int)(
                value
                        * density
                        + 0.5f
        );
    }


    // ============================================================
    // GET PRODUCTS
    // ============================================================

    public static ArrayList<String> getProducts(
            Activity activity
    ) {

        ArrayList<String> products =
                new ArrayList<String>();


        if (activity == null) {

            products.add(
                    "Select Product"
            );

            return products;
        }


        SharedPreferences preferences =
                activity.getSharedPreferences(
                        PRODUCTS_PREF,
                        Context.MODE_PRIVATE
                );


        String saved =
                preferences.getString(
                        PRODUCTS_KEY,
                        ""
                );


        if (
                saved != null
                &&
                saved.trim().length() > 0
        ) {

            String[] values =
                    saved.split(
                            "\\|",
                            -1
                    );


            for (
                    String value :
                    values
            ) {

                if (value == null) {
                    continue;
                }


                value =
                        value.trim();


                if (
                        value.length() > 0
                ) {

                    products.add(
                            value
                    );
                }
            }
        }


        if (
                products.size() == 0
        ) {

            products.add(
                    "Select Product"
            );
        }


        return products;
    }


    // ============================================================
    // SAVE PRODUCTS
    // ============================================================

    public static void saveProducts(
            Activity activity,
            ArrayList<String> products
    ) {

        if (activity == null) {
            return;
        }


        StringBuilder builder =
                new StringBuilder();


        if (products != null) {

            for (
                    String product :
                    products
            ) {

                if (product == null) {
                    continue;
                }


                product =
                        product.trim();


                if (
                        product.length() == 0
                ) {
                    continue;
                }


                if (
                        builder.length() > 0
                ) {

                    builder.append(
                            "|"
                    );
                }


                builder.append(
                        product
                );
            }
        }


        activity
                .getSharedPreferences(
                        PRODUCTS_PREF,
                        Context.MODE_PRIVATE
                )
                .edit()
                .putString(
                        PRODUCTS_KEY,
                        builder.toString()
                )
                .apply();
    }


    // ============================================================
    // ADD PRODUCT
    // ============================================================

    public static void addProduct(
            Activity activity,
            String product
    ) {

        if (
                activity == null
                ||
                product == null
        ) {
            return;
        }


        product =
                product.trim();


        if (
                product.length() == 0
        ) {
            return;
        }


        ArrayList<String> products =
                getProducts(
                        activity
                );


        if (
                products.size() == 1
                &&
                products.get(0)
                        .equals(
                                "Select Product"
                        )
        ) {

            products.clear();
        }


        if (
                !products.contains(
                        product
                )
        ) {

            products.add(
                    product
            );
        }


        saveProducts(
                activity,
                products
        );
    }


    // ============================================================
    // REMOVE PRODUCT
    // ============================================================

    public static void removeProduct(
            Activity activity,
            String product
    ) {

        if (
                activity == null
                ||
                product == null
        ) {
            return;
        }


        ArrayList<String> products =
                getProducts(
                        activity
                );


        products.remove(
                product
        );


        saveProducts(
                activity,
                products
        );
    }

    // RECORD FILE EXISTS
    // ============================================================

    public static boolean recordsFileExists(
            Activity activity
    ) {

        if (activity == null) {
            return false;
        }


        try {

            FileInputStream input =
                    activity.openFileInput(
                            RECORDS_FILE
                    );


            input.close();


            return true;


        } catch (Exception e) {

            return false;
        }
    }


    // ============================================================
    // GET FILE URI
    // ============================================================

    public static Uri getFileUri(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }


        try {

            return Uri.parse(
                    activity
                            .getFilesDir()
                            .getAbsolutePath()
            );


        } catch (Exception e) {

            return null;
        }
    }
