private static int dp(            
        Context context,            
        int value            
) {            
        
    if (context == null) {            
        return value;            
    }            
        
    return (int) (            
            value            
            *            
            context.getResources()            
                    .getDisplayMetrics()            
                    .density            
            +            
            0.5f            
    );            
}

/*            
 * ============================================================            
 * GET PRODUCTS            
 * ============================================================            
 */            
        
public static ArrayList<String> getProducts(            
        Activity activity            
) {            
        
    ArrayList<String> products =            
            new ArrayList<String>();            
        
        
    if (activity == null) {            
        
        products.add(            
                "Select Product"            
        );            
        
        return products;            
    }            
        
        
    SharedPreferences preferences =            
            activity.getSharedPreferences(            
                    PRODUCTS_PREF,            
                    Context.MODE_PRIVATE            
            );            
        
        
    String saved =            
            preferences.getString(            
                    PRODUCTS_KEY,            
                    ""            
            );            
        
        
    if (            
            saved != null            
            &&            
            saved.trim().length() > 0            
    ) {            
        
        String[] values =            
                saved.split(            
                        "\\|",            
                        -1            
                );            
        
        
        for (            
                String value :            
                values            
        ) {            
        
            if (value == null) {            
                continue;            
            }            
        
        
            value =            
                    value.trim();            
        
        
            if (            
                    value.length() > 0            
            ) {            
        
                products.add(            
                        value            
                );            
            }            
        }            
    }            
        
        
    if (            
            products.size() == 0            
    ) {            
        
        products.add(            
                "Select Product"            
        );            
    }            
        
        
    return products;            
}            
        
        
/*            
 * ============================================================            
 * SAVE PRODUCTS            
 * ============================================================            
 */            
        
public static void saveProducts(            
        Activity activity,            
        ArrayList<String> products            
) {            
        
    if (activity == null) {            
        return;            
    }            
        
        
    StringBuilder builder =            
            new StringBuilder();            
        
        
    if (products != null) {            
        
        for (            
                String product :            
                products            
        ) {            
        
            if (product == null) {            
                continue;            
            }            
        
        
            product =            
                    product.trim();            
        
        
            if (            
                    product.length() == 0            
            ) {            
        
                continue;            
            }            
        
        
            if (            
                    builder.length() > 0            
            ) {            
        
                builder.append(            
                        "|"            
                );            
            }            
        
        
            builder.append(            
                    product            
            );            
        }            
    }            
        
        
    activity            
            .getSharedPreferences(            
                    PRODUCTS_PREF,            
                    Context.MODE_PRIVATE            
            )            
            .edit()            
            .putString(            
                    PRODUCTS_KEY,            
                    builder.toString()            
            )            
            .apply();            
}            
        
        
/*            
 * ============================================================            
 * ADD PRODUCT            
 * ============================================================            
 */            
        
public static void addProduct(            
        Activity activity,            
        String product            
) {            
        
    if (            
            activity == null            
            ||            
            product == null            
    ) {            
        
        return;            
    }            
        
        
    product =            
            product.trim();            
        
        
    if (            
            product.length() == 0            
    ) {            
        
        return;            
    }            
        
        
    ArrayList<String> products =            
            getProducts(            
                    activity            
            );            
        
        
    if (            
            products.size() == 1            
            &&            
            products.get(0)            
                    .equals(            
                            "Select Product"            
                    )            
    ) {            
        
        products.clear();            
    }            
        
        
    if (            
            !products.contains(            
                    product            
            )            
    ) {            
        
        products.add(            
                product            
        );            
    }            
        
        
    saveProducts(            
            activity,            
            products            
    );            
}            
        
        
/*            
 * ============================================================            
 * REMOVE PRODUCT            
 * ============================================================            
 */            
        
public static void removeProduct(            
        Activity activity,            
        String product            
) {            
        
    if (            
            activity == null            
            ||            
            product == null            
    ) {            
        
        return;            
    }            
        
        
    ArrayList<String> products =            
            getProducts(            
                    activity            
            );            
        
        
    products.remove(            
            product            
    );            
        
        
    saveProducts(            
            activity,            
            products            
    );            
}

/*            
 * ============================================================            
 * ADD PRODUCT            
 * ============================================================            
 */            
        
private static void addProduct(            
        SettingsActivity context            
) {            
        
    if (productInput == null) {            
        return;            
    }            
        
        
    String product =            
            productInput            
                    .getText()            
                    .toString()            
                    .trim();            
        
        
    /*            
     * EMPTY            
     */            
    if (product.length() == 0) {            
        
        Toast.makeText(            
                context,            
                "Enter a product name",            
                Toast.LENGTH_SHORT            
        ).show();            
        
        return;            
    }            
        
        
    /*            
     * PLACEHOLDER IS NOT A PRODUCT            
     */            
    if (product.equalsIgnoreCase(            
            SELECT_PRODUCTS            
    )            
            || product.equalsIgnoreCase(            
            "Select Product"            
    )) {            
        
        Toast.makeText(            
                context,            
                "Select Products is not a product",            
                Toast.LENGTH_SHORT            
        ).show();            
        
        return;            
    }            
        
        
    /*            
     * DUPLICATE CHECK            
     */            
    if (productExists(            
            context,            
            product            
    )) {            
        
        Toast.makeText(            
                context,            
                "Product already exists",            
                Toast.LENGTH_SHORT            
        ).show();            
        
        return;            
    }            
        
        
    /*            
     * LOCAL PRODUCT STORAGE            
     */            
    addProduct(            
            context,            
            product            
    );            
        
        
    productInput.setText(            
            ""            
    );            
        
        
    refreshProducts(            
            context            
    );            
        
        
    Toast.makeText(            
            context,            
            "Product added",            
            Toast.LENGTH_SHORT            
    ).show();            
}            
        
        
/*            
 * ============================================================            
 * CHECK DUPLICATE            
 * ============================================================            
 */            
        
private static boolean productExists(            
        SettingsActivity context,            
        String product            
) {            
        
    ArrayList<String> products =            
            getProducts(            
                    context            
            );            
        
        
    for (String existing :            
            products) {            
        
        if (existing == null) {            
            continue;            
        }            
        
        
        existing =            
                existing.trim();            
        
        
        if (existing.length() == 0) {            
            continue;            
        }            
        
        
        if (existing.equalsIgnoreCase(            
                SELECT_PRODUCTS            
        )            
                || existing.equalsIgnoreCase(            
                "Select Product"            
        )) {            
        
            continue;            
        }            
        
        
        if (existing.equalsIgnoreCase(            
                product            
        )) {            
        
            return true;            
        }            
    }            
        
        
    return false;            
}

                    dp(context, 12)            
            )            
    );            
        
        
    LinearLayout container =            
            new LinearLayout(context);            
        
    container.setPadding(            
            dp(context, 20),            
            dp(context, 4),            
            dp(context, 20),            
            dp(context, 4)            
    );            
        
    container.addView(            
            editInput,            
            new LinearLayout.LayoutParams(            
                    LinearLayout.LayoutParams.MATCH_PARENT,            
                    LinearLayout.LayoutParams.WRAP_CONTENT            
            )            
    );            
        
        
    final AlertDialog editDialog =            
            new AlertDialog.Builder(            
                    context            
            )            
            .setTitle(            
                    "Edit Product"            
            )            
            .setView(            
                    container            
            )            
            .setNegativeButton(            
                    "CANCEL",            
                    null            
            )            
            .setPositiveButton(            
                    "SAVE",            
                    null            
            )            
            .create();            
        
        
    editDialog.setOnShowListener(            
            new android.content.DialogInterface.OnShowListener() {            
        
                @Override            
                public void onShow(            
                        android.content.DialogInterface d            
                ) {            
        
                    Button saveButton =            
                            editDialog.getButton(            
                                    AlertDialog.BUTTON_POSITIVE            
                            );            
        
                    if (saveButton != null) {            
        
                        saveButton.setTextColor(            
                                COLOR_PRIMARY            
                        );            
        
                        saveButton.setOnClickListener(            
                                new View.OnClickListener() {            
        
                                    @Override            
                                    public void onClick(            
                                            View v            
                                    ) {            
        
                                        String newProduct =            
                                                editInput            
                                                        .getText()            
                                                        .toString()            
                                                        .trim();            
        
        
                                        if (            
                                                newProduct.length()            
                                                == 0            
                                        ) {            
        
                                            Toast.makeText(            
                                                    context,            
                                                    "Enter a product name",            
                                                    Toast.LENGTH_SHORT            
                                            ).show();            
        
                                            return;            
                                        }            
        
        
                                        if (            
                                                newProduct.equalsIgnoreCase(            
                                                        SELECT_PRODUCTS            
                                                )            
                                                ||            
                                                newProduct.equalsIgnoreCase(            
                                                        "Select Product"            
                                                )            
                                        ) {            
        
                                            Toast.makeText(            
                                                    context,            
                                                    "Select Products is not a product",            
                                                    Toast.LENGTH_SHORT            
                                            ).show();            
        
                                            return;            
                                        }            
        
        
                                        /*            
                                         * Same name is allowed            
                                         * when capitalization only            
                                         * changes.            
                                         */            
                                        if (            
                                                !newProduct.equalsIgnoreCase(            
                                                        oldProduct            
                                                )            
                                                &&            
                                                productExists(            
                                                        context,            
                                                        newProduct            
                                                )            
                                        ) {            
        
                                            Toast.makeText(            
                                                    context,            
                                                    "Product already exists",            
                                                    Toast.LENGTH_SHORT            
                                            ).show();            
        
                                            return;            
                                        }            
        
        
                                        ArrayList<String> products =            
                                                getProducts(            
                                                        context            
                                                );            
        
        
                                        for (            
                                                int i = 0;            
                                                i < products.size();            
                                                i++            
                                        ) {            
        
                                            String current =            
                                                    products.get(i);            
        
        
                                            if (            
                                                    current != null            
                                                    &&            
                                                    current.equals(            
                                                            oldProduct            
                                                    )            
                                            ) {            
        
                                                products.set(            
                                                        i,            
                                                        newProduct            
                                                );            
        
                                                break;            
                                            }            
                                        }            
        
        
                                        saveProducts(            
                                                context,            
                                                products            
                                        );            
        
        
                                        editDialog.dismiss();            
        
        
                                        refreshProducts(            
                                                context            
                                        );            
        
        
                                        Toast.makeText(            
                                                context,            
                                                "Product updated",            
                                                Toast.LENGTH_SHORT            
                                        ).show();            
                                    }            
                                }            
                        );            
                    }            
                }            
            }            
    );            
        
        
    editDialog.show();            
}

/*            
 * ============================================================            
 * IMPORT RESULT            
 * ============================================================            
 */            
        
public static boolean handleImportResult(            
        SettingsActivity context,            
        int requestCode,            
        int resultCode,            
        Intent data            
) {            
        
    if (            
            requestCode != IMPORT_TXT_REQUEST            
            &&            
            requestCode != IMPORT_CSV_REQUEST            
            &&            
            requestCode != IMPORT_XLSX_REQUEST            
    ) {            
        
        return false;            
    }            
        
        
    if (            
            resultCode !=            
            android.app.Activity.RESULT_OK            
    ) {            
        
        return true;            
    }            
        
        
    if (            
            data == null            
            ||            
            data.getData() == null            
    ) {            
        
        Toast.makeText(            
                context,            
                "No file selected",            
                Toast.LENGTH_SHORT            
        ).show();            
        
        return true;            
    }            
        
        
    Uri uri =            
            data.getData();            
        
        
    try {            
        
        context.getContentResolver()            
                .takePersistableUriPermission(            
                        uri,            
                        Intent.FLAG_GRANT_READ_URI_PERMISSION            
                );            
        
    } catch (Exception ignored) {            
    }            
        
        
    try {            
        
        int imported;            
        
        
        if (            
                requestCode ==            
                IMPORT_XLSX_REQUEST            
        ) {            
        
            imported =            
                    importXlsx(            
                            context,            
                            uri            
                    );            
        
        } else {            
        
            imported =            
                    importText(            
                            context,            
                            uri,            
                            requestCode ==            
                                    IMPORT_CSV_REQUEST            
                    );            
        }            
        
        
        refreshProducts(            
                context            
        );            
        
        
        Toast.makeText(            
                context,            
                imported            
                        + " product(s) imported",            
                Toast.LENGTH_SHORT            
        ).show();            
        
        
    } catch (Exception e) {            
        
        Toast.makeText(            
                context,            
                "Import failed: "            
                        + errorMessage(e),            
                Toast.LENGTH_LONG            
        ).show();            
    }            
        
        
    return true;            
}            
        
        
/*            
 * ============================================================            
 * TXT / CSV IMPORT            
 * ============================================================            
 */            
        
private static int importText(            
        SettingsActivity context,            
        Uri uri,            
        boolean csv            
) throws Exception {            
        
    InputStream input =            
            context.getContentResolver()            
                    .openInputStream(uri);            
        
        
    if (input == null) {            
        
        throw new Exception(            
                "Cannot open file"            
        );            
    }            
        
        
    BufferedReader reader =            
            new BufferedReader(            
                    new InputStreamReader(            
                            input,            
                            StandardCharsets.UTF_8            
                    )            
            );            
        
        
    int imported = 0;            
        
        
    try {            
        
        String line;            
        
        
        Set<String> importedNames =            
                new HashSet<String>();            
        
        
        while (            
                (line =            
                        reader.readLine())            
                        != null            
        ) {            
        
            line =            
                    line.trim();            
        
        
            if (line.length() == 0) {            
                continue;            
            }            
        
        
            /*            
             * Remove BOM.            
             */            
            if (            
                    line.charAt(0) ==            
                    '\uFEFF'            
            ) {            
        
                line =            
                        line.substring(1)            
                                .trim();            
            }            
        
        
            /*            
             * TXT:            
             *            
             * One product per line.            
             *            
             * CSV:            
             *            
             * Product,Product,Product            
             */            
            String[] values;            
        
        
            if (csv) {            
        
                values =            
                        splitCsvLine(            
                                line            
                        );            
        
            } else {            
        
                values =            
                        new String[] {            
                                line            
                        };            
            }            
        
        
            for (            
                    String value :            
                    values            
            ) {            
        
                if (value == null) {            
                    continue;            
                }            
        
        
                String product =            
                        cleanProduct(            
                                value            
                        );            
        
        
                if (            
                        !validImportedProduct(            
                                product            
                        )            
                ) {            
        
                    continue;            
                }            
        
        
                String key =            
                        product.toLowerCase(            
                                Locale.US            
                        );            
        
        
                if (            
                        importedNames.contains(            
                                key            
                        )            
                ) {            
        
                    continue;            
                }            
        
        
                if (            
                        productExists(            
                                context,            
                                product            
                        )            
                ) {            
        
                    importedNames.add(            
                            key            
                    );            
        
                    continue;            
                }            
        
        
                addProduct(            
                        context,            
                        product            
                );            
        
        
                importedNames.add(            
                        key            
                );            
        
        
                imported++;            
            }            
        }            
        
        
    } finally {            
        
        reader.close();            
    }            
        
        
    return imported;            
}            
        
        
/*            
 * ============================================================            
 * CSV SPLITTER            
 * ============================================================            
 */            
        
private static String[] splitCsvLine(            
        String line            
) {            
        
    ArrayList<String> values =            
            new ArrayList<String>();            
        
        
    StringBuilder current =            
            new StringBuilder();            
        
        
    boolean quoted =            
            false;            
        
        
    for (            
            int i = 0;            
            i < line.length();            
            i++            
    ) {            
        
        char c =            
                line.charAt(i);            
        
        
        if (c == '"') {            
        
            if (            
                    quoted            
                    &&            
                    i + 1 < line.length()            
                    &&            
                    line.charAt(i + 1) == '"'            
            ) {            
        
                current.append(            
                        '"'            
                );            
        
                i++;            
        
            } else {            
        
                quoted =            
                        !quoted;            
            }            
        
        } else if (            
                c == ','            
                &&            
                !quoted            
        ) {            
        
            values.add(            
                    current.toString()            
            );            
        
            current.setLength(            
                    0            
            );            
        
        } else {            
        
            current.append(            
                    c            
            );            
        }            
    }            
        
        
    values.add(            
            current.toString()            
    );            
        
        
    return values.toArray(            
            new String[values.size()]            
    );            
}            
        
        
/*            
 * ============================================================            
 * XLSX IMPORT            
 * ============================================================            
 */            
        
private static int importXlsx(            
        SettingsActivity context,            
        Uri uri            
) throws Exception {            
        
    InputStream input =            
            context.getContentResolver()            
                    .openInputStream(uri);            
        
        
    if (input == null) {            
        
        throw new Exception(            
                "Cannot open XLSX file"            
        );            
    }            
        
        
    java.io.File temp =            
            java.io.File.createTempFile(            
                    "products_",            
                    ".xlsx",            
                    context.getCacheDir()            
            );            
        
        
    java.io.FileOutputStream output =            
            new java.io.FileOutputStream(            
                    temp            
            );            
        
        
    try {            
        
        byte[] buffer =            
                new byte[8192];            
        
        
        int length;            
        
        
        while (            
                (length =            
                        input.read(buffer))            
                        != -1            
        ) {            
        
            output.write(            
                    buffer,            
                    0,            
                    length            
            );            
        }            
        
        
    } finally {            
        
        try {            
            input.close();            
        } catch (Exception ignored) {            
        }            
        
        
        try {            
            output.close();            
        } catch (Exception ignored) {            
        }            
    }            
        
        
    int imported = 0;            
        
        
    ZipFile zip =            
            new ZipFile(            
                    temp            
            );            
        
        
    try {            
        
        ArrayList<String> sharedStrings =            
                readSharedStrings(            
                        zip            
                );            
        
        
        /*            
         * Find worksheet XML files.            
         */            
        java.util.Enumeration<? extends ZipEntry>            
                entries =            
                zip.entries();            
        
        
        while (            
                entries.hasMoreElements()            
        ) {            
        
            ZipEntry entry =            
                    entries.nextElement();            
        
        
            String name =            
                    entry.getName();            
        
        
            if (            
                    !name.startsWith(            
                            "xl/worksheets/"            
                    )            
            ) {            
        
                continue;            
            }            
        
        
            if (            
                    !name.endsWith(            
                            ".xml"            
                    )            
            ) {            
        
                continue;            
            }            
        
        
            InputStream sheet =            
                    zip.getInputStream(            
                            entry            
                    );            
        
        
            try {            
        
                imported +=            
                        readWorksheetProducts(            
                                context,            
                                sheet,            
                                sharedStrings            
                        );            
        
            } finally {            
        
                sheet.close();            
            }            
        }            
        
        
    } finally {            
        
        zip.close();            
        
        
        try {            
            temp.delete();            
        } catch (Exception ignored) {            
        }            
    }            
        
        
    return imported;            
}            
        
        
/*            
 * ============================================================            
 * READ XLSX SHARED STRINGS            
 * ============================================================            
 */            
        
private static ArrayList<String>            
readSharedStrings(            
        ZipFile zip            
) throws Exception {            
        
    ArrayList<String> strings =            
            new ArrayList<String>();            
        
        
    ZipEntry entry =            
            zip.getEntry(            
                    "xl/sharedStrings.xml"            
            );            
        
        
    if (entry == null) {            
        return strings;            
    }            
        
        
    InputStream input =            
            zip.getInputStream(            
                    entry            
            );            
        
        
    XmlPullParserFactory factory =            
            XmlPullParserFactory            
                    .newInstance();            
        
        
    XmlPullParser parser =            
            factory.newPullParser();            
        
        
    parser.setInput(            
            input,            
            "UTF-8"            
    );            
        
        
    StringBuilder current =            
            null;            
        
        
    int event;            
        
        
    while (            
            (event =            
                    parser.next())            
                    != XmlPullParser.END_DOCUMENT            
    ) {            
        
        if (            
                event ==            
                XmlPullParser.START_TAG            
        ) {            
        
            String tag =            
                    parser.getName();            
        
        
            if (            
                    "si".equals(tag)            
            ) {            
        
                current =            
                        new StringBuilder();            
        
            } else if (            
                    "t".equals(tag)            
                    &&            
                    current != null            
            ) {            
        
                current.append(            
                        parser.nextText()            
                );            
            }            
        
        
        } else if (            
                event ==            
                XmlPullParser.END_TAG            
        ) {            
        
            if (            
                    "si".equals(            
                            parser.getName()            
                    )            
                    &&            
                    current != null            
            ) {            
        
                strings.add(            
                        current.toString()            
                );            
        
                current =            
                        null;            
            }            
        }            
    }            
        
        
    input.close();            
        
        
    return strings;            
}            
        
        
/*            
 * ============================================================            
 * READ XLSX WORKSHEET            
 * ============================================================            
 */            
        
private static int readWorksheetProducts(            
        SettingsActivity context,            
        InputStream input,            
        ArrayList<String> sharedStrings            
) throws Exception {            
        
    XmlPullParserFactory factory =            
            XmlPullParserFactory            
                    .newInstance();            
        
        
    XmlPullParser parser =            
            factory.newPullParser();            
        
        
    parser.setInput(            
            input,            
            "UTF-8"            
    );            
        
        
    int imported = 0;            
        
        
    String cellType =            
            null;            
        
        
    String cellValue =            
            null;            
        
        
    boolean inValue =            
            false;            
        
        
    int event;            
        
        
    while (            
            (event =            
                    parser.next())            
                    != XmlPullParser.END_DOCUMENT            
    ) {            
        
        if (            
                event ==            
                XmlPullParser.START_TAG            
        ) {            
        
            String tag =            
                    parser.getName();            
        
        
            if (            
                    "c".equals(tag)            
            ) {            
        
                cellType =            
                        parser.getAttributeValue(            
                                null,            
                                "t"            
                        );            
        
                cellValue =            
                        null;            
            }            
        
        
            if (            
                    "v".equals(tag)            
            ) {            
        
                inValue =            
                        true;            
            }            
        
        
            if (            
                    "t".equals(tag)            
                    &&            
                    "inlineStr".equals(            
                            cellType            
                    )            
            ) {            
        
                cellValue =            
                        parser.nextText();            
        
        
                addXlsxProductIfValid(            
                        context,            
                        cellValue            
                );            
            }            
        
        
        } else if (            
                event ==            
                XmlPullParser.TEXT            
        ) {            
        
            if (inValue) {            
        
                cellValue =            
                        parser.getText();            
            }            
        
        
        } else if (            
                event ==            
                XmlPullParser.END_TAG            
        ) {            
        
            String tag =            
                    parser.getName();            
        
        
            if (            
                    "v".equals(tag)            
            ) {            
        
                inValue =            
                        false;            
        
        
                if (            
                        cellValue != null            
                ) {            
        
                    String product =            
                            cellValue.trim();            
        
        
                    /*            
                     * Shared string index.            
                     */            
                    if (            
                            "s".equals(            
                                    cellType            
                            )            
                    ) {            
        
                        try {            
        
                            int index =            
                                    Integer.parseInt(            
                                            product            
                                    );            
        
        
                            if (            
                                    index >= 0            
                                    &&            
                                    index <            
                                    sharedStrings.size()            
                            ) {            
        
                                product =            
                                        sharedStrings.get(            
                                                index            
                                        );            
                            }            
        
                        } catch (            
                                NumberFormatException ignored            
                        ) {            
                        }            
                    }            
        
        
                    /*            
                     * We cannot directly count            
                     * here because add method            
                     * returns boolean.            
                     */            
                    if (            
                            addXlsxProductIfValid(            
                                    context,            
                                    product            
                            )            
                    ) {            
        
                        imported++;            
                    }            
                }            
            }            
        
        
            if (            
                    "c".equals(tag)            
            ) {            
        
                cellType =            
                        null;            
        
                cellValue =            
                        null;            
            }            
        }            
    }            
        
        
    return imported;            
}            
        
        
/*            
 * ============================================================            
 * ADD XLSX PRODUCT            
 * ============================================================            
 */            
        
private static boolean            
addXlsxProductIfValid(            
        SettingsActivity context,            
        String value            
) {            
        
    String product =            
            cleanProduct(            
                    value            
            );            
        
        
    if (            
            !validImportedProduct(            
                    product            
            )            
    ) {            
        
        return false;            
    }            
        
        
    if (            
            productExists(            
                    context,            
                    product            
            )            
    ) {            
        
        return false;            
    }            
        
        
    addProduct(            
            context,            
            product            
    );            
        
        
    return true;            
}            
        
        
/*            
 * ============================================================            
 * CLEAN PRODUCT            
 * ============================================================            
 */            
        
private static String cleanProduct(            
        String product            
) {            
        
    if (product == null) {            
        return "";            
    }            
        
        
    product =            
            product.trim();            
        
        
    if (            
            product.length() >= 2            
            &&            
            product.startsWith("\"")            
            &&            
            product.endsWith("\"")            
    ) {            
        
        product =            
                product.substring(            
                        1,            
                        product.length() - 1            
                ).trim();            
    }            
        
        
    return product;            
}            
        
        
/*            
 * ============================================================            
 * VALIDATE IMPORTED PRODUCT            
 * ============================================================            
 */            
        
private static boolean            
validImportedProduct(            
        String product            
) {            
        
    if (            
            product == null            
            ||            
            product.length() == 0            
    ) {            
        
        return false;            
    }            
        
        
    /*            
     * Never import placeholder.            
     */            
    if (            
            product.equalsIgnoreCase(            
                    SELECT_PRODUCTS            
            )            
            ||            
            product.equalsIgnoreCase(            
                    "Select Product"            
            )            
    ) {            
        
        return false;            
    }            
        
        
    /*            
     * Ignore common column headers.            
     */            
    String lower =            
            product.toLowerCase(            
                    Locale.US            
            );            
        
        
    if (            
            lower.equals("product")            
            ||            
            lower.equals("products")            
            ||            
            lower.equals("product name")            
            ||            
            lower.equals("productname")            
            ||            
            lower.equals("name")            
    ) {            
        
        return false;            
    }            
        
        
    return true;            
}            
        
        
/*            
 * ============================================================            
 * REFRESH PRODUCT LIST            
 * ============================================================            
 */            
        
private static void refreshProducts(            
        final SettingsActivity context            
) {            
        
    if (productContainer == null) {            
        return;            
    }            
        
        
    productContainer.removeAllViews();            
        
        
    ArrayList<String> products =            
            getProducts(            
                    context            
            );            
        
        
    ArrayList<String> realProducts =            
            new ArrayList<String>();            
        
        
    /*            
     * IMPORTANT:            
     *            
     * Select Products is removed            
     * from the real product list.            
     */            
    for (            
            String product :            
            products            
    ) {            
        
        if (product == null) {            
            continue;            
        }            
        
        
        product =            
                product.trim();            
        
        
        if (product.length() == 0) {            
            continue;            
        }            
        
        
        if (            
                product.equalsIgnoreCase(            
                        SELECT_PRODUCTS            
                )            
                ||            
                product.equalsIgnoreCase(            
                        "Select Product"            
                )            
        ) {            
        
            continue;            
        }            
        
        
        boolean duplicate =            
                false;            
        
        
        for (            
                String existing :            
                realProducts            
        ) {            
        
            if (            
                    existing.equalsIgnoreCase(            
                            product            
                    )            
            ) {            
        
                duplicate =            
                        true;            
        
                break;            
            }            
        }            
        
        
        if (!duplicate) {            
        
            realProducts.add(            
                    product            
            );            
        }            
    }            
        
        
    /*            
     * NO PRODUCTS            
     */            
    if (            
            realProducts.size() == 0            
    ) {            
        
        TextView empty =            
                new TextView(context);            
        
        empty.setText(            
                "No products added"            
        );            
        
        empty.setTextSize(            
                16            
        );            
        
        empty.setTextColor(            
                Color.rgb(            
                        100,            
                        100,            
                        100            
                )            
        );            
        
        empty.setGravity(            
                Gravity.CENTER            
        );            
        
        empty.setPadding(            
                0,            
                dp(context, 30),            
                0,            
                dp(context, 30)            
        );            
        
        
        productContainer.addView(            
                empty            
        );            
        
        
        /*            
         * Save ONLY real products.            
         *            
         * Do NOT save Select Products.            
         */            
        saveProducts(            
                context,            
                new ArrayList<String>()            
        );            
        
        
        return;            
    }            
        
        
    /*            
     * DISPLAY PRODUCTS            
     */            
    for (            
            final String product :            
            realProducts            
    ) {            
        
        addProductRow(            
                context,            
                product            
        );            
    }            
}

/*

============================================================

DRAG THUMB

============================================================
*/
dragThumb.setOnTouchListener(
new View.OnTouchListener() {

private float lastY;            

     private boolean dragging = false;            


     @Override            
     public boolean onTouch(            
             View v,            
             MotionEvent event            
     ) {            

         switch (            
                 event.getActionMasked()            
         ) {            


             /*            
              * ====================================================            
              * START DRAG            
              * ====================================================            
              */            
             case MotionEvent.ACTION_DOWN:            

                 lastY =            
                         event.getRawY();            

                 dragging =            
                         true;            


                 /*            
                  * Prevent the ScrollView from taking            
                  * control of the finger movement.            
                  */            
                 if (v.getParent() != null) {            

                     v.getParent()            
                             .requestDisallowInterceptTouchEvent(            
                                     true            
                             );            
                 }            


                 /*            
                  * Visual indication that            
                  * the row is being dragged.            
                  */            
                 row.setElevation(            
                         dp(            
                                 context,            
                                 8            
                         )            
                 );            


                 dragThumb.setBackground(            
                         background(            
                                 COLOR_ACCENT,            
                                 dp(            
                                         context,            
                                         9            
                                 )            
                         )            
                 );            


                 /*            
                  * Keep receiving all touch events.            
                  */            
                 return true;            


             /*            
              * ====================================================            
              * MOVE DRAG            
              * ====================================================            
              */            
             case MotionEvent.ACTION_MOVE:            

                 if (!dragging) {            

                     return true;            
                 }            


                 float currentY =            
                         event.getRawY();            


                 /*            
                  * Only move when the finger            
                  * has actually changed position.            
                  */            
                 if (            
                         currentY != lastY            
                 ) {            

                     moveRowWhileDragging(            
                             row,            
                             currentY            
                     );            


                     lastY =            
                             currentY;            
                 }            


                 return true;            


             /*            
              * ====================================================            
              * FINISH DRAG            
              * ====================================================            
              */            
             case MotionEvent.ACTION_UP:            

             case MotionEvent.ACTION_CANCEL:            

                 if (dragging) {            

                     dragging =            
                             false;            


                     /*            
                      * Restore row appearance.            
                      */            
                     row.setElevation(            
                             0            
                     );            


                     /*            
                      * Restore drag thumb.            
                      */            
                     dragThumb.setBackground(            
                             background(            
                                     COLOR_LIGHT,            
                                     dp(            
                                             context,            
                                             9            
                                     )            
                             )            
                     );            


                     /*            
                      * Allow ScrollView to receive            
                      * touch events again.            
                      */            
                     if (            
                             v.getParent() != null            
                     ) {            

                         v.getParent()            
                                 .requestDisallowInterceptTouchEvent(            
                                         false            
                                 );            
                     }            


                     /*            
                      * Save the new product order.            
                      */            
                     saveCurrentProductOrder(            
                             context            
                     );            
                 }            


                 return true;            
         }            


         return true;            
     }            
 }


);

}

/*

============================================================

MOVE ROW WHILE DRAGGING

============================================================
*/
private static void moveRowWhileDragging(
View row,
float rawY
) {

if (
productContainer == null
||
row == null
) {

return;

}

int currentIndex =
productContainer.indexOfChild(
row
);

if (currentIndex < 0) {

return;

}

int childCount =
productContainer.getChildCount();

if (childCount <= 1) {

return;

}

int targetIndex =
currentIndex;

for (
int i = 0;
i < childCount;
i++
) {

View child =            
         productContainer.getChildAt(            
                 i            
         );            


 if (child == null) {            

     continue;            
 }            


 if (child == row) {            

     continue;            
 }            


 /*            
  * IMPORTANT:            
  *            
  * getLocationOnScreen() requires            
  * int[] — NOT float[].            
  */            
 int[] location =            
         new int[2];            


 child.getLocationOnScreen(            
         location            
 );            


 float childTop =            
         (float) location[1];            


 float childCenter =            
         childTop            
         +            
         (            
                 child.getHeight()            
                 /            
                 2f            
         );            


 /*            
  * Finger is above this row's center.            
  */            
 if (            
         rawY < childCenter            
 ) {            

     targetIndex =            
             i;            

     break;            
 }            


 /*            
  * Finger is below this row's center.            
  */            
 targetIndex =            
         i + 1;

}

/*

Remove the dragged row from

its current position first.
*/
if (
targetIndex != currentIndex
) {

productContainer.removeView(
row
);

/*

After removing the row,

the child count is smaller.
*/
int newChildCount =
productContainer.getChildCount();


/*

Correct the target position

after removal.
*/
if (
targetIndex > newChildCount
) {

targetIndex =
newChildCount;
}


if (
targetIndex < 0
) {

targetIndex =            
         0;

}

/*

Insert the row at the

new position.
*/
productContainer.addView(
row,
targetIndex
);


row.requestLayout();
}
}


/*

============================================================

SAVE CURRENT DRAG ORDER

============================================================
*/


private static void saveCurrentProductOrder(
SettingsActivity context
) {

if (            
         context == null            
         ||            
         productContainer == null            
 ) {            

     return;            
 }            


 ArrayList<String> orderedProducts =            
         new ArrayList<String>();            


 for (            
         int i = 0;            
         i < productContainer.getChildCount();            
         i++            
 ) {            

     View child =            
             productContainer.getChildAt(            
                     i            
             );            


     if (!(child instanceof LinearLayout)) {            
         continue;            
     }            


     LinearLayout row =            
             (LinearLayout) child;            


     /*            
      * Product name is the second            
      * child after the drag thumb.            
      */            
     if (row.getChildCount() < 2) {            
         continue;            
     }            


     View nameView =            
             row.getChildAt(            
                     1            
             );            


     if (!(nameView instanceof TextView)) {            
         continue;            
     }            


     String product =            
             ((TextView) nameView)            
                     .getText()            
                     .toString()            
                     .trim();            


     if (            
             product.length() == 0            
             ||            
             product.equalsIgnoreCase(            
                     SELECT_PRODUCTS            
             )            
             ||            
             product.equalsIgnoreCase(            
                     "Select Product"            
             )            
     ) {            

         continue;            
     }            


     orderedProducts.add(            
             product            
     );            
 }            


 saveProducts(            
         context,            
         orderedProducts            
 );

}

/*

============================================================

DELETE

============================================================
*/


private static void deleteProduct(
SettingsActivity context,
String product
) {

if (product == null) {            
     return;            
 }            


 product =            
         product.trim();            


 if (product.length() == 0) {            
     return;            
 }            


 if (            
         product.equalsIgnoreCase(            
                 SELECT_PRODUCTS            
         )            
         ||            
         product.equalsIgnoreCase(            
                 "Select Product"            
         )            
 ) {            

     Toast.makeText(            
             context,            
             "Select Products is not a product",            
             Toast.LENGTH_SHORT            
     ).show();            

     return;            
 }            


 removeProduct(            
         context,            
         product            
 );            


 refreshProducts(            
         context            
 );            


 Toast.makeText(            
         context,            
         "Product deleted",            
         Toast.LENGTH_SHORT            
 ).show();

}

}

/*

============================================================

ERROR MESSAGE

============================================================
*/


private static String errorMessage(
Exception e
) {

if (e == null) {            
     return "Unknown error";            
 }            


 String message =            
         e.getMessage();            


 if (            
         message == null            
         ||            
         message.trim().length() == 0            
 ) {            

     return "Unknown error";            
 }            


 return message;

}
