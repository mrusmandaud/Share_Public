    /*
     * SAVE RECORD
     */
    public static boolean saveRecord(
            Activity activity,
            EditText nameView,
            EditText phoneView,
            EditText addressView,
            Spinner productSpinner,
            CheckBox confirm
    ) {

        if (
                confirm == null
                        || !confirm.isChecked()
        ) {

            Toast.makeText(
                    activity,
                    "Please confirm and agree to save",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        String name = "";

        if (
                nameView != null
                        && nameView.getText() != null
        ) {

            name =
                    nameView
                            .getText()
                            .toString()
                            .trim();
        }


        String phone = "";

        if (
                phoneView != null
                        && phoneView.getText() != null
        ) {

            phone =
                    phoneView
                            .getText()
                            .toString()
                            .trim();
        }


        String address = "";

        if (
                addressView != null
                        && addressView.getText() != null
        ) {

            address =
                    addressView
                            .getText()
                            .toString()
                            .trim();
        }


        String product = "";

        if (
                productSpinner != null
                        && productSpinner
                        .getSelectedItem() != null
        ) {

            product =
                    productSpinner
                            .getSelectedItem()
                            .toString()
                            .trim();
        }


        if (name.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Name",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (phone.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Phone",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (address.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Address",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (
                product.length() == 0
                        || product.equalsIgnoreCase(
                        "Select Product"
                )
        ) {

            Toast.makeText(
                    activity,
                    "Please select Product",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        String record =
                name
                + " | "
                + phone
                + " | "
                + address
                + " | "
                + product
                + "\n";


        FileOutputStream output =
                null;


        try {

            output =
                    activity.openFileOutput(
                            RECORDS_FILE,
                            Context.MODE_APPEND
                    );

            output.write(
                    record.getBytes(
                            "UTF-8"
                    )
            );

            output.flush();


            Toast.makeText(
                    activity,
                    "Record saved successfully",
                    Toast.LENGTH_SHORT
            ).show();


            clearRecord(
                    nameView,
                    phoneView,
                    addressView,
                    productSpinner,
                    confirm
            );


            return true;


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to save record",
                    Toast.LENGTH_SHORT
            ).show();

            return false;


        } finally {

            if (output != null) {

                try {

                    output.close();

                } catch (Exception ignored) {
                }
            }
        }
    }


    /*
     * CLEAR RECORD
     */
    public static void clearRecord(
            EditText nameView,
            EditText phoneView,
            EditText addressView,
            Spinner productSpinner,
            CheckBox confirm
    ) {

        if (nameView != null) {
            nameView.setText("");
        }


        if (phoneView != null) {
            phoneView.setText("");
        }


        if (addressView != null) {
            addressView.setText("");
        }


        if (
                productSpinner != null
                        && productSpinner.getAdapter() != null
                        && productSpinner
                        .getAdapter()
                        .getCount() > 0
        ) {

            productSpinner.setSelection(
                    0
            );
        }


        if (confirm != null) {
            confirm.setChecked(false);
        }
    }


    /*
     * GET RECORDS
     */
    public static ArrayList<String> getRecords(
            Activity activity
    ) {

        ArrayList<String> records =
                new ArrayList<String>();

        BufferedReader reader =
                null;


        try {

            FileInputStream input =
                    activity.openFileInput(
                            RECORDS_FILE
                    );

            reader =
                    new BufferedReader(
                            new InputStreamReader(
                                    input,
                                    "UTF-8"
                            )
                    );


            String line;


            while (
                    (line = reader.readLine())
                            != null
            ) {

                line =
                        line.trim();


                if (
                        line.length() > 0
                ) {

                    records.add(
                            line
                    );
                }
            }


        } catch (Exception e) {


        } finally {

            if (reader != null) {

                try {

                    reader.close();

                } catch (Exception ignored) {
                }
            }
        }


        return records;
    }

    /*
     * GET PRODUCTS
     */
    public static ArrayList<String> getProducts(
            Activity activity
    ) {

        ArrayList<String> products =
                new ArrayList<String>();


        SharedPreferences preferences =
                activity.getSharedPreferences(
                        PRODUCTS_PREF,
                        Context.MODE_PRIVATE
                );


        String saved =
                preferences.getString(
                        PRODUCTS_KEY,
                        ""
                );


        if (
                saved != null
                        && saved.trim().length() > 0
        ) {

            String[] values =
                    saved.split(
                            "\\|",
                            -1
                    );


            for (
                    String value :
                    values
            ) {

                value =
                        value.trim();


                if (
                        value.length() > 0
                ) {

                    products.add(
                            value
                    );
                }
            }
        }


        if (
                products.size() == 0
        ) {

            products.add(
                    "Select Product"
            );
        }


        return products;
    }


    /*
     * SAVE PRODUCTS
     */
    public static void saveProducts(
            Activity activity,
            ArrayList<String> products
    ) {

        StringBuilder builder =
                new StringBuilder();


        if (products != null) {

            for (
                    String product :
                    products
            ) {

                if (product == null) {
                    continue;
                }


                product =
                        product.trim();


                if (
                        product.length() == 0
                ) {
                    continue;
                }


                if (
                        builder.length() > 0
                ) {

                    builder.append(
                            "|"
                    );
                }


                builder.append(
                        product
                );
            }
        }


        activity.getSharedPreferences(
                PRODUCTS_PREF,
                Context.MODE_PRIVATE
        )
        .edit()
        .putString(
                PRODUCTS_KEY,
                builder.toString()
        )
        .apply();
    }


    /*
     * ADD PRODUCT
     */
    public static void addProduct(
            Activity activity,
            String product
    ) {

        if (product == null) {
            return;
        }


        product =
                product.trim();


        if (
                product.length() == 0
        ) {
            return;
        }


        ArrayList<String> products =
                getProducts(
                        activity
                );


        if (
                products.size() == 1
                        && products
                        .get(0)
                        .equals(
                                "Select Product"
                        )
        ) {

            products.clear();
        }


        if (
                !products.contains(
                        product
                )
        ) {

            products.add(
                    product
            );
        }


        saveProducts(
                activity,
                products
        );
    }


    /*
     * REMOVE PRODUCT
     */
    public static void removeProduct(
            Activity activity,
            String product
    ) {

        if (product == null) {
            return;
        }


        ArrayList<String> products =
                getProducts(
                        activity
                );


        products.remove(
                product
        );


        saveProducts(
                activity,
                products
        );
    }


    /*
     * RECORD FILE EXISTS
     */
    public static boolean recordsFileExists(
            Activity activity
    ) {

        try {

            FileInputStream input =
                    activity.openFileInput(
                            RECORDS_FILE
                    );

            input.close();

            return true;


        } catch (Exception e) {

            return false;
        }
    }


    /*
     * GET FILE URI
     */
    public static Uri getFileUri(
            Activity activity
    ) {

        try {

            return Uri.parse(
                    activity
                            .getFilesDir()
                            .getAbsolutePath()
            );

        } catch (Exception e) {

            return null;
        }
    }
