findViewById(R.id.settingsBackupCard).setOnClickListener(
    new View.OnClickListener() {
        @Override
        public void onClick(View v) {

            getSharedPreferences(
                "settings_state",
                MODE_PRIVATE
            )
            .edit()
            .putString(
                "backup_action",
                "backup"
            )
            .putBoolean(
                "backup_result_ready",
                false
            )
            .remove(
                "backup_result_uri"
            )
            .remove(
                "backup_result_code"
            )
            .apply();

            android.content.Intent intent =
                new android.content.Intent(
                    android.content.Intent.ACTION_CREATE_DOCUMENT
                );

            intent.addCategory(
                android.content.Intent.CATEGORY_OPENABLE
            );

            intent.setType("text/plain");

            intent.putExtra(
                android.content.Intent.EXTRA_TITLE,
                "myrecords_backup.txt"
            );

            startActivityForResult(
                intent,
                7001
            );
        }
    }
);

findViewById(R.id.settingsExportCard).setOnClickListener(
    new View.OnClickListener() {
        @Override
        public void onClick(View v) {

            final android.content.SharedPreferences prefs =
                getSharedPreferences(
                    "settings_state",
                    MODE_PRIVATE
                );

            final String[] formats = {
                "TXT",
                "CSV",
                "XLSX"
            };

            new android.app.AlertDialog.Builder(
                SettingsActivity.this
            )
            .setTitle(
                "Select Export Format"
            )
            .setItems(
                formats,
                new android.content.DialogInterface.OnClickListener() {

                    @Override
                    public void onClick(
                        android.content.DialogInterface dialog,
                        int which
                    ) {

                        String selectedFormat =
                            formats[which];

                        java.text.SimpleDateFormat dateFormat =
                            new java.text.SimpleDateFormat(
                                "yyyyMMdd",
                                java.util.Locale.US
                            );

                        String date =
                            dateFormat.format(
                                new java.util.Date()
                            );

                        int serial =
                            prefs.getInt(
                                "export_serial",
                                1
                            );

                        String extension =
                            selectedFormat.toLowerCase(
                                java.util.Locale.US
                            );

                        String filename =
                            "SaveRecords_"
                            + date
                            + "_"
                            + serial
                            + "."
                            + extension;

                        prefs.edit()
                        .putString(
                            "export_format",
                            selectedFormat
                        )
                        .putString(
                            "export_filename",
                            filename
                        )
                        .putBoolean(
                            "export_result_ready",
                            false
                        )
                        .remove(
                            "export_result_uri"
                        )
                        .apply();

                        android.content.Intent intent =
                            new android.content.Intent(
                                android.content.Intent.ACTION_CREATE_DOCUMENT
                            );

                        intent.addCategory(
                            android.content.Intent.CATEGORY_OPENABLE
                        );

                        if (
                            selectedFormat.equals("TXT")
                        ) {

                            intent.setType(
                                "text/plain"
                            );

                        } else if (
                            selectedFormat.equals("CSV")
                        ) {

                            intent.setType(
                                "text/csv"
                            );

                        } else {

                            intent.setType(
                                "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
                            );
                        }

                        intent.putExtra(
                            android.content.Intent.EXTRA_TITLE,
                            filename
                        );

                        startActivityForResult(
                            intent,
                            8001
                        );
                    }
                }
            )
            .setNegativeButton(
                "CANCEL",
                null
            )
            .show();
        }
    }
);
