@Override
protected void onCreate(Bundle savedInstanceState) {
    super.onCreate(savedInstanceState);

    setContentView(R.layout.settings);

    // 1. BACK
    findViewById(R.id.settingsBack).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                ButtonsController.back(SettingsActivity.this);
            }
        }
    );

    // 2. APPEARANCE
    findViewById(R.id.settingsAppearanceCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                Intent intent = new Intent(
                    SettingsActivity.this,
                    AppearanceActivity.class
                );
                startActivity(intent);
            }
        }
    );

    // 3. THEME
    findViewById(R.id.settingsThemeCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                Intent intent = new Intent(
                    SettingsActivity.this,
                    ThemeActivity.class
                );
                startActivity(intent);
            }
        }
    );

    // 4. PRODUCTS
    findViewById(R.id.settingsProductsCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                Intent intent = new Intent(
                    SettingsActivity.this,
                    ProductManagement.class
                );
                startActivity(intent);
            }
        }
    );

    // 5. BACKUP
    findViewById(R.id.settingsBackupCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {

                getSharedPreferences(
                    "settings_state",
                    MODE_PRIVATE
                )
                .edit()
                .putString(
                    "backup_action",
                    "backup"
                )
                .putBoolean(
                    "backup_result_ready",
                    false
                )
                .remove(
                    "backup_result_uri"
                )
                .remove(
                    "backup_result_code"
                )
                .apply();

                BACKUP();
            }
        }
    );

    // 6. RESTORE
    findViewById(R.id.settingsRestoreCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {

                getSharedPreferences(
                    "settings_state",
                    MODE_PRIVATE
                )
                .edit()
                .putString(
                    "backup_action",
                    "restore"
                )
                .putBoolean(
                    "backup_result_ready",
                    false
                )
                .remove(
                    "backup_result_uri"
                )
                .remove(
                    "backup_result_code"
                )
                .apply();

                BACKUP();
            }
        }
    );

    // 7. EXPORT
    findViewById(R.id.settingsExportCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {

                getSharedPreferences(
                    "settings_state",
                    MODE_PRIVATE
                )
                .edit()
                .putBoolean(
                    "export_result_ready",
                    false
                )
                .remove(
                    "export_result_uri"
                )
                .remove(
                    "export_format"
                )
                .remove(
                    "export_filename"
                )
                .apply();

                EXPORT();
            }
        }
    );

    // 8. CONTACT US
    findViewById(R.id.settingsContactCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                setContentView(
                    R.layout.contactus
                );
            }
        }
    );

    // 9. ABOUT
    findViewById(R.id.settingsAboutCard).setOnClickListener(
        new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                setContentView(
                    R.layout.about
                );
            }
        }
    );
}
