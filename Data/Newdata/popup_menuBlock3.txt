// ==========================================================
// popup_menu
// ALL UI CREATED IN JAVA
// NO POPUP XML
// ==========================================================


// ==========================================================
// FIND TABLE WEBVIEW
// ==========================================================

android.webkit.WebView popupTableView = null;

for (
        int i = 0;
        i < binding.recordsContainer.getChildCount();
        i++
) {

    android.view.View child =
            binding.recordsContainer.getChildAt(i);

    if (
            child instanceof android.webkit.WebView
    ) {

        popupTableView =
                (android.webkit.WebView) child;

        break;
    }
}


if (popupTableView == null) {
    return;
}


final android.webkit.WebView finalPopupTableView =
        popupTableView;


// ==========================================================
// POPUP ROOT
// ==========================================================

final android.widget.LinearLayout popupLayout =
        new android.widget.LinearLayout(
                MyrecordsActivity.this
        );

popupLayout.setOrientation(
        android.widget.LinearLayout.VERTICAL
);


// ==========================================================
// UI PROPERTIES
// ==========================================================

int popupWidth =
        (int) (
                190 *
                getResources()
                        .getDisplayMetrics()
                        .density
        );

int buttonHeight =
        (int) (
                48 *
                getResources()
                        .getDisplayMetrics()
                        .density
        );

int horizontalPadding =
        (int) (
                16 *
                getResources()
                        .getDisplayMetrics()
                        .density
        );

int verticalPadding =
        (int) (
                4 *
                getResources()
                        .getDisplayMetrics()
                        .density
        );

int cornerRadius =
        (int) (
                12 *
                getResources()
                        .getDisplayMetrics()
                        .density
        );


// ==========================================================
// POPUP BACKGROUND
// ==========================================================

android.graphics.drawable.GradientDrawable popupBackground =
        new android.graphics.drawable.GradientDrawable();

popupBackground.setColor(
        android.graphics.Color.WHITE
);

popupBackground.setCornerRadius(
        cornerRadius
);

popupBackground.setStroke(
        1,
        android.graphics.Color.LTGRAY
);

popupLayout.setBackground(
        popupBackground
);

popupLayout.setPadding(
        4,
        4,
        4,
        4
);


// ==========================================================
// CREATE BUTTON FUNCTION
// ==========================================================

java.util.function.Function<String, android.widget.Button>
        createPopupButton =
        new java.util.function.Function<
                String,
                android.widget.Button
        >() {

            @Override
            public android.widget.Button apply(
                    String text
            ) {

                android.widget.Button button =
                        new android.widget.Button(
                                MyrecordsActivity.this
                        );

                button.setText(text);

                button.setTextSize(
                        15
                );

                button.setTextColor(
                        android.graphics.Color.BLACK
                );

                button.setGravity(
                        android.view.Gravity.CENTER_VERTICAL |
                        android.view.Gravity.LEFT
                );

                button.setAllCaps(false);

                button.setPadding(
                        horizontalPadding,
                        verticalPadding,
                        horizontalPadding,
                        verticalPadding
                );

                button.setMinHeight(0);

                button.setMinWidth(0);

                button.setBackgroundColor(
                        android.graphics.Color.TRANSPARENT
                );

                button.setLayoutParams(
                        new android.widget.LinearLayout.LayoutParams(
                                -1,
                                buttonHeight
                        )
                );

                return button;
            }
        };


// ==========================================================
// CREATE POPUP BUTTONS
// ==========================================================

final android.widget.Button copyButton =
        createPopupButton.apply("Copy");

final android.widget.Button shareButton =
        createPopupButton.apply("Share");

final android.widget.Button exportButton =
        createPopupButton.apply("Export");

final android.widget.Button selectAllButton =
        createPopupButton.apply("Select All");

final android.widget.Button deselectAllButton =
        createPopupButton.apply("Deselect All");


// ==========================================================
// ADD BUTTONS
// ==========================================================

popupLayout.addView(
        copyButton
);

popupLayout.addView(
        shareButton
);

popupLayout.addView(
        exportButton
);

popupLayout.addView(
        selectAllButton
);

popupLayout.addView(
        deselectAllButton
);


// ==========================================================
// POPUP WINDOW
// ==========================================================

final android.widget.PopupWindow popupWindow =
        new android.widget.PopupWindow(
                popupLayout,
                popupWidth,
                android.view.ViewGroup.LayoutParams.WRAP_CONTENT,
                true
        );

popupWindow.setBackgroundDrawable(
        popupBackground
);

popupWindow.setOutsideTouchable(true);

popupWindow.setFocusable(true);

popupWindow.setElevation(
        8 *
        getResources()
                .getDisplayMetrics()
                .density
);


// ==========================================================
// COPY
// ==========================================================

copyButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(android.view.View v) {

                ButtonsController.copy(
                        MyrecordsActivity.this,
                        finalPopupTableView,
                        popupWindow
                );
            }
        }
);


// ==========================================================
// SHARE
// ==========================================================

shareButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(android.view.View v) {

                ButtonsController.share(
                        MyrecordsActivity.this,
                        finalPopupTableView,
                        popupWindow
                );
            }
        }
);


// ==========================================================
// EXPORT
// ==========================================================

exportButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(android.view.View v) {

                ButtonsController.export(
                        MyrecordsActivity.this
                );

                popupWindow.dismiss();
            }
        }
);


// ==========================================================
// SELECT ALL
// ==========================================================

selectAllButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(android.view.View v) {

                ButtonsController.selectAll(
                        finalPopupTableView
                );

                popupWindow.dismiss();
            }
        }
);


// ==========================================================
// DESELECT ALL
// ==========================================================

deselectAllButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(android.view.View v) {

                ButtonsController.deselectAll(
                        finalPopupTableView
                );

                popupWindow.dismiss();
            }
        }
);


// ==========================================================
// SHOW POPUP
// ==========================================================

popupWindow.showAsDropDown(
        finalPopupTableView,
        0,
        0
);
