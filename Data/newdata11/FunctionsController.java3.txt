    // ============================================================
    // EDIT SESSION STATE
    // ============================================================

    private static final Map<Activity, EditSession>
            EDIT_SESSIONS =
            new WeakHashMap<Activity, EditSession>();


    // ============================================================
    // EDIT MENU ACTION CALLBACKS
    // ============================================================

    public interface EditMenuActions {

        void onClear();

        void onDelete();

        void onSave();

        void onClose();
    }


    private static final Map<Activity, EditMenuActions>
            EDIT_MENU_ACTIONS =
            new WeakHashMap<Activity, EditMenuActions>();


    public static void setEditMenuActions(
            Activity activity,
            EditMenuActions actions
    ) {

        if (activity == null) {
            return;
        }

        if (actions == null) {

            EDIT_MENU_ACTIONS.remove(
                    activity
            );

        } else {

            EDIT_MENU_ACTIONS.put(
                    activity,
                    actions
            );
        }
    }


    private static EditMenuActions getEditMenuActions(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }

        return EDIT_MENU_ACTIONS.get(
                activity
        );
    }

    // ============================================================
    // EDIT SESSION
    // ============================================================

    private static class EditSession {

        boolean editMode = false;

        final Set<Integer> deletedRows =
                new HashSet<Integer>();

        final Set<TextView> removedCells =
                new HashSet<TextView>();

        final ArrayList<String> originalRecords =
                new ArrayList<String>();

        boolean menuConnected = false;

        TextView focusedCell = null;

        final Set<String> preEditSelectedCells =
                new HashSet<String>();

        final IdentityHashMap<View, Integer>
                originalRowHeights =
                new IdentityHashMap<View, Integer>();

        final IdentityHashMap<View, Float>
                originalRowWeights =
                new IdentityHashMap<View, Float>();

        boolean rowsFrozen = false;

        View editPopupWindow = null;
    }


    // ============================================================
    // GET EDIT SESSION
    // ============================================================

    private static EditSession getEditSession(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }

        EditSession session =
                EDIT_SESSIONS.get(
                        activity
                );

        if (session == null) {

            session =
                    new EditSession();

            EDIT_SESSIONS.put(
                    activity,
                    session
            );
        }

        return session;
    }


    // ============================================================
    // LOCAL SELECTION STATE
    //
    // IMPORTANT:
    // Selection itself remains in ButtonsController.
    //
    // This small wrapper only gives Edit Mode access to the current
    // records stored on recordsContainer.
    // ============================================================

    private static class SelectionState {

        final ArrayList<ArrayList<TextView>>
                records =
                new ArrayList<ArrayList<TextView>>();

        final ArrayList<TextView>
                cells =
                new ArrayList<TextView>();
    }


    private static SelectionState getSelectionState(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }

        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );

        if (container == null) {
            return null;
        }

        Object object =
                container.getTag();

        if (!(object instanceof ArrayList)) {
            return null;
        }

        SelectionState state =
                new SelectionState();

        try {

            ArrayList<?> outer =
                    (ArrayList<?>) object;

            for (Object rowObject : outer) {

                if (!(rowObject instanceof ArrayList)) {
                    continue;
                }

                ArrayList<?> sourceRow =
                        (ArrayList<?>) rowObject;

                ArrayList<TextView> row =
                        new ArrayList<TextView>();

                for (Object cellObject : sourceRow) {

                    if (cellObject instanceof TextView) {

                        TextView cell =
                                (TextView) cellObject;

                        row.add(cell);
                        state.cells.add(cell);
                    }
                }

                state.records.add(row);
            }

        } catch (Exception ignored) {
        }

        return state;
    }


    // ============================================================
    // SELECTION HELPERS
    //
    // These do not move Selection out of ButtonsController.
    // They only preserve the Edit Menu's previous selected-cell
    // bookkeeping.
    // ============================================================

    private static boolean isSelected(
            TextView cell
    ) {

        if (cell == null) {
            return false;
        }

        return cell.isSelected();
    }


    private static void setCellSelected(
            TextView cell,
            boolean selected
    ) {

        if (cell == null) {
            return;
        }

        cell.setSelected(
                selected
        );
    }


    private static void clearSelection(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }

        for (TextView cell : state.cells) {

            if (cell != null) {

                cell.setSelected(
                        false
                );
            }
        }
    }


    private static void deselectState(
            SelectionState state
    ) {

        clearSelection(
                state
        );
    }


    private static void selection(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }

        /*
         * Selection remains owned by ButtonsController.
         */
        ButtonsController.selection(
                activity
        );
    }


    private static void deselectAll(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }

        ButtonsController.deselectAll(
                activity
        );
    }


    // ============================================================
    // EDIT
    // ============================================================

    public static void edit(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }

        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );

        if (container == null) {
            return;
        }

        Object object =
                container.getTag();

        if (!(object instanceof ArrayList)) {

            Toast.makeText(
                    activity,
                    "No records available",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }

        ArrayList<ArrayList<TextView>>
                taggedRecords =
                (ArrayList<ArrayList<TextView>>)
                        object;

        if (taggedRecords.size() == 0) {

            Toast.makeText(
                    activity,
                    "No records available",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }

        EditSession session =
                getEditSession(
                        activity
                );

        if (session == null) {
            return;
        }

        if (session.editMode) {

            closeEditSession(
                    activity
            );

            return;
        }


        // --------------------------------------------------------
        // SAVE PRE-EDIT SELECTION
        // --------------------------------------------------------

        session.preEditSelectedCells.clear();

        SelectionState oldState =
                getSelectionState(
                        activity
                );

        if (oldState != null) {

            for (
                    int r = 0;
                    r < oldState.records.size();
                    r++
            ) {

                ArrayList<TextView> oldRecord =
                        oldState.records.get(r);

                if (oldRecord == null) {
                    continue;
                }

                for (
                        int c = 0;
                        c < oldRecord.size();
                        c++
                ) {

                    TextView oldCell =
                            oldRecord.get(c);

                    if (
                            oldCell != null
                            &&
                            isSelected(oldCell)
                    ) {

                        session.preEditSelectedCells.add(
                                r + ":" + c
                        );
                    }
                }
            }
        }


        session.editMode = true;

        session.deletedRows.clear();

        session.removedCells.clear();

        session.originalRecords.clear();

        session.originalRowHeights.clear();

        session.originalRowWeights.clear();

        session.rowsFrozen = false;

        session.focusedCell = null;


        ArrayList<String> records =
                getRecords(
                        activity
                );

        session.originalRecords.addAll(
                records
        );


        makeCellsEditable(
                activity
        );


        SelectionState currentState =
                getSelectionState(
                        activity
                );


        freezeRecordRowsForEdit(
                currentState,
                session
        );


        clearSelection(
                currentState
        );


        restorePreEditSelection(
                activity
        );


        showEditPopup(
                activity
        );
    }


    // ============================================================
    // GET RECORDS
    // ============================================================

    private static ArrayList<String> getRecords(
            Activity activity
    ) {

        ArrayList<String> records =
                new ArrayList<String>();

        if (activity == null) {
            return records;
        }

        try {

            java.io.BufferedReader reader =
                    new java.io.BufferedReader(
                            new java.io.InputStreamReader(
                                    activity.openFileInput(
                                            RECORDS_FILE
                                    ),
                                    "UTF-8"
                            )
                    );

            String line;

            while (
                    (line = reader.readLine())
                            != null
            ) {

                if (line.trim().length() > 0) {

                    records.add(
                            line
                    );
                }
            }

            reader.close();

        } catch (Exception ignored) {
        }

        return records;
    }


    // ============================================================
    // RESTORE PRE-EDIT SELECTION
    // ============================================================

    private static void restorePreEditSelection(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }

        EditSession session =
                getEditSession(
                        activity
                );

        SelectionState state =
                getSelectionState(
                        activity
                );

        if (
                session == null
                ||
                state == null
                ||
                session.preEditSelectedCells.isEmpty()
        ) {

            return;
        }

        for (
                String key :
                session.preEditSelectedCells
        ) {

            int separator =
                    key.indexOf(":");

            if (separator <= 0) {
                continue;
            }

            try {

                int row =
                        Integer.parseInt(
                                key.substring(
                                        0,
                                        separator
                                )
                        );

                int column =
                        Integer.parseInt(
                                key.substring(
                                        separator + 1
                                )
                        );

                if (
                        row >= 0
                        &&
                        row < state.records.size()
                ) {

                    ArrayList<TextView> record =
                            state.records.get(row);

                    if (
                            record != null
                            &&
                            column >= 0
                            &&
                            column < record.size()
                    ) {

                        setCellSelected(
                                record.get(column),
                                true
                        );
                    }
                }

            } catch (Exception ignored) {
            }
        }
    }


    // ============================================================
    // MAKE CELLS EDITABLE
    // ============================================================

    @SuppressWarnings("unchecked")
    private static void makeCellsEditable(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }

        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );

        if (container == null) {
            return;
        }

        Object object =
                container.getTag();

        if (!(object instanceof ArrayList)) {
            return;
        }

        ArrayList<ArrayList<TextView>>
                records =
                (ArrayList<ArrayList<TextView>>)
                        object;

        final EditSession session =
                getEditSession(
                        activity
                );

        if (session == null) {
            return;
        }

        for (
                int r = 0;
                r < records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    records.get(r);

            if (record == null) {
                continue;
            }

            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView oldCell =
                        record.get(c);

                if (oldCell == null) {
                    continue;
                }

                if (
                        oldCell instanceof EditText
                ) {

                    final EditText existing =
                            (EditText) oldCell;

                    configureEditTextInput(
                            existing,
                            c
                    );

                    existing.setFocusable(
                            true
                    );

                    existing.setFocusableInTouchMode(
                            true
                    );

                    existing.setCursorVisible(
                            true
                    );

                    existing.setEnabled(
                            true
                    );

                    existing.setOnFocusChangeListener(
                            new View.OnFocusChangeListener() {

                                @Override
                                public void onFocusChange(
                                        View v,
                                        boolean hasFocus
                                ) {

                                    if (
                                            hasFocus
                                    ) {

                                        session.focusedCell =
                                                (TextView) v;
                                    }
                                }
                            }
                    );

                    continue;
                }

                ViewParent parentObject =
                        oldCell.getParent();

                if (
                        !(parentObject instanceof ViewGroup)
                ) {
                    continue;
                }

                ViewGroup parent =
                        (ViewGroup)
                                parentObject;

                int position =
                        parent.indexOfChild(
                                oldCell
                        );

                if (position < 0) {
                    continue;
                }

                EditText editText =
                        new EditText(
                                activity
                        );

                editText.setText(
                        oldCell.getText()
                );

                editText.setTextColor(
                        BLACK
                );

                editText.setTextSize(
                        android.util.TypedValue
                                .COMPLEX_UNIT_PX,
                        oldCell.getTextSize()
                );

                editText.setGravity(
                        oldCell.getGravity()
                );

                editText.setPadding(
                        oldCell.getPaddingLeft(),
                        oldCell.getPaddingTop(),
                        oldCell.getPaddingRight(),
                        oldCell.getPaddingBottom()
                );

                editText.setTypeface(
                        oldCell.getTypeface()
                );

                editText.setIncludeFontPadding(
                        oldCell.getIncludeFontPadding()
                );

                editText.setMinHeight(
                        oldCell.getMinHeight()
                );

                editText.setMinWidth(
                        oldCell.getMinWidth()
                );

                Drawable background =
                        oldCell.getBackground();

                if (background != null) {

                    editText.setBackground(
                            background
                    );
                }

                configureEditTextInput(
                        editText,
                        c
                );

                editText.setFocusable(
                        true
                );

                editText.setFocusableInTouchMode(
                        true
                );

                editText.setCursorVisible(
                        true
                );

                editText.setEnabled(
                        true
                );

                ViewGroup.LayoutParams params =
                        oldCell.getLayoutParams();

                editText.setLayoutParams(
                        params
                );

                editText.setOnFocusChangeListener(
                        new View.OnFocusChangeListener() {

                            @Override
                            public void onFocusChange(
                                    View v,
                                    boolean hasFocus
                            ) {

                                if (
                                        hasFocus
                                ) {

                                    session.focusedCell =
                                            (TextView) v;
                                }
                            }
                        }
                );

                parent.removeViewAt(
                        position
                );

                parent.addView(
                        editText,
                        position
                );

                record.set(
                        c,
                        editText
                );
            }
        }

        container.setTag(
                records
        );

        /*
         * Selection stays in ButtonsController.
         */
        selection(
                activity
        );
    }


    // ============================================================
    // CONFIGURE EDIT TEXT
    // ============================================================

    private static void configureEditTextInput(
            EditText editText,
            int column
    ) {

        if (editText == null) {
            return;
        }

        editText.setHorizontallyScrolling(
                false
        );

        editText.setMaxLines(
                2
        );

        if (column == 0) {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_WORDS
            );

            editText.setSingleLine(
                    false
            );

        } else if (column == 1) {

            editText.setInputType(
                    InputType.TYPE_CLASS_PHONE
            );

            editText.setSingleLine(
                    false
            );

        } else if (column == 2) {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_SENTENCES
                            |
                            InputType.TYPE_TEXT_FLAG_MULTI_LINE
            );

            editText.setSingleLine(
                    false
            );

        } else {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_SENTENCES
            );

            editText.setSingleLine(
                    false
            );
        }
    }


    // ============================================================
    // DISABLE EDITING
    // ============================================================

    private static void disableEditingOnCells(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }

        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            TextView cell =
                    state.cells.get(i);

            if (
                    cell instanceof EditText
            ) {

                EditText editText =
                        (EditText) cell;

                editText.clearFocus();

                editText.setCursorVisible(
                        false
                );

                editText.setFocusable(
                        false
                );

                editText.setFocusableInTouchMode(
                        false
                );

                editText.setEnabled(
                        true
                );
            }
        }
    }


    // ============================================================
    // SHOW EDIT POPUP
    // ============================================================

    private static void showEditPopup(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }

        final EditSession session =
                getEditSession(
                        activity
                );

        if (session == null) {
            return;
        }

        View reusablePopup = null;

        if (
                session.editPopupWindow != null
        ) {

            ViewParent existingParent =
                    session.editPopupWindow.getParent();

            if (
                    existingParent instanceof ViewGroup
            ) {

                reusablePopup =
                        session.editPopupWindow;

            } else {

                session.editPopupWindow =
                        null;
            }
        }

        final View anchor =
                activity.findViewById(
                        R.id.editMenuButton
                );

        if (anchor == null) {
            return;
        }

        ViewParent parentObject =
                anchor.getParent();

        if (
                !(parentObject instanceof ViewGroup)
        ) {
            return;
        }

        ViewGroup originalParent =
                (ViewGroup) parentObject;

        final View popupView;
        final View popupContainer;

        if (
                reusablePopup != null
        ) {

            popupView =
                    reusablePopup;

            popupContainer =
                    reusablePopup;

        } else {

            popupView =
                    LayoutInflater.from(
                            activity
                    ).inflate(
                            R.layout.edit_popup_menu,
                            originalParent,
                            false
                    );

            if (popupView == null) {
                return;
            }

            popupContainer =
                    popupView.findViewById(
                            R.id.editPopupContainer
                    );
        }

        if (popupContainer == null) {
            return;
        }


        // --------------------------------------------------------
        // CLEAR
        // --------------------------------------------------------

        View clearButton =
                popupView.findViewById(
                        R.id.clearButton
                );

        if (clearButton != null) {

            clearButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            EditMenuActions actions =
                                    getEditMenuActions(
                                            activity
                                    );

                            if (actions != null) {

                                actions.onClear();

                            } else {

                                confirmClear(
                                        activity
                                );
                            }
                        }
                    }
            );
        }


        // --------------------------------------------------------
        // DELETE
        // --------------------------------------------------------

        View deleteButton =
                popupView.findViewById(
                        R.id.deleteButton
                );

        if (deleteButton != null) {

            deleteButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            EditMenuActions actions =
                                    getEditMenuActions(
                                            activity
                                    );

                            if (actions != null) {

                                actions.onDelete();

                            } else {

                                deleteFocusedRow(
                                        activity
                                );
                            }
                        }
                    }
            );
        }


        // --------------------------------------------------------
        // SAVE
        // --------------------------------------------------------

        View saveButton =
                popupView.findViewById(
                        R.id.saveButton
                );

        if (saveButton != null) {

            saveButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            EditMenuActions actions =
                                    getEditMenuActions(
                                            activity
                                    );

                            if (actions != null) {

                                actions.onSave();

                            } else {

                                save(
                                        activity
                                );
                            }
                        }
                    }
            );
        }


        // --------------------------------------------------------
        // CLOSE
        // --------------------------------------------------------

        View closeButton =
                popupView.findViewById(
                        R.id.closeButton
                );

        if (closeButton != null) {

            closeButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            EditMenuActions actions =
                                    getEditMenuActions(
                                            activity
                                    );

                            if (actions != null) {

                                actions.onClose();

                            } else {

                                closeEditSession(
                                        activity
                                );
                            }
                        }
                    }
            );
        }


        int originalIndex =
                originalParent.indexOfChild(
                        anchor
                );

        if (originalIndex < 0) {
            return;
        }

        if (
                reusablePopup == null
        ) {

            originalParent.addView(
                    popupContainer,
                    originalIndex + 1
            );
        }

        popupContainer.setVisibility(
                View.VISIBLE
        );

        originalParent.requestLayout();

        originalParent.invalidate();

        session.editPopupWindow =
                popupContainer;
    }


    // ============================================================
    // DISMISS EDIT POPUP
    // ============================================================

    private static void dismissEditPopup(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }

        EditSession session =
                getEditSession(
                        activity
                );

        if (session == null) {
            return;
        }

        if (
                session.editPopupWindow != null
        ) {

            try {

                ViewParent parentObject =
                        session.editPopupWindow.getParent();

                if (
                        parentObject instanceof ViewGroup
                ) {

                    session.editPopupWindow.setVisibility(
                            View.INVISIBLE
                    );

                    ((ViewGroup) parentObject)
                            .requestLayout();

                    ((ViewGroup) parentObject)
                            .invalidate();

                } else {

                    session.editPopupWindow =
                            null;
                }

            } catch (Exception ignored) {
            }
        }
    }


    // ============================================================
    // PREPARE EDIT POPUP PLACEHOLDER
    // ============================================================

    private static void prepareEditPopupPlaceholder(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }

        final EditSession session =
                getEditSession(
                        activity
                );

        if (
                session == null
                ||
                session.editPopupWindow != null
        ) {
            return;
        }

        final View anchor =
                activity.findViewById(
                        R.id.editMenuButton
                );

        if (anchor == null) {
            return;
        }

        ViewParent parentObject =
                anchor.getParent();

        if (
                !(parentObject instanceof ViewGroup)
        ) {
            return;
        }

        ViewGroup originalParent =
                (ViewGroup) parentObject;

        try {

            View popupView =
                    LayoutInflater.from(
                            activity
                    ).inflate(
                            R.layout.edit_popup_menu,
                            originalParent,
                            false
                    );

            if (popupView == null) {
                return;
            }

            View popupContainer =
                    popupView.findViewById(
                            R.id.editPopupContainer
                    );

            if (popupContainer == null) {
                return;
            }

            int originalIndex =
                    originalParent.indexOfChild(
                            anchor
                    );

            if (originalIndex < 0) {
                return;
            }

            popupContainer.setVisibility(
                    View.INVISIBLE
            );

            originalParent.addView(
                    popupContainer,
                    originalIndex + 1
            );

            session.editPopupWindow =
                    popupContainer;

            originalParent.requestLayout();

            originalParent.invalidate();

        } catch (Exception ignored) {
        }
    }

    // ============================================================
    // CLOSE EDIT SESSION
    // ============================================================

    private static void clearEditActionFocus(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }

        try {

            View currentFocus =
                    activity.getCurrentFocus();

            if (currentFocus != null) {
                currentFocus.clearFocus();
            }

            activity.getWindow()
                    .setSoftInputMode(
                            android.view.WindowManager
                                    .LayoutParams
                                    .SOFT_INPUT_STATE_ALWAYS_HIDDEN
                    );

        } catch (Exception ignored) {
        }
    }


    private static void closeEditSession(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }

        EditSession session =
                getEditSession(
                        activity
                );

        if (
                session == null
                ||
                !session.editMode
        ) {

            dismissEditPopup(
                    activity
            );

            return;
        }

        SelectionState state =
                getSelectionState(
                        activity
                );

        if (state != null) {

            restoreOriginalRecords(
                    activity,
                    state,
                    session.originalRecords
            );

            showAllRecordRows(
                    state
            );

            restoreRecordRowLayout(
                    session
            );
        }

        session.editMode =
                false;

        session.deletedRows.clear();

        session.removedCells.clear();

        session.originalRecords.clear();

        session.focusedCell =
                null;

        SelectionState newState =
                getSelectionState(
                        activity
                );

        if (newState != null) {

            disableEditingOnCells(
                    newState
            );

            renumberVisibleSerialNumbers(
                    activity,
                    newState
            );
        }

        selection(
                activity
        );

        dismissEditPopup(
                activity
        );

        clearEditActionFocus(
                activity
        );
    }

    // ============================================================
    // RESTORE ORIGINAL RECORDS
    // ============================================================

    private static void restoreOriginalRecords(
            Activity activity,
            SelectionState state,
            ArrayList<String> originalRecords
    ) {

        if (
                activity == null
                ||
                state == null
                ||
                originalRecords == null
        ) {
            return;
        }

        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);

            if (record == null) {
                continue;
            }

            if (
                    r >= originalRecords.size()
            ) {
                continue;
            }

            String original =
                    originalRecords.get(r);

            String[] parts =
                    original.split(
                            "\\s*\\|\\s*",
                            -1
                    );

            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                String value = "";

                if (
                        c < parts.length
                ) {

                    value =
                            parts[c].trim();
                }

                TextView oldCell =
                        record.get(c);

                if (oldCell == null) {
                    continue;
                }

                if (
                        !(oldCell instanceof EditText)
                ) {

                    oldCell.setText(
                            value
                    );

                    continue;
                }

                ViewParent parentObject =
                        oldCell.getParent();

                if (
                        !(parentObject instanceof ViewGroup)
                ) {

                    oldCell.setText(
                            value
                    );

                    continue;
                }

                ViewGroup parent =
                        (ViewGroup) parentObject;

                int position =
                        parent.indexOfChild(
                                oldCell
                        );

                if (position < 0) {

                    oldCell.setText(
                            value
                    );

                    continue;
                }

                TextView textView =
                        createNormalTextView(
                                activity,
                                oldCell,
                                value
                        );

                parent.removeViewAt(
                        position
                );

                parent.addView(
                        textView,
                        position
                );

                record.set(
                        c,
                        textView
                );
            }
        }

        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );

        if (container != null) {

            container.setTag(
                    state.records
            );
        }
    }


    // ============================================================
    // CREATE NORMAL TEXTVIEW
    // ============================================================

    private static TextView createNormalTextView(
            Activity activity,
            TextView source,
            String value
    ) {

        TextView textView =
                new TextView(
                        activity
                );

        textView.setText(
                value
        );

        textView.setTextColor(
                BLACK
        );

        textView.setTextSize(
                android.util.TypedValue
                        .COMPLEX_UNIT_PX,
                source.getTextSize()
        );

        textView.setGravity(
                source.getGravity()
        );

        textView.setPadding(
                source.getPaddingLeft(),
                source.getPaddingTop(),
                source.getPaddingRight(),
                source.getPaddingBottom()
        );

        textView.setTypeface(
                source.getTypeface()
        );

        textView.setIncludeFontPadding(
                source.getIncludeFontPadding()
        );

        textView.setMinHeight(
                source.getMinHeight()
        );

        textView.setMinWidth(
                source.getMinWidth()
        );

        Drawable background =
                source.getBackground();

        if (background != null) {

            textView.setBackground(
                    background
            );
        }

        textView.setClickable(
                true
        );

        textView.setEnabled(
                true
        );

        ViewGroup.LayoutParams params =
                source.getLayoutParams();

        if (params != null) {

            textView.setLayoutParams(
                    params
            );
        }

        return textView;
    }


    // ============================================================
    // SHOW ALL RECORD ROWS
    // ============================================================

    private static void showAllRecordRows(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }

        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            showRecordRow(
                    state.records.get(r)
            );
        }
    }


    // ============================================================
    // SHOW RECORD ROW
    // ============================================================

    private static void showRecordRow(
            ArrayList<TextView> record
    ) {

        if (
                record == null
                ||
                record.size() == 0
        ) {
            return;
        }

        TextView firstCell =
                record.get(0);

        if (firstCell == null) {
            return;
        }

        ViewParent parent =
                firstCell.getParent();

        if (parent instanceof View) {

            ((View) parent).setVisibility(
                    View.VISIBLE
            );
        }
    }

    // ============================================================
    // DELETE FOCUSED ROW
    // ============================================================

    private static void deleteFocusedRow(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }

        final EditSession session =
                getEditSession(
                        activity
                );

        if (
                session == null
                ||
                !session.editMode
        ) {
            return;
        }

        final SelectionState state =
                getSelectionState(
                        activity
                );

        if (state == null) {
            return;
        }

        final Set<Integer> rowsToDelete =
                new HashSet<Integer>();

        for (
                String key :
                session.preEditSelectedCells
        ) {

            int separator =
                    key.indexOf(":");

            if (separator > 0) {

                try {

                    int row =
                            Integer.parseInt(
                                    key.substring(
                                            0,
                                            separator
                                    )
                            );

                    if (
                            row >= 0
                            &&
                            row < state.records.size()
                    ) {

                        rowsToDelete.add(
                                row
                        );
                    }

                } catch (Exception ignored) {
                }
            }
        }

        if (rowsToDelete.isEmpty()) {

            TextView focusedCell =
                    getFocusedEditCell(
                            activity
                    );

            if (focusedCell == null) {
                return;
            }

            int row =
                    getRecordIndexForView(
                            activity,
                            focusedCell
                    );

            if (row >= 0) {

                rowsToDelete.add(
                        row
                );
            }
        }

        if (rowsToDelete.isEmpty()) {
            return;
        }

        final int count =
                rowsToDelete.size();

        showDeleteConfirmation(
                activity,
                rowsToDelete,
                state,
                session,
                count
        );
    }

    // ============================================================
    // CONFIRMATION WINDOW POSITION
    // ============================================================

    private static void setConfirmationWindowPosition(
            Activity activity,
            android.view.Window window
    ) {

        if (
                activity == null
                ||
                window == null
        ) {
            return;
        }

        android.view.WindowManager.LayoutParams params =
                window.getAttributes();

        switch (
                CONFIRM_POSITION_MODE
        ) {

            case CONFIRM_POSITION_TOP_LEFT:

                params.gravity =
                        Gravity.TOP
                                |
                                Gravity.LEFT;

                params.x =
                        dp(
                                activity,
                                CONFIRM_X_DP
                        );

                params.y =
                        dp(
                                activity,
                                CONFIRM_Y_DP
                        );

                break;


            case CONFIRM_POSITION_TOP_RIGHT:

                params.gravity =
                        Gravity.TOP
                                |
                                Gravity.RIGHT;

                params.x =
                        dp(
                                activity,
                                CONFIRM_X_DP
                        );

                params.y =
                        dp(
                                activity,
                                CONFIRM_Y_DP
                        );

                break;


            case CONFIRM_POSITION_BOTTOM_LEFT:

                params.gravity =
                        Gravity.BOTTOM
                                |
                                Gravity.LEFT;

                params.x =
                        dp(
                                activity,
                                CONFIRM_X_DP
                        );

                params.y =
                        dp(
                                activity,
                                CONFIRM_Y_DP
                        );

                break;


            case CONFIRM_POSITION_BOTTOM_RIGHT:

                params.gravity =
                        Gravity.BOTTOM
                                |
                                Gravity.RIGHT;

                params.x =
                        dp(
                                activity,
                                CONFIRM_X_DP
                        );

                params.y =
                        dp(
                                activity,
                                CONFIRM_Y_DP
                        );

                break;


            case CONFIRM_POSITION_CENTER:

                params.gravity =
                        Gravity.CENTER;

                params.x =
                        0;

                params.y =
                        dp(
                                activity,
                                CONFIRM_Y_DP
                        );

                break;


            default:

                params.gravity =
                        Gravity.CENTER;

                params.x =
                        0;

                params.y =
                        0;

                break;
        }

        window.setAttributes(
                params
        );
    }

    // ============================================================
    // GET FOCUSED EDIT CELL
    // ============================================================

    private static TextView getFocusedEditCell(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }

        EditSession session =
                getEditSession(
                        activity
                );

        if (
                session == null
                ||
                !session.editMode
        ) {
            return null;
        }

        if (
                session.focusedCell != null
                &&
                session.focusedCell.getVisibility()
                        != View.GONE
                &&
                session.focusedCell.getParent()
                        != null
        ) {

            return session.focusedCell;
        }

        SelectionState state =
                getSelectionState(
                        activity
                );

        if (state == null) {
            return null;
        }

        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            TextView cell =
                    state.cells.get(i);

            if (
                    cell instanceof EditText
                    &&
                    cell.hasFocus()
            ) {

                session.focusedCell =
                        cell;

                return cell;
            }
        }

        return null;
    }

    // CLEAR FOCUSED CELL
    // ============================================================

            Activity activity
    ) {

        if (activity == null) {

        }

        EditSession session =
                getEditSession(

        if (
                session == null
                ||

        SelectionState state =
                getSelectionState(
                        activity
                );

        if (
                state != null
                &&
                !session.preEditSelectedCells.isEmpty()
        ) {

            boolean changed =
                    false;

            for (
                    int r = 0;
                    r < state.records.size();
                    r++
            ) {

                ArrayList<TextView> record =
                        state.records.get(r);

                if (record == null) {
                    continue;
                }

                for (
                        int c = 0;
                        c < record.size();
                        c++
                ) {

                    if (
                            session.preEditSelectedCells
                                    .contains(
                                            r + ":" + c
                                    )
                    ) {

                        TextView cell =
                                record.get(c);

                        if (cell != null) {

                            cell.setText(
                                    ""
                            );

                            session.removedCells.add(
                                    cell
                            );

                            changed =
                                    true;
                        }
                    }
                }
            }

            if (changed) {

                deselectState(
                        state
                );

                session.preEditSelectedCells.clear();

                session.focusedCell =
                        null;

                clearEditActionFocus(
                        activity
                );

                return;
            }
        }

        TextView cell =
                getFocusedEditCell(
                        activity
                );

        if (cell == null) {
            return;
        }

        session.focusedCell =
                cell;

        cell.requestFocus();

        cell.setText(
                ""
        );

        if (
                cell instanceof EditText
        ) {

            ((EditText) cell)
                    .setSelection(
                            0
                    );
        }

        session.removedCells.add(
                cell
        );

        clearEditActionFocus(
                activity
        );
    }


    // ============================================================
    // HIDE RECORD ROW
    // ============================================================

    private static void hideRecordRow(
            ArrayList<TextView> record
    ) {

        if (
                record == null
                ||
                record.size() == 0
        ) {
            return;
        }

        TextView firstCell =
                record.get(0);

        if (firstCell == null) {
            return;
        }

        ViewParent parent =
                firstCell.getParent();

        if (
                parent instanceof ViewGroup
        ) {

            ((ViewGroup) parent)
                    .setVisibility(
                            View.GONE
                    );
        }
    }


    // ============================================================
    // GET RECORD INDEX FOR VIEW
    // ============================================================

    private static int getRecordIndexForView(
            Activity activity,
            TextView target
    ) {

        if (
                activity == null
                ||
                target == null
        ) {
            return -1;
        }

        SelectionState state =
                getSelectionState(
                        activity
                );

        if (state == null) {
            return -1;
        }

        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);

            if (record == null) {
                continue;
            }

            for (
                    TextView cell :
                    record
            ) {

                if (cell == target) {
                    return r;
                }
            }
        }

        return -1;
    }


    // ============================================================
    // FREEZE RECORD ROWS
    // ============================================================

    private static void freezeRecordRowsForEdit(
            SelectionState state,
            EditSession session
    ) {

        if (
                state == null
                ||
                session == null
                ||
                session.rowsFrozen
        ) {
            return;
        }

        session.originalRowHeights.clear();

        session.originalRowWeights.clear();

        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);

            if (
                    record == null
                    ||
                    record.size() == 0
            ) {
                continue;
            }

            TextView firstCell =
                    record.get(0);

            if (firstCell == null) {
                continue;
            }

            ViewParent parentObject =
                    firstCell.getParent();

            if (
                    !(parentObject instanceof ViewGroup)
            ) {
                continue;
            }

            View row =
                    (View) parentObject;

            ViewGroup.LayoutParams params =
                    row.getLayoutParams();

            if (params == null) {
                continue;
            }

            int measuredHeight =
                    row.getHeight();

            if (measuredHeight <= 0) {

                measuredHeight =
                        firstCell.getHeight();
            }

            if (measuredHeight <= 0) {
                continue;
            }

            session.originalRowHeights.put(
                    row,
                    Integer.valueOf(
                            params.height
                    )
            );

            if (
                    params instanceof LinearLayout.LayoutParams
            ) {

                LinearLayout.LayoutParams lp =
                        (LinearLayout.LayoutParams)
                                params;

                session.originalRowWeights.put(
                        row,
                        Float.valueOf(
                                lp.weight
                        )
                );

                lp.height =
                        measuredHeight;

                lp.weight =
                        0f;

            } else {

                params.height =
                        measuredHeight;
            }

            row.setLayoutParams(
                    params
            );
        }

        session.rowsFrozen =
                true;
    }


    // ============================================================
    // FINALIZE ROW LAYOUT
    // ============================================================

    private static void finalizeRecordRowLayoutAfterSave(
            EditSession session
    ) {

        if (
                session == null
                ||
                !session.rowsFrozen
        ) {
            return;
        }

        for (
                Map.Entry<View, Integer> entry :
                session.originalRowHeights.entrySet()
        ) {

            View row =
                    entry.getKey();

            if (row == null) {
                continue;
            }

            ViewGroup.LayoutParams params =
                    row.getLayoutParams();

            if (params == null) {
                continue;
            }

            Integer originalHeight =
                    entry.getValue();

            if (
                    originalHeight != null
                    &&
                    originalHeight.intValue() > 0
            ) {

                params.height =
                        originalHeight.intValue();
            }

            if (
                    params instanceof LinearLayout.LayoutParams
            ) {

                ((LinearLayout.LayoutParams) params)
                        .weight =
                        0f;
            }

            row.setLayoutParams(
                    params
            );
        }

        session.originalRowHeights.clear();

        session.originalRowWeights.clear();

        session.rowsFrozen =
                false;
    }


    // ============================================================
    // RESTORE ROW LAYOUT
    // ============================================================

    private static void restoreRecordRowLayout(
            EditSession session
    ) {

        if (
                session == null
                ||
                !session.rowsFrozen
        ) {
            return;
        }

        for (
                Map.Entry<View, Integer> entry :
                session.originalRowHeights.entrySet()
        ) {

            View row =
                    entry.getKey();

            if (row == null) {
                continue;
            }

            ViewGroup.LayoutParams params =
                    row.getLayoutParams();

            if (params == null) {
                continue;
            }

            Integer originalHeight =
                    entry.getValue();

            if (originalHeight != null) {

                params.height =
                        originalHeight.intValue();
            }

            if (
                    params instanceof LinearLayout.LayoutParams
            ) {

                Float originalWeight =
                        session.originalRowWeights.get(
                                row
                        );

                if (originalWeight != null) {

                    ((LinearLayout.LayoutParams) params)
                            .weight =
                            originalWeight.floatValue();
                }
            }

            row.setLayoutParams(
                    params
            );
        }

        session.originalRowHeights.clear();

        session.originalRowWeights.clear();

        session.rowsFrozen =
                false;
    }


    // ============================================================
    // RESTORE EDITED RECORDS AS TEXTVIEWS
    // ============================================================

    private static void restoreEditedRecordsAsTextViews(
            Activity activity,
            SelectionState state
    ) {

        if (
                activity == null
                ||
                state == null
        ) {
            return;
        }

        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);

            if (record == null) {
                continue;
            }

            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView oldCell =
                        record.get(c);

                if (
                        oldCell == null
                        ||
                        !(oldCell instanceof EditText)
                ) {
                    continue;
                }

                ViewParent parentObject =
                        oldCell.getParent();

                if (
                        !(parentObject instanceof ViewGroup)
                ) {
                    continue;
                }

                ViewGroup parent =
                        (ViewGroup) parentObject;

                int position =
                        parent.indexOfChild(
                                oldCell
                        );

                if (position < 0) {
                    continue;
                }

                TextView textView =
                        createNormalTextView(
                                activity,
                                oldCell,
                                oldCell.getText().toString()
                        );

                parent.removeViewAt(
                        position
                );

                parent.addView(
                        textView,
                        position
                );

                record.set(
                        c,
                        textView
                );
            }
        }

        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );

        if (container != null) {

            container.setTag(
                    state.records
            );
        }
    }


    // ============================================================
    // SAVE EDITED RECORDS
    // ============================================================

    private static void saveEditedRecords(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }

        EditSession session =
                getEditSession(
                        activity
                );

        if (
                session == null
                ||
                !session.editMode
        ) {
            return;
        }

        SelectionState state =
                getSelectionState(
                        activity
                );

        if (state == null) {
            return;
        }

        ArrayList<String> updatedRecords =
                new ArrayList<String>();

        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            if (
                    session.deletedRows.contains(
                            r
                    )
            ) {
                continue;
            }

            ArrayList<TextView> record =
                    state.records.get(r);

            if (record == null) {
                continue;
            }

            String name = "";
            String phone = "";
            String address = "";
            String product = "";

            if (
                    record.size() > 0
                    &&
                    record.get(0) != null
            ) {

                name =
                        record.get(0)
                                .getText()
                                .toString()
                                .trim();
            }

            if (
                    record.size() > 1
                    &&
                    record.get(1) != null
            ) {

                phone =
                        record.get(1)
                                .getText()
                                .toString()
                                .trim();
            }

            if (
                    record.size() > 2
                    &&
                    record.get(2) != null
            ) {

                address =
                        record.get(2)
                                .getText()
                                .toString()
                                .trim();
            }

            if (
                    record.size() > 3
                    &&
                    record.get(3) != null
            ) {

                product =
                        record.get(3)
                                .getText()
                                .toString()
                                .trim();
            }

            updatedRecords.add(
                    name
                            + " | "
                            + phone
                            + " | "
                            + address
                            + " | "
                            + product
            );
        }

        FileOutputStream output =
                null;

        try {

            output =
                    activity.openFileOutput(
                            RECORDS_FILE,
                            Context.MODE_PRIVATE
                    );

            for (
                    String line :
                    updatedRecords
            ) {

                output.write(
                        (
                                line
                                        +
                                        "\n"
                        ).getBytes(
                                "UTF-8"
                        )
                );
            }

            output.flush();

            output.close();

            output =
                    null;


            restoreEditedRecordsAsTextViews(
                    activity,
                    state
            );


            finalizeRecordRowLayoutAfterSave(
                    session
            );


            session.editMode =
                    false;

            session.deletedRows.clear();

            session.removedCells.clear();

            session.originalRecords.clear();

            session.preEditSelectedCells.clear();

            session.focusedCell =
                    null;


            SelectionState newState =
                    getSelectionState(
                            activity
                    );

            if (newState != null) {

                disableEditingOnCells(
                        newState
                );

                clearSelection(
                        newState
                );

                renumberVisibleSerialNumbers(
                        activity,
                        newState
                );
            }


            deselectAll(
                    activity
            );


            selection(
                    activity
            );


            dismissEditPopup(
                    activity
            );


            clearEditActionFocus(
                    activity
            );


            Toast.makeText(
                    activity,
                    "Data Update",
                    Toast.LENGTH_SHORT
            ).show();

        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to update records",
                    Toast.LENGTH_SHORT
            ).show();

        } finally {

            if (output != null) {

                try {
                    output.close();
                } catch (Exception ignored) {
                }
            }
        }
    }


    // ============================================================
    // RENUMBER SERIAL NUMBERS
    // ============================================================

    private static void renumberVisibleSerialNumbers(
            Activity activity,
            SelectionState state
    ) {

        if (
                activity == null
                ||
                state == null
        ) {
            return;
        }

        int serialNumber =
                1;

        Set<TextView> recordCells =
                new HashSet<TextView>();

        for (
                ArrayList<TextView> record :
                state.records
        ) {

            if (record == null) {
                continue;
            }

            for (
                    TextView cell :
                    record
            ) {

                if (cell != null) {

                    recordCells.add(
                            cell
                    );
                }
            }
        }

        for (
                ArrayList<TextView> record :
                state.records
        ) {

            if (
                    record == null
                    ||
                    record.size() == 0
            ) {
                continue;
            }

            TextView firstCell =
                    record.get(0);

            if (firstCell == null) {
                continue;
            }

            ViewParent parent =
                    firstCell.getParent();

            if (
                    !(parent instanceof ViewGroup)
            ) {
                continue;
            }

            ViewGroup row =
                    (ViewGroup) parent;

            if (
                    row.getVisibility()
                            !=
                            View.VISIBLE
            ) {
                continue;
            }

            TextView serialView =
                    findSerialTextView(
                            row,
                            recordCells
                    );

            if (serialView != null) {

                serialView.setText(
                        String.valueOf(
                                serialNumber
                        )
                );

                serialNumber++;
            }
        }
    }


    // ============================================================
    // FIND SERIAL TEXTVIEW
    // ============================================================

    private static TextView findSerialTextView(
            ViewGroup row,
            Set<TextView> recordCells
    ) {

        if (
                row == null
                ||
                recordCells == null
        ) {
            return null;
        }

        for (
                int i = 0;
                i < row.getChildCount();
                i++
        ) {

            View child =
                    row.getChildAt(i);

            if (
                    child instanceof TextView
                    &&
                    !(child instanceof EditText)
            ) {

                TextView textView =
                        (TextView) child;

                if (
                        !recordCells.contains(
                                textView
                        )
                ) {

                    return textView;
                }
            }
        }

        for (
                int i = 0;
                i < row.getChildCount();
                i++
        ) {

            View child =
                    row.getChildAt(i);

            if (
                    child instanceof ViewGroup
            ) {

                TextView found =
                        findSerialTextView(
                                (ViewGroup) child,
                                recordCells
                        );

                if (found != null) {
                    return found;
                }
            }
        }

        return null;
    }


    // ============================================================
    // DP HELPER
    // ============================================================

    private static int dp(
            Activity activity,
            int value
    ) {

        if (activity == null) {
            return value;
        }

        float density =
                activity.getResources()
                        .getDisplayMetrics()
                        .density;

        return (int)
                (value * density + 0.5f);
    }
