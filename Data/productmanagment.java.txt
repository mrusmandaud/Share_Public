public class ProductManagement extends Activity {

    private static final int IMPORT_PRODUCTS = 1001;

    private static final int TYPE_CSV = 1;
    private static final int TYPE_TXT = 2;
    private static final int TYPE_XLSX = 3;

    private int importType = TYPE_TXT;

    private ArrayList<String> productList;

        super.onCreate(savedInstanceState);

    }


    // ============================================================
    // LOAD PRODUCTS
    // ============================================================

    private void loadProducts() {

        productList =
                ButtonsController.getProducts(this);

                    new ArrayList<String>();
        }
    }

    // ============================================================
    // ADD PRODUCT
    // ============================================================

    private void showAddProduct() {

        final EditText input =
                new EditText(this);

        input.setHint(
                "Product name"
        );

        input.setSingleLine(true);


        new AlertDialog.Builder(this)

                .setTitle(
                        "Add Product"
                )

                .setView(input)

                .setPositiveButton(
                        "ADD",
                        new DialogInterface.OnClickListener() {

                    @Override
                    public void onClick(
                            DialogInterface dialog,
                            int which) {

                        String product =
                                input.getText()
                                        .toString()
                                        .trim();


                        if (product.length() == 0) {

                            Toast.makeText(
                                    ProductManagement.this,
                                    "Enter product name",
                                    Toast.LENGTH_SHORT
                            ).show();

                            return;
                        }


                        boolean added =
                                ButtonsController.addProduct(
                                        ProductManagement.this,
                                        product
                                );


                        if (added) {

                            loadProducts();

                            Toast.makeText(
                                    ProductManagement.this,
                                    "Product Added",
                                    Toast.LENGTH_SHORT
                            ).show();

                        } else {

                            Toast.makeText(
                                    ProductManagement.this,
                                    "Product already exists",
                                    Toast.LENGTH_SHORT
                            ).show();
                        }
                    }
                })

                .setNegativeButton(
                        "CANCEL",
                        null
                )

                .show();
    }


    // ============================================================
    // IMPORT OPTIONS
    // ============================================================

    private void showImportOptions() {

        final String[] options = {

                "CSV",
                "TXT",
                "XLSX"

        };


        new AlertDialog.Builder(this)

                .setTitle(
                        "Import Products List"
                )

                .setItems(
                        options,
                        new DialogInterface.OnClickListener() {

                    @Override
                    public void onClick(
                            DialogInterface dialog,
                            int which) {

                        if (which == 0) {

                            importType =
                                    TYPE_CSV;

                            openPicker(
                                    "text/csv"
                            );

                        } else if (which == 1) {

                            importType =
                                    TYPE_TXT;

                            openPicker(
                                    "text/plain"
                            );

                        } else {

                            importType =
                                    TYPE_XLSX;

                            openPicker(
                                    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
                            );
                        }
                    }
                })

                .show();
    }


    // ============================================================
    // FILE PICKER
    // ============================================================

    private void openPicker(
            String mimeType
    ) {

        Intent intent =
                new Intent(
                        Intent.ACTION_OPEN_DOCUMENT
                );


        intent.addCategory(
                Intent.CATEGORY_OPENABLE
        );


        intent.setType(
                mimeType
        );


        startActivityForResult(
                intent,
                IMPORT_PRODUCTS
        );
    }


    // ============================================================
    // IMPORT RESULT
    // ============================================================

    @Override
    protected void onActivityResult(
            int requestCode,
            int resultCode,
            Intent data
    ) {

        super.onActivityResult(
                requestCode,
                resultCode,
                data
        );


        if (
                requestCode != IMPORT_PRODUCTS
                || resultCode != RESULT_OK
                || data == null
                || data.getData() == null
        ) {

            return;
        }


        Uri uri =
                data.getData();


        if (importType == TYPE_XLSX) {

            importXLSX(uri);

        } else {

            importTextFile(uri);
        }
    }


    // ============================================================
    // TXT / CSV IMPORT
    // ============================================================

    private void importTextFile(
            Uri uri
    ) {

        int added = 0;


        try {

            InputStream input =
                    getContentResolver()
                            .openInputStream(uri);


            if (input == null) {

                throw new Exception(
                        "Unable to open file"
                );
            }


            BufferedReader reader =
                    new BufferedReader(
                            new InputStreamReader(
                                    input,
                                    "UTF-8"
                            )
                    );


            String line;


            while (
                    (line = reader.readLine()) != null
            ) {

                line =
                        line.trim();


                if (
                        line.length() == 0
                ) {

                    continue;
                }


                if (importType == TYPE_CSV) {

                    ArrayList<String> values =
                            parseCSVLine(line);


                    for (
                            String value :
                            values
                    ) {

                        String product =
                                value.trim();


                        if (
                                product.length() > 0
                        ) {

                            if (
                                    ButtonsController.addProduct(
                                            ProductManagement.this,
                                            product
                                    )
                            ) {

                                added++;
                            }
                        }
                    }

                } else {

                    if (
                            ButtonsController.addProduct(
                                    ProductManagement.this,
                                    line
                            )
                    ) {

                        added++;
                    }
                }
            }


            reader.close();


            loadProducts();


            Toast.makeText(
                    ProductManagement.this,
                    added +
                    " product(s) imported",
                    Toast.LENGTH_SHORT
            ).show();


        } catch (Exception e) {

            Toast.makeText(
                    ProductManagement.this,
                    "Import failed",
                    Toast.LENGTH_SHORT
            ).show();
        }
    }


    // ============================================================
    // CSV PARSER
    // ============================================================

    private ArrayList<String> parseCSVLine(
            String line
    ) {

        ArrayList<String> result =
                new ArrayList<String>();


        StringBuilder value =
                new StringBuilder();


        boolean quoted =
                false;


        for (
                int i = 0;
                i < line.length();
                i++
        ) {

            char c =
                    line.charAt(i);


            if (c == '"') {

                if (
                        quoted
                        && i + 1 < line.length()
                        && line.charAt(i + 1) == '"'
                ) {

                    value.append('"');

                    i++;

                } else {

                    quoted =
                            !quoted;
                }

            } else if (
                    c == ','
                    && !quoted
            ) {

                result.add(
                        value.toString()
                );

                value.setLength(0);

            } else {

                value.append(c);
            }
        }


        result.add(
                value.toString()
        );


        return result;
    }


    // ============================================================
    // XLSX IMPORT
    // ============================================================

    private void importXLSX(
            Uri uri
    ) {

        int added = 0;


        try {

            ArrayList<String> sharedStrings =
                    readSharedStrings(uri);


            ArrayList<String> cells =
                    readWorksheet(
                            uri,
                            sharedStrings
                    );


            for (
                    String product :
                    cells
            ) {

                product =
                        product.trim();


                if (
                        product.length() == 0
                ) {

                    continue;
                }


                if (
                        ButtonsController.addProduct(
                                ProductManagement.this,
                                product
                        )
                ) {

                    added++;
                }
            }


            loadProducts();


            Toast.makeText(
                    ProductManagement.this,
                    added +
                    " product(s) imported",
                    Toast.LENGTH_SHORT
            ).show();


        } catch (Exception e) {

            Toast.makeText(
                    ProductManagement.this,
                    "XLSX import failed",
                    Toast.LENGTH_SHORT
            ).show();
        }
    }


    // ============================================================
    // READ SHARED STRINGS
    // ============================================================

    private ArrayList<String> readSharedStrings(
            Uri uri
    ) throws Exception {

        ArrayList<String> result =
                new ArrayList<String>();


        InputStream input =
                getContentResolver()
                        .openInputStream(uri);


        if (input == null) {

            throw new Exception(
                    "Unable to open XLSX"
            );
        }


        ZipInputStream zip =
                new ZipInputStream(input);


        ZipEntry entry;


        while (
                (entry = zip.getNextEntry()) != null
        ) {

            if (
                    "xl/sharedStrings.xml"
                            .equals(
                                    entry.getName()
                            )
            ) {

                XmlPullParser parser =
                        Xml.newPullParser();


                parser.setInput(
                        zip,
                        "UTF-8"
                );


                StringBuilder text =
                        new StringBuilder();


                int event;


                while (
                        (event = parser.next())
                        != XmlPullParser.END_DOCUMENT
                ) {

                    if (
                            event == XmlPullParser.START_TAG
                            && "t".equals(
                                    parser.getName()
                            )
                    ) {

                        text.append(
                                parser.nextText()
                        );
                    }


                    if (
                            event == XmlPullParser.END_TAG
                            && "si".equals(
                                    parser.getName()
                            )
                    ) {

                        result.add(
                                text.toString()
                        );

                        text.setLength(0);
                    }
                }


                break;
            }
        }


        zip.close();


        return result;
    }


    // ============================================================
    // READ WORKSHEET
    // ============================================================

    private ArrayList<String> readWorksheet(
            Uri uri,
            ArrayList<String> sharedStrings
    ) throws Exception {

        ArrayList<String> result =
                new ArrayList<String>();


        InputStream input =
                getContentResolver()
                        .openInputStream(uri);


        if (input == null) {

            throw new Exception(
                    "Unable to open XLSX"
            );
        }


        ZipInputStream zip =
                new ZipInputStream(input);


        ZipEntry entry;


        while (
                (entry = zip.getNextEntry()) != null
        ) {

            if (
                    "xl/worksheets/sheet1.xml"
                            .equals(
                                    entry.getName()
                            )
            ) {

                XmlPullParser parser =
                        Xml.newPullParser();


                parser.setInput(
                        zip,
                        "UTF-8"
                );


                int event;


                boolean insideCell =
                        false;


                String cellType =
                        "";


                String cellValue =
                        "";


                String inlineText =
                        "";


                while (
                        (event = parser.next())
                        != XmlPullParser.END_DOCUMENT
                ) {

                    if (
                            event ==
                            XmlPullParser.START_TAG
                    ) {

                        String tag =
                                parser.getName();


                        if (
                                "c".equals(tag)
                        ) {

                            insideCell =
                                    true;

                            cellType =
                                    parser.getAttributeValue(
                                            null,
                                            "t"
                                    );

                            if (cellType == null) {

                                cellType =
                                        "";
                            }

                            cellValue =
                                    "";

                            inlineText =
                                    "";
                        }


                        else if (
                                insideCell
                                && "v".equals(tag)
                        ) {

                            cellValue =
                                    parser.nextText();
                        }


                        else if (
                                insideCell
                                && "t".equals(tag)
                        ) {

                            inlineText =
                                    parser.nextText();
                        }
                    }


                    else if (
                            event ==
                            XmlPullParser.END_TAG
                    ) {

                        if (
                                "c".equals(
                                        parser.getName()
                                )
                        ) {

                            String value;


                            if (
                                    "s".equals(
                                            cellType
                                    )
                            ) {

                                try {

                                    int index =
                                            Integer.parseInt(
                                                    cellValue
                                            );


                                    if (
                                            index >= 0
                                            && index <
                                            sharedStrings.size()
                                    ) {

                                        value =
                                                sharedStrings
                                                        .get(index);

                                    } else {

                                        value =
                                                "";
                                    }

                                } catch (Exception e) {

                                    value =
                                            "";
                                }

                            }

                            else if (
                                    "inlineStr".equals(
                                            cellType
                                    )
                            ) {

                                value =
                                        inlineText;

                            }

                            else {

                                value =
                                        cellValue;
                            }


                            if (
                                    value != null
                                    && value.trim()
                                            .length() > 0
                            ) {

                                result.add(
                                        value
                                );
                            }


                            insideCell =
                                    false;
                        }
                    }
                }


                break;
            }
        }


        zip.close();


        return result;
    }


    // ============================================================
    // PRODUCT LIST
    // ============================================================

    private void showProductList() {

        loadProducts();


        if (
                productList.size() == 0
        ) {

            new AlertDialog.Builder(this)

                    .setTitle(
                            "Product List"
                    )

                    .setMessage(
                            "No products added yet."
                    )

                    .setPositiveButton(
                            "OK",
                            null
                    )

                    .show();

            return;
        }


        StringBuilder message =
                new StringBuilder();


        for (
                int i = 0;
                i < productList.size();
                i++
        ) {

            message.append(
                    i + 1
            );

            message.append(
                    ". "
            );

            message.append(
                    productList.get(i)
            );


            if (
                    i <
                    productList.size() - 1
            ) {

                message.append(
                        "\n"
                );
            }
        }


        ScrollView scroll =
                new ScrollView(this);


        TextView text =
                new TextView(this);


        text.setText(
                message.toString()
        );


        text.setTextSize(
                18
        );


        text.setPadding(
                20,
                20,
                20,
                20
        );


        scroll.addView(
                text
        );


        new AlertDialog.Builder(this)

                .setTitle(
                        "Product List"
                )

                .setView(
                        scroll
                )

                .setPositiveButton(
                        "OK",
                        null
                )

                .show();
    }


    // ============================================================
    // REMOVE PRODUCT
    // ============================================================

    private void showRemoveProduct() {

        loadProducts();


        if (
                productList.size() == 0
        ) {

            new AlertDialog.Builder(this)

                    .setTitle(
                            "Remove Product"
                    )

                    .setMessage(
                            "No products available."
                    )

                    .setPositiveButton(
                            "OK",
                            null
                    )

                    .show();

            return;
        }


        final String[] products =
                productList.toArray(
                        new String[
                                productList.size()
                        ]
                );


        new AlertDialog.Builder(this)

                .setTitle(
                        "Remove Product"
                )

                .setItems(
                        products,
                        new DialogInterface.OnClickListener() {

                    @Override
                    public void onClick(
                            DialogInterface dialog,
                            final int which
                    ) {

                        final String product =
                                products[which];


                        new AlertDialog.Builder(
                                ProductManagement.this
                        )

                                .setTitle(
                                        "Remove Product"
                                )

                                .setMessage(
                                        "Remove \"" +
                                        product +
                                        "\"?"
                                )

                                .setPositiveButton(
                                        "REMOVE",
                                        new DialogInterface.OnClickListener() {

                                    @Override
                                    public void onClick(
                                            DialogInterface dialog,
                                            int which
                                    ) {

                                        boolean removed =
                                                ButtonsController
                                                        .removeProduct(
                                                                ProductManagement.this,
                                                                product
                                                        );


                                        if (removed) {

                                            loadProducts();


                                            Toast.makeText(
                                                    ProductManagement.this,
                                                    "Product Removed",
                                                    Toast.LENGTH_SHORT
                                            ).show();

                                        } else {

                                            Toast.makeText(
                                                    ProductManagement.this,
                                                    "Remove Failed",
                                                    Toast.LENGTH_SHORT
                                            ).show();
                                        }
                                    }
                                })

                                .setNegativeButton(
                                        "CANCEL",
                                        null
                                )

                                .show();
                    }
                })

                .show();
    }
