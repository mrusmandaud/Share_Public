    /*

     * ---------------------------------------------------------------

     * SECTION: RECORD SAVE

     * PURPOSE: Validates and saves a new record to the existing records file.

     * ACTIVITY / VIEW CONNECTION: Activity: NewrecordActivity. Views: nameView, phoneView, addressView, productSpinner, confirm.

     * WORK FLOW: Connection: SAVE button onClick / More Block → ButtonsController.saveRecord(activity, nameEditText, phoneEditText, addressEditText, productSpinner, confirmCheckBox).

     * INPUT / OUTPUT: Inputs: Activity + four form controls. Return: boolean success/failure.

     * DETAILS: Verified behavior: confirmation, Name, Phone, Address and Product are validated; product placeholders Select Product/Select Products are rejected; data is appended to myrecords.txt.

     * ---------------------------------------------------------------

     */

    public static boolean saveRecord(
            Activity activity,
            EditText nameView,
            EditText phoneView,
            EditText addressView,
            Spinner productSpinner,
            CheckBox confirm
    ) {

        if (activity == null) {
            return false;
        }

        if (
                confirm == null
                ||
                !confirm.isChecked()
        ) {

            Toast.makeText(
                    activity,
                    "Please confirm and agree to save",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        String name = "";

        if (
                nameView != null
                &&
                nameView.getText() != null
        ) {

            name =
                    nameView
                            .getText()
                            .toString()
                            .trim();
        }


        String phone = "";

        if (
                phoneView != null
                &&
                phoneView.getText() != null
        ) {

            phone =
                    phoneView
                            .getText()
                            .toString()
                            .trim();
        }


        String address = "";

        if (
                addressView != null
                &&
                addressView.getText() != null
        ) {

            address =
                    addressView
                            .getText()
                            .toString()
                            .trim();
        }


        String product = "";

        if (
                productSpinner != null
                &&
                productSpinner.getSelectedItem() != null
        ) {

            product =
                    productSpinner
                            .getSelectedItem()
                            .toString()
                            .trim();
        }


        if (name.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Name",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (phone.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Phone",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (address.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Address",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (
                product.length() == 0
                ||
                product.equalsIgnoreCase(
                        "Select Product"
                )
                ||
                product.equalsIgnoreCase(
                        "Select Products"
                )
        ) {

            Toast.makeText(
                    activity,
                    "Please select Product",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        String record =
                name
                        + " | "
                        + phone
                        + " | "
                        + address
                        + " | "
                        + product
                        + "\n";


        FileOutputStream output = null;


        try {

            output =
                    activity.openFileOutput(
                            RECORDS_FILE,
                            Context.MODE_APPEND
                    );

            output.write(
                    record.getBytes("UTF-8")
            );

            output.flush();


            Toast.makeText(
                    activity,
                    "Record saved successfully",
                    Toast.LENGTH_SHORT
            ).show();


            clearRecord(
                    nameView,
                    phoneView,
                    addressView,
                    productSpinner,
                    confirm
            );


            return true;


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to save record",
                    Toast.LENGTH_SHORT
            ).show();

            return false;


        } finally {

            if (output != null) {

                try {
                    output.close();
                } catch (Exception ignored) {
                }
            }
        }
    }

    /*

     * ---------------------------------------------------------------

     * SECTION: RECORD DATA ACCESS

     * PURPOSE: Reads existing saved records from the application file.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity or another screen that needs the stored records. View/Button: none directly.

     * WORK FLOW: Connection: Activity/controller logic → ButtonsController.getRecords(activity).

     * INPUT / OUTPUT: Input: Activity. Return: ArrayList<String> containing non-empty UTF-8 record lines.

     * DETAILS: Verified file: myrecords.txt.

     * ---------------------------------------------------------------

     */

    public static ArrayList<String> getRecords(
            Activity activity
    ) {

        ArrayList<String> records =
                new ArrayList<String>();


        if (activity == null) {
            return records;
        }


        BufferedReader reader = null;


        try {

            FileInputStream input =
                    activity.openFileInput(
                            RECORDS_FILE
                    );


            reader =
                    new BufferedReader(
                            new InputStreamReader(
                                    input,
                                    "UTF-8"
                            )
                    );


            String line;


            while (
                    (line = reader.readLine()) != null
            ) {

                line =
                        line.trim();


                if (line.length() > 0) {

                    records.add(line);
                }
            }


        } catch (Exception e) {

        } finally {

            if (reader != null) {

                try {
                    reader.close();
                } catch (Exception ignored) {
                }
            }
        }


        return records;
    }

    /*
     * ---------------------------------------------------------------
     * SECTION: SELECTION STATE MODEL
     * PURPOSE: Holds the current record/cell selection structure for
     *          one Activity.
     * ACTIVITY / VIEW CONNECTION: MyrecordsActivity; the state is
     * created from the ArrayList<ArrayList<TextView>> stored in the
     * existing recordsContainer tag.
     * WORK FLOW: selection() creates the state → selection actions
     * update it → copy/share/popup/edit logic reads it.
     * INPUT / OUTPUT: record cell lists in; internal SelectionState out.
     * DETAILS: Internal class only. No XML ID or More Block should call
     * this class directly.
     * ---------------------------------------------------------------
     */
    private static class SelectionState {

        final ArrayList<ArrayList<TextView>>
                records;

        final ArrayList<TextView>
                cells;

        final ArrayList<Integer>
                cellRecordIndexes;


        int firstIndex = -1;

        int lastIndex = -1;

        int firstRecord = -1;

        int lastRecord = -1;


        SelectionState(
                ArrayList<ArrayList<TextView>>
                        records
        ) {

            this.records =
                    records;

            this.cells =
                    new ArrayList<TextView>();

            this.cellRecordIndexes =
                    new ArrayList<Integer>();


            if (records == null) {
                return;
            }


            for (
                    int r = 0;
                    r < records.size();
                    r++
            ) {

                ArrayList<TextView> record =
                        records.get(r);


                if (record == null) {
                    continue;
                }


                for (
                        int c = 0;
                        c < record.size();
                        c++
                ) {

                    TextView cell =
                            record.get(c);


                    if (cell != null) {

                        cells.add(cell);

                        cellRecordIndexes.add(r);
                    }
                }
            }
        }
    }

    // ============================================================
    // GET RECORD INDEX FOR CELL
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION HELPERS

     * PURPOSE: Maps a flattened cell index to its record index.

     * ACTIVITY / VIEW CONNECTION: Activity/View: internal only.

     * WORK FLOW: Connection: selection logic.

     * INPUT / OUTPUT: Inputs: SelectionState + cell index. Return: record index or -1.

     * DETAILS: No direct UI connection.

     * ---------------------------------------------------------------

     */

    private static int getRecordIndexForCell(
            SelectionState state,
            int cellIndex
    ) {

        if (state == null) {
            return -1;
        }


        if (
                cellIndex < 0
                ||
                cellIndex >=
                        state.cellRecordIndexes.size()
        ) {

            return -1;
        }


        return state.cellRecordIndexes.get(
                cellIndex
        );
    }

     */

    private static int getRecordIndexForView(
            Activity activity,
            TextView target
    ) {

        if (
                activity == null
                ||
                target == null
        ) {
            return -1;
        }


        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return -1;
        }


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                if (record.get(c) == target) {
                    return r;
                }
            }
        }


        return -1;
    }


    // ============================================================
    // GET CELL POSITION IN RECORD
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION HELPERS

     * PURPOSE: Finds a cell's column position inside a record.

     * ACTIVITY / VIEW CONNECTION: Activity/View: internal.

     * WORK FLOW: Connection: range-selection logic.

     * INPUT / OUTPUT: Inputs: SelectionState + record index + TextView. Return: column index or -1.

     * DETAILS: No direct XML connection.

     * ---------------------------------------------------------------

     */

    private static int getCellPositionInRecord(
            SelectionState state,
            int recordIndex,
            TextView target
    ) {

        if (
                state == null
                ||
                target == null
                ||
                recordIndex < 0
                ||
                recordIndex >= state.records.size()
        ) {

            return -1;
        }


        ArrayList<TextView> record =
                state.records.get(
                        recordIndex
                );


        if (record == null) {
            return -1;
        }


        for (
                int i = 0;
                i < record.size();
                i++
        ) {

            if (record.get(i) == target) {
                return i;
            }
        }


        return -1;
    }


    // ============================================================
    // SET CELL SELECTED
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION UI

     * PURPOSE: Applies or removes the existing teal selection appearance.

     * ACTIVITY / VIEW CONNECTION: View: record TextView cell.

     * WORK FLOW: Connection: selection(), selectAll(), deselectState() and range-selection helpers.

     * INPUT / OUTPUT: Inputs: TextView + selected flag. Changes: background/text colors.

     * DETAILS: Verified selection color: #009688 (Color.rgb(0,150,136)); selected text is white and unselected text is black.

     * ---------------------------------------------------------------

     */

    private static void setCellSelected(
            TextView cell,
            boolean selected
    ) {

        if (cell == null) {
            return;
        }


        if (selected) {

            cell.setBackgroundColor(
                    SELECTED_COLOR
            );

            cell.setTextColor(
                    WHITE
            );

        } else {

            cell.setBackgroundColor(
                    WHITE
            );

            cell.setTextColor(
                    BLACK
            );
        }
    }


    // ============================================================
    // SELECT RECORD
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION HELPERS

     * PURPOSE: Selects every cell in one record row.

     * ACTIVITY / VIEW CONNECTION: View: record cell TextViews.

     * WORK FLOW: Connection: selectAll() and selectRecordRange().

     * INPUT / OUTPUT: Input: ArrayList<TextView>. Return: none.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void selectRecord(
            ArrayList<TextView> record
    ) {

        if (record == null) {
            return;
        }


        for (
                int i = 0;
                i < record.size();
                i++
        ) {

            setCellSelected(
                    record.get(i),
                    true
            );
        }
    }


    // ============================================================
    // CLEAR SELECTION
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION HELPERS

     * PURPOSE: Clears the visual selection from all cells in the selection state.

     * ACTIVITY / VIEW CONNECTION: View: record cells.

     * WORK FLOW: Connection: selection(), deselectState(), range-selection logic.

     * INPUT / OUTPUT: Input: SelectionState. Return: none.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void clearSelection(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }


        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            setCellSelected(
                    state.cells.get(i),
                    false
            );
        }
    }


    // ============================================================
    // DESELECT STATE
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION

     * PURPOSE: Clears selection and resets the stored selection boundaries.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: record cells.

     * WORK FLOW: Connection: deselectAll() and single-cell toggle behavior inside selection().

     * INPUT / OUTPUT: Input: SelectionState. Changes: selected colors and selection indexes.

     * DETAILS: Internal helper; user-facing action is normally through deselectAll().

     * ---------------------------------------------------------------

     */

    private static void deselectState(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }


        clearSelection(
                state
        );


        state.firstIndex =
                -1;

        state.lastIndex =
                -1;

        state.firstRecord =
                -1;

        state.lastRecord =
                -1;
    }


    // ============================================================
    // SELECT RECORD RANGE
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION HELPERS

     * PURPOSE: Selects all record rows between two record indexes.

     * ACTIVITY / VIEW CONNECTION: View: record cell TextViews.

     * WORK FLOW: Connection: normal selection range behavior inside selection().

     * INPUT / OUTPUT: Inputs: SelectionState + start/end record indexes.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void selectRecordRange(
            SelectionState state,
            int startRecord,
            int endRecord
    ) {

        if (state == null) {
            return;
        }


        if (startRecord > endRecord) {

            int temp =
                    startRecord;

            startRecord =
                    endRecord;

            endRecord =
                    temp;
        }


        for (
                int r = startRecord;
                r <= endRecord;
                r++
        ) {

            if (
                    r < 0
                    ||
                    r >= state.records.size()
            ) {

                continue;
            }


            selectRecord(
                    state.records.get(r)
            );
        }
    }


    // ============================================================
    // SELECTION
    // ============================================================

    @SuppressWarnings("unchecked")
    /*
     * ---------------------------------------------------------------
     * SECTION: SELECTION CONTROLLER
     * PURPOSE: Connects the recordsContainer cells to normal selection, range selection, long-press popup, and edit-mode behavior.
     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. Verified View ID: R.id.recordsContainer.
     * WORK FLOW: Connection: Activity onCreate / More Block after records are rendered → ButtonsController.selection(activity). The recordsContainer tag must contain the existing ArrayList<ArrayList<TextView>> structure.
     * INPUT / OUTPUT: Behavior: single-cell select/deselect, same-row range selection, multi-row range selection, edit-mode cell focus, and long press → popupMenu(activity).
     * DETAILS: Important: this is the main selection hookup; do not duplicate competing listeners in the Activity unless intentionally coordinated.
     * ---------------------------------------------------------------
     */
    public static void selection(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }


        final LinearLayout
                selectionContainer =
                activity.findViewById(
                        R.id.recordsContainer
                );


        if (selectionContainer == null) {
            return;
        }


        Object selectionObject =
                selectionContainer.getTag();


        if (!(selectionObject instanceof ArrayList)) {

            SELECTION_STATES.remove(
                    activity
            );

            return;
        }


        final ArrayList<ArrayList<TextView>>
                selectionRecords =
                (ArrayList<ArrayList<TextView>>)
                        selectionObject;


        if (selectionRecords.size() == 0) {

            SELECTION_STATES.remove(
                    activity
            );

            return;
        }


        final SelectionState state =
                new SelectionState(
                        selectionRecords
                );


        if (state.cells.size() == 0) {

            SELECTION_STATES.remove(
                    activity
            );

            return;
        }


        SELECTION_STATES.put(
                activity,
                state
        );


        clearSelection(
                state
        );


        final EditSession editSession =
                getEditSession(
                        activity
                );


        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            final TextView cell =
                    state.cells.get(i);


            if (cell == null) {
                continue;
            }


            final int index =
                    i;


            cell.setClickable(
                    true
            );


            // ====================================================
            // EDIT MODE
            // ====================================================

            if (
                    editSession != null
                    &&
                    editSession.editMode
            ) {

                cell.setOnClickListener(
                        new View.OnClickListener() {

                            @Override
                            public void onClick(
                                    View v
                            ) {

                                if (v instanceof EditText) {

                                    EditText editText =
                                            (EditText) v;

                                    editSession.focusedCell =
                                            editText;

                                    editText.requestFocus();

                                    editText.setSelection(
                                            editText.length()
                                    );
                                }
                            }
                        }
                );


                cell.setOnLongClickListener(
                        new View.OnLongClickListener() {

                            @Override
                            public boolean onLongClick(
                                    View v
                            ) {

                                if (v instanceof EditText) {

                                    EditText editText =
                                            (EditText) v;

                                    editSession.focusedCell =
                                            editText;

                                    editText.requestFocus();

                                    editText.setSelection(
                                            editText.length()
                                    );
                                }

                                return true;
                            }
                        }
                );

                continue;
            }


            // ====================================================
            // NORMAL SELECTION MODE
            // ====================================================

            cell.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            SelectionState currentState =
                                    getSelectionState(
                                            activity
                                    );


                            if (currentState == null) {
                                return;
                            }


                            int tappedRecord =
                                    getRecordIndexForCell(
                                            currentState,
                                            index
                                    );


                            if (tappedRecord < 0) {
                                return;
                            }


                            if (
                                    currentState.firstIndex
                                            == -1
                            ) {

                                clearSelection(
                                        currentState
                                );


                                setCellSelected(
                                        cell,
                                        true
                                );


                                currentState.firstIndex =
                                        index;

                                currentState.lastIndex =
                                        index;

                                currentState.firstRecord =
                                        tappedRecord;

                                currentState.lastRecord =
                                        tappedRecord;


                                return;
                            }


                            if (
                                    currentState.firstIndex
                                            ==
                                            currentState.lastIndex
                                    &&
                                    index
                                            ==
                                            currentState.firstIndex
                            ) {

                                deselectState(
                                        currentState
                                );

                                return;
                            }


                            int firstRecord =
                                    currentState.firstRecord;


                            if (firstRecord < 0) {

                                firstRecord =
                                        getRecordIndexForCell(
                                                currentState,
                                                currentState.firstIndex
                                        );
                            }


                            if (
                                    firstRecord
                                            ==
                                            tappedRecord
                            ) {

                                ArrayList<TextView>
                                        record =
                                        currentState.records
                                                .get(
                                                        tappedRecord
                                                );


                                int firstPosition =
                                        getCellPositionInRecord(
                                                currentState,
                                                tappedRecord,
                                                currentState.cells
                                                        .get(
                                                                currentState.firstIndex
                                                        )
                                        );


                                int secondPosition =
                                        getCellPositionInRecord(
                                                currentState,
                                                tappedRecord,
                                                cell
                                        );


                                if (
                                        firstPosition < 0
                                        ||
                                        secondPosition < 0
                                ) {

                                    return;
                                }


                                int start =
                                        firstPosition;

                                int end =
                                        secondPosition;


                                if (start > end) {

                                    int temp =
                                            start;

                                    start =
                                            end;

                                    end =
                                            temp;
                                }


                                clearSelection(
                                        currentState
                                );


                                for (
                                        int c = start;
                                        c <= end;
                                        c++
                                ) {

                                    if (
                                            c >= 0
                                            &&
                                            c < record.size()
                                    ) {

                                        setCellSelected(
                                                record.get(c),
                                                true
                                        );
                                    }
                                }


                                currentState.firstRecord =
                                        tappedRecord;

                                currentState.lastRecord =
                                        tappedRecord;

                                currentState.firstIndex =
                                        index;

                                currentState.lastIndex =
                                        index;


                                return;
                            }


                            int startRecord =
                                    firstRecord;

                            int endRecord =
                                    tappedRecord;


                            if (
                                    startRecord
                                            >
                                            endRecord
                            ) {

                                int temp =
                                        startRecord;

                                startRecord =
                                        endRecord;

                                endRecord =
                                        temp;
                            }


                            clearSelection(
                                    currentState
                            );


                            selectRecordRange(
                                    currentState,
                                    startRecord,
                                    endRecord
                            );


                            currentState.firstRecord =
                                    startRecord;

                            currentState.lastRecord =
                                    endRecord;

                            currentState.firstIndex =
                                    index;

                            currentState.lastIndex =
                                    index;
                        }
                    }
            );


            // ====================================================
            // LONG PRESS → POPUP
            // ====================================================

            cell.setOnLongClickListener(
                    new View.OnLongClickListener() {

                        @Override
                        public boolean onLongClick(
                                View v
                        ) {

                            popupMenu(
                                    activity
                            );

                            return true;
                        }
                    }
            );
        }


        editMenu(
                activity
        );
    }


    // ============================================================
    // SELECT ALL
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION ACTION

     * PURPOSE: Selects all record cells currently represented by the SelectionState.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: recordsContainer data cells.

     * WORK FLOW: Connection: Selection popup/menu Select All button → ButtonsController.selectAll(activity).

     * INPUT / OUTPUT: Input: Activity. Action: selects all records and updates selection boundaries.

     * DETAILS: No separate XML ID is referenced inside this method.

     * ---------------------------------------------------------------

     */

    public static void selectAll(
            Activity activity
    ) {

        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return;
        }


        if (state.cells.size() == 0) {
            return;
        }


        clearSelection(
                state
        );


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            selectRecord(
                    state.records.get(r)
            );
        }


        state.firstRecord =
                0;

        state.lastRecord =
                state.records.size() - 1;


        state.firstIndex =
                0;

        state.lastIndex =
                state.cells.size() - 1;
    }


    // ============================================================
    // DESELECT ALL
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION ACTION

     * PURPOSE: Clears all current record selection.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: recordsContainer data cells.

     * WORK FLOW: Connection: Selection popup/menu Deselect All button → ButtonsController.deselectAll(activity).

     * INPUT / OUTPUT: Input: Activity. Action: clears selected cells and resets selection state.

     * DETAILS: No separate XML ID is referenced inside this method.

     * ---------------------------------------------------------------

     */

    public static void deselectAll(
            Activity activity
    ) {

        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return;
        }


        deselectState(
                state
        );
    }


    // ============================================================
    // CHECK SELECTED
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION HELPERS

     * PURPOSE: Checks whether a cell currently uses the selected background color.

     * ACTIVITY / VIEW CONNECTION: View: record TextView/EditText cell.

     * WORK FLOW: Connection: internal selection/edit/export/copy logic.

     * INPUT / OUTPUT: Input: TextView. Return: boolean.

     * DETAILS: Verified against SELECTED_COLOR.

     * ---------------------------------------------------------------

     */

    private static boolean isSelected(
            TextView cell
    ) {

        if (cell == null) {
            return false;
        }


        if (
                cell.getBackground()
                        instanceof ColorDrawable
        ) {

            ColorDrawable background =
                    (ColorDrawable)
                            cell.getBackground();


            return background.getColor()
                    ==
                    SELECTED_COLOR;
        }


        return false;
    }


    // ============================================================
    // GET SELECTED TEXT
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION DATA

     * PURPOSE: Builds the text representation of currently selected records.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: selected record cells.

     * WORK FLOW: Connection: copy() and share(); indirectly supports selected-data actions.

     * INPUT / OUTPUT: Input: Activity. Return: selected records as text separated by record/column delimiters.

     * DETAILS: Internal helper; no direct button hookup.

     * ---------------------------------------------------------------

     */

    private static String getSelectedText(
            Activity activity
    ) {

        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return "";
        }


        StringBuilder selectedText =
                new StringBuilder();


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            boolean recordSelected =
                    false;


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);


                if (
                        cell != null
                        &&
                        isSelected(cell)
                ) {

                    recordSelected =
                            true;

                    break;
                }
            }


            if (!recordSelected) {
                continue;
            }


            StringBuilder recordText =
                    new StringBuilder();


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);


                if (cell == null) {
                    continue;
                }


                if (
                        recordText.length() > 0
                ) {

                    recordText.append(
                            " | "
                    );
                }


                recordText.append(
                        cell.getText()
                                .toString()
                );
            }


            if (
                    recordText.length() == 0
            ) {

                continue;
            }


            if (
                    selectedText.length() > 0
            ) {

                selectedText.append(
                        "\n"
                );
            }


            selectedText.append(
                    recordText.toString()
            );
        }


        return selectedText.toString();
    }


    // ============================================================
    // COPY
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION ACTION / COPY

     * PURPOSE: Copies selected record data to the Android clipboard.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View/Button: Copy action in the selection popup/menu.

     * WORK FLOW: Connection: popup Copy button → ButtonsController.copy(activity).

     * INPUT / OUTPUT: Input: Activity. Action: copies selected text; shows existing Toast messages.

     * DETAILS: Requires selected data; no direct XML ID is hard-coded here.

     * ---------------------------------------------------------------

     */

    public static void copy(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }


        String selectedText =
                getSelectedText(
                        activity
                );


        if (
                selectedText == null
                ||
                selectedText.length() == 0
        ) {

            Toast.makeText(
                    activity,
                    "No data selected",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        ClipboardManager clipboard =
                (ClipboardManager)
                        activity.getSystemService(
                                Context.CLIPBOARD_SERVICE
                        );


        if (clipboard == null) {

            Toast.makeText(
                    activity,
                    "Clipboard unavailable",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        clipboard.setPrimaryClip(
                ClipData.newPlainText(
                        "Selected Records",
                        selectedText
                )
        );


        Toast.makeText(
                activity,
                "Copied",
                Toast.LENGTH_SHORT
        ).show();
    }


    // ============================================================
    // SHARE
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: SELECTION ACTION / SHARE

     * PURPOSE: Shares selected record data using Android's text sharing chooser.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View/Button: Share action in the selection popup/menu.

     * WORK FLOW: Connection: popup Share button → ButtonsController.share(activity).

     * INPUT / OUTPUT: Input: Activity. Action: ACTION_SEND with text/plain and selected data.

     * DETAILS: Requires selected data; no direct XML ID is hard-coded here.

     * ---------------------------------------------------------------

     */

    public static void share(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }


        String selectedText =
                getSelectedText(
                        activity
                );


        if (
                selectedText == null
                ||
                selectedText.length() == 0
        ) {

            Toast.makeText(
                    activity,
                    "No data selected",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        try {

            Intent shareIntent =
                    new Intent(
                            Intent.ACTION_SEND
                    );


            shareIntent.setType(
                    "text/plain"
            );


            shareIntent.putExtra(
                    Intent.EXTRA_TEXT,
                    selectedText
            );


            activity.startActivity(
                    Intent.createChooser(
                            shareIntent,
                            "Share"
                    )
            );


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to share",
                    Toast.LENGTH_SHORT
            ).show();
        }
    }


    // ============================================================
    // EDIT
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT MODE ENTRY

     * PURPOSE: Enters or exits record edit mode and prepares the current records for editing.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View/Button: main Edit Menu trigger.

     * WORK FLOW: Connection: Activity Edit action → ButtonsController.edit(activity).

     * INPUT / OUTPUT: Input: Activity. Action: loads records, stores edit session state, converts cells to editable controls, restores prior selection, and shows the Edit popup.

     * DETAILS: Verified View dependency: R.id.recordsContainer; popup anchor is connected later through editMenu()/showEditPopup().

     * ---------------------------------------------------------------

     */

    public static void edit(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }


        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );


        if (container == null) {
            return;
        }


        Object object =
                container.getTag();


        if (!(object instanceof ArrayList)) {

            Toast.makeText(
                    activity,
                    "No records available",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        ArrayList<ArrayList<TextView>>
                taggedRecords =
                (ArrayList<ArrayList<TextView>>)
                        object;


        if (taggedRecords.size() == 0) {

            Toast.makeText(
                    activity,
                    "No records available",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        EditSession session =
                getEditSession(
                        activity
                );


        if (session == null) {
            return;
        }


        if (session.editMode) {

            // Second tap on Edit Menu exits edit mode and closes the popup.
            closeEditSession(
                    activity
            );

            return;
        }


        // Remember the normal-mode cell selection before replacing
        // TextViews with EditTexts.
        session.preEditSelectedCells.clear();

        SelectionState oldState =
                getSelectionState(
                        activity
                );

        if (oldState != null) {
            for (int r = 0; r < oldState.records.size(); r++) {
                ArrayList<TextView> oldRecord = oldState.records.get(r);
                if (oldRecord == null) continue;
                for (int c = 0; c < oldRecord.size(); c++) {
                    TextView oldCell = oldRecord.get(c);
                    if (oldCell != null && isSelected(oldCell)) {
                        session.preEditSelectedCells.add(r + ":" + c);
                    }
                }
            }
        }

        session.editMode =
                true;

        session.deletedRows.clear();

        session.removedCells.clear();

        session.originalRecords.clear();

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();
        session.rowsFrozen = false;

        session.focusedCell =
                null;


        ArrayList<String> records =
                getRecords(
                        activity
                );


        session.originalRecords.addAll(
                records
        );


        makeCellsEditable(
                activity
        );


        SelectionState currentState =
                getSelectionState(
                        activity
                );


        freezeRecordRowsForEdit(
                currentState,
                session
        );


        clearSelection(
                currentState
        );


        restorePreEditSelection(
                activity
        );


        showEditPopup(
                activity
        );
    }


    // ============================================================
    // RESTORE PRE-EDIT SELECTION
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT / SELECTION BRIDGE

     * PURPOSE: Restores cell selection saved before entering edit mode.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. Views: record cells.

     * WORK FLOW: Connection: edit() → restorePreEditSelection(activity).

     * INPUT / OUTPUT: Input: Activity. Action: reapplies stored record:column selections.

     * DETAILS: Internal helper; no direct XML connection.

     * ---------------------------------------------------------------

     */

    private static void restorePreEditSelection(
            Activity activity
    ) {

        if (activity == null) return;

        EditSession session = getEditSession(activity);
        SelectionState state = getSelectionState(activity);

        if (
                session == null
                ||
                state == null
                ||
                session.preEditSelectedCells.isEmpty()
        ) return;

        for (String key : session.preEditSelectedCells) {
            int separator = key.indexOf(":");
            if (separator <= 0) continue;

            try {
                int row = Integer.parseInt(key.substring(0, separator));
                int column = Integer.parseInt(key.substring(separator + 1));

                if (row >= 0 && row < state.records.size()) {
                    ArrayList<TextView> record = state.records.get(row);
                    if (
                            record != null
                            &&
                            column >= 0
                            &&
                            column < record.size()
                    ) {
                        setCellSelected(record.get(column), true);
                    }
                }
            } catch (Exception ignored) {
            }
        }
    }


    // ============================================================
    // MAKE CELLS EDITABLE
    // ============================================================

    @SuppressWarnings("unchecked")
    /*
     * ---------------------------------------------------------------
     * SECTION: EDIT MODE / CELL CONVERSION
     * PURPOSE: Converts record TextViews to editable controls while preserving their layout and displayed content.
     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. Verified View ID: R.id.recordsContainer.
     * WORK FLOW: Connection: edit() → makeCellsEditable(activity).
     * INPUT / OUTPUT: Input: Activity. Action: replaces cells, configures input, updates container tag, then reconnects selection().
     * DETAILS: Internal controller operation; Activity only needs to invoke edit().
     * ---------------------------------------------------------------
     */
    private static void makeCellsEditable(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }


        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );


        if (container == null) {
            return;
        }


        Object object =
                container.getTag();


        if (!(object instanceof ArrayList)) {
            return;
        }


        ArrayList<ArrayList<TextView>>
                records =
                (ArrayList<ArrayList<TextView>>)
                        object;


        final EditSession session =
                getEditSession(
                        activity
                );


        for (
                int r = 0;
                r < records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    records.get(r);


            if (record == null) {
                continue;
            }


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView oldCell =
                        record.get(c);


                if (oldCell == null) {
                    continue;
                }


                if (
                        oldCell instanceof EditText
                ) {

                    EditText existing =
                            (EditText)
                                    oldCell;


                    configureEditTextInput(
                            existing,
                            c
                    );


                    existing.setFocusable(
                            true
                    );

                    existing.setFocusableInTouchMode(
                            true
                    );

                    existing.setCursorVisible(
                            true
                    );

                    existing.setEnabled(
                            true
                    );


                    final EditSession
                            currentSession =
                            session;


                    existing.setOnFocusChangeListener(
                            new View.OnFocusChangeListener() {

                                @Override
                                public void onFocusChange(
                                        View v,
                                        boolean hasFocus
                                ) {

                                    if (
                                            hasFocus
                                            &&
                                            currentSession != null
                                    ) {

                                        currentSession.focusedCell =
                                                (TextView) v;
                                    }
                                }
                            }
                    );

                    continue;
                }


                ViewParent parentObject =
                        oldCell.getParent();


                if (!(parentObject instanceof ViewGroup)) {
                    continue;
                }


                ViewGroup parent =
                        (ViewGroup)
                                parentObject;


                int position =
                        parent.indexOfChild(
                                oldCell
                        );


                if (position < 0) {
                    continue;
                }


                EditText editText =
                        new EditText(
                                activity
                        );


                editText.setText(
                        oldCell.getText()
                );


                editText.setTextColor(
                        BLACK
                );


                editText.setTextSize(
                        android.util.TypedValue
                                .COMPLEX_UNIT_PX,
                        oldCell.getTextSize()
                );


                editText.setGravity(
                        oldCell.getGravity()
                );


                editText.setPadding(
                        oldCell.getPaddingLeft(),
                        oldCell.getPaddingTop(),
                        oldCell.getPaddingRight(),
                        oldCell.getPaddingBottom()
                );


                editText.setTypeface(
                        oldCell.getTypeface()
                );


                editText.setIncludeFontPadding(
                        oldCell.getIncludeFontPadding()
                );


                editText.setMinHeight(
                        oldCell.getMinHeight()
                );


                editText.setMinWidth(
                        oldCell.getMinWidth()
                );


                editText.setBackground(
                        oldCell.getBackground()
                );


                configureEditTextInput(
                        editText,
                        c
                );


                editText.setFocusable(
                        true
                );


                editText.setFocusableInTouchMode(
                        true
                );


                editText.setCursorVisible(
                        true
                );


                editText.setEnabled(
                        true
                );


                ViewGroup.LayoutParams
                        params =
                        oldCell.getLayoutParams();


                editText.setLayoutParams(
                        params
                );


                final EditSession
                        currentSession =
                        session;


                editText.setOnFocusChangeListener(
                        new View.OnFocusChangeListener() {

                            @Override
                            public void onFocusChange(
                                    View v,
                                    boolean hasFocus
                            ) {

                                if (
                                        hasFocus
                                        &&
                                        currentSession != null
                                ) {

                                    currentSession.focusedCell =
                                            (TextView) v;
                                }
                            }
                        }
                );


                parent.removeViewAt(
                        position
                );


                parent.addView(
                        editText,
                        position
                );


                record.set(
                        c,
                        editText
                );
            }
        }


        container.setTag(
                records
        );


        selection(
                activity
        );
    }


    // ============================================================
    // CONFIGURE EDIT TEXT INPUT
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT MODE / INPUT RULES

     * PURPOSE: Applies the existing keyboard/input behavior by record column.

     * ACTIVITY / VIEW CONNECTION: View: EditText record cell.

     * WORK FLOW: Connection: makeCellsEditable() for every record column.

     * INPUT / OUTPUT: Inputs: EditText + column index. Changes: input type, line behavior and wrapping.

     * DETAILS: Column mapping is preserved from the current source; no external XML hookup.

     * ---------------------------------------------------------------

     */

    private static void configureEditTextInput(
            EditText editText,
            int column
    ) {

        if (editText == null) {
            return;
        }


        // Keep record cells wrapped while editing instead of
        // forcing them into one long horizontal line.
        editText.setHorizontallyScrolling(
                false
        );

        editText.setMaxLines(
                2
        );


        if (column == 0) {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_WORDS
            );

            editText.setSingleLine(
                    false
            );

        }


        else if (column == 1) {

            editText.setInputType(
                    InputType.TYPE_CLASS_PHONE
            );

            editText.setSingleLine(
                    false
            );

        }


        else if (column == 2) {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_SENTENCES
                            |
                            InputType.TYPE_TEXT_FLAG_MULTI_LINE
            );

            editText.setSingleLine(
                    false
            );

        }


        else {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_SENTENCES
            );

            editText.setSingleLine(
                    false
            );
        }
    }


    // ============================================================
    // DISABLE EDITING
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT MODE / EXIT PREPARATION

     * PURPOSE: Disables active editing interaction on current editable cells without removing the controls.

     * ACTIVITY / VIEW CONNECTION: View: record EditText cells.

     * WORK FLOW: Connection: internal edit-session close/save flow.

     * INPUT / OUTPUT: Input: SelectionState. Action: clears focus and disables editing interaction.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void disableEditingOnCells(
            SelectionState state
    ) {

        if (state == null) {
            return;
        }


        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            TextView cell =
                    state.cells.get(i);


            if (
                    cell instanceof EditText
            ) {

                EditText editText =
                        (EditText)
                                cell;


                editText.clearFocus();

                editText.setCursorVisible(
                        false
                );

                editText.setFocusable(
                        false
                );

                editText.setFocusableInTouchMode(
                        false
                );

                editText.setEnabled(
                        true
                );
            }
        }
    }

/*

 * ---------------------------------------------------------------

 * SECTION: EDIT MENU POPUP / UI CONNECTION

 * PURPOSE: Builds and attaches the existing on-page Edit popup beside the Edit Menu button.

 * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. Main button ID verified: R.id.editMenuButton. Popup layout verified in code: R.layout.edit_popup_menu.

 * WORK FLOW: Connection: edit() → showEditPopup(activity). Popup controls verified: clearButton, deleteButton, saveButton, closeButton; container: editPopupContainer.

 * INPUT / OUTPUT: Actions: Clear → confirmClear(); Delete → delete(); Save → save(); Close → closeEditSession().

 * DETAILS: Architecture preserved: uses the existing real page FrameLayout wrapper; no PopupWindow/showAtLocation/showAsDropDown is introduced.

 * ---------------------------------------------------------------

 */

private static void showEditPopup(
        final Activity activity
) {

    if (activity == null) {
        return;
    }

    final EditSession session =
            getEditSession(activity);

    if (session == null) {
        return;
    }

    // Reuse the existing XML popup if it is already attached. Keeping the
    // popup in the real page layout (but INVISIBLE when closed) prevents the
    // record area from moving when Edit Menu opens/closes.
    View reusablePopup = null;

    if (session.editPopupWindow != null) {
        ViewParent existingParent = session.editPopupWindow.getParent();
        if (existingParent instanceof ViewGroup) {
            reusablePopup = session.editPopupWindow;
        } else {
            session.editPopupWindow = null;
        }
    }

    // Main Edit Menu button.
    final View anchor =
            activity.findViewById(
                    R.id.editMenuButton
            );

    if (anchor == null) {
        return;
    }

    final ViewParent parentObject =
            anchor.getParent();

    if (!(parentObject instanceof ViewGroup)) {
        return;
    }

    final ViewGroup originalParent =
            (ViewGroup) parentObject;

    // Inflate the real on-page submenu only when no XML placeholder exists.
    // All popup size, padding, spacing, margins and visual positioning remain
    // controlled by edit_popup_menu.xml.
    final View popupView;
    final View popupContainer;

    if (reusablePopup != null) {
        popupView = reusablePopup;
        popupContainer = reusablePopup;
    } else {
        popupView =
                LayoutInflater.from(activity).inflate(
                        R.layout.edit_popup_menu,
                        originalParent,
                        false
                );

        if (popupView == null) {
            return;
        }

        popupContainer =
                popupView.findViewById(
                        R.id.editPopupContainer
                );
    }

    if (popupContainer == null) {
        return;
    }

    // ------------------------------------------------------------
    // BUTTON ACTIONS
    // ------------------------------------------------------------

    View clearButton =
            popupView.findViewById(
                    R.id.clearButton
            );

    if (clearButton != null) {
        clearButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onRemove();
                        }
                    }
                }
        );
    }

    View deleteButton =
            popupView.findViewById(
                    R.id.deleteButton
            );

    if (deleteButton != null) {
        deleteButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onDelete();
                        }
                    }
                }
        );
    }

    View saveButton =
            popupView.findViewById(
                    R.id.saveButton
            );

    if (saveButton != null) {
        saveButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onSave();
                        }
                    }
                }
        );
    }

    View closeButton =
            popupView.findViewById(
                    R.id.closeButton
            );

    if (closeButton != null) {
        closeButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onClose();
                        } else {
                            closeEditSession(activity);
                        }
                    }
                }
        );
    }

    // ------------------------------------------------------------
    // IMPORTANT ARCHITECTURE
    // ------------------------------------------------------------
    //
    // Do NOT use PopupWindow.
    // Do NOT use showAtLocation().
    // Do NOT use showAsDropDown().
    // Do NOT use screen X/Y coordinates.
    //
    // Do NOT set popup width/height here.
    // Do NOT set popup left/right/top/bottom margins here.
    // Do NOT set popup padding or spacing here.
    // Do NOT set popup translation here.
    // Do NOT set popup elevation here.
    //
    // The Edit Menu XML and edit_popup_menu.xml control their own
    // size, margins, padding, spacing and vertical/horizontal layout.
    // The existing page parent remains the real container.
    //
    // The popup is inserted immediately after Edit Menu in the same
    // parent. The current parent is the horizontal Edit Menu row, so
    // the XML controls the exact appearance and dimensions.
    // ------------------------------------------------------------

    int originalIndex =
            originalParent.indexOfChild(anchor);

    if (originalIndex < 0) {
        return;
    }

    // Add only a newly inflated popup. A prepared XML placeholder is already
    // in the parent and is simply made visible.
    if (reusablePopup == null) {
        originalParent.addView(
                popupContainer,
                originalIndex + 1
        );
    }

    popupContainer.setVisibility(View.VISIBLE);
    originalParent.requestLayout();
    originalParent.invalidate();

    // Store the actual on-page submenu root.
    session.editPopupWindow = popupContainer;
}

/*

 * ---------------------------------------------------------------

 * SECTION: EDIT MENU POPUP / CLEANUP

 * PURPOSE: Removes the currently attached on-page edit popup container.

 * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: existing edit popup container.

 * WORK FLOW: Connection: internal edit-session close/cleanup flow.

 * INPUT / OUTPUT: Input: Activity. Action: removes popup view from its parent and clears session reference.

 * DETAILS: Internal helper.

 * ---------------------------------------------------------------

 */

private static void dismissEditPopup(
        Activity activity
) {

    if (activity == null) {
        return;
    }


    EditSession session =
            getEditSession(
                    activity
            );


    if (session == null) {
        return;
    }


    if (session.editPopupWindow != null) {

        try {
            ViewParent parentObject =
                    session.editPopupWindow.getParent();

            if (parentObject instanceof ViewGroup) {
                // Keep the XML popup in the layout as an invisible placeholder.
                // This preserves the exact page layout when Edit Menu disappears.
                session.editPopupWindow.setVisibility(View.INVISIBLE);
                ((ViewGroup) parentObject).requestLayout();
                ((ViewGroup) parentObject).invalidate();
            } else {
                session.editPopupWindow = null;
            }
        } catch (Exception ignored) {
        }
    }
}

 * WORK FLOW: Connection: Activity onCreate / More Block → ButtonsController.editMenu(activity).

 * INPUT / OUTPUT: Input: Activity. Action: installs the existing Edit Menu interaction.

 * DETAILS: Keep this as the central Edit Menu connection rather than duplicating edit listeners elsewhere.

 * ---------------------------------------------------------------

private static void prepareEditPopupPlaceholder(
        final Activity activity
) {

    if (activity == null) {
        return;
    }

    final EditSession session = getEditSession(activity);
    if (session == null || session.editPopupWindow != null) {
        return;
    }

    final View anchor = activity.findViewById(R.id.editMenuButton);
    if (anchor == null) {
        return;
    }

    final ViewParent parentObject = anchor.getParent();
    if (!(parentObject instanceof ViewGroup)) {
        return;
    }

    final ViewGroup originalParent = (ViewGroup) parentObject;

    try {
        final View popupView =
                LayoutInflater.from(activity).inflate(
                        R.layout.edit_popup_menu,
                        originalParent,
                        false
                );

        if (popupView == null) {
            return;
        }

        final View popupContainer =
                popupView.findViewById(R.id.editPopupContainer);

        if (popupContainer == null) {
            return;
        }

        int originalIndex = originalParent.indexOfChild(anchor);
        if (originalIndex < 0) {
            return;
        }

        popupContainer.setVisibility(View.INVISIBLE);
        originalParent.addView(
                popupContainer,
                originalIndex + 1
        );

        session.editPopupWindow = popupContainer;
        originalParent.requestLayout();
        originalParent.invalidate();
    } catch (Exception ignored) {
    }
}

/*

 * ---------------------------------------------------------------

 * SECTION: EDIT SESSION / CLOSE

 * PURPOSE: Ends edit mode and closes/restores the existing edit UI state.

 * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View/Button: Close in the Edit popup.

 * WORK FLOW: Connection: edit popup closeButton → closeEditSession(activity); also used by edit() when toggling edit mode.

 * INPUT / OUTPUT: Input: Activity. Action: closes the edit session and restores the normal screen state according to existing code.

 * DETAILS: Internal edit-session lifecycle method.

 * ---------------------------------------------------------------

 */

private static void clearEditActionFocus(
        Activity activity
) {

    if (activity == null) {
        return;
    }

    try {
        // Clear input focus only. Never request focus on recordsContainer.
        // Requesting focus there can force the parent ScrollView to jump.
        View currentFocus = activity.getCurrentFocus();

        if (currentFocus != null) {
            currentFocus.clearFocus();
        }

        activity.getWindow().setSoftInputMode(
                android.view.WindowManager.LayoutParams.SOFT_INPUT_STATE_ALWAYS_HIDDEN
        );
    } catch (Exception ignored) {
    }
}

private static void closeEditSession(
        Activity activity
) {

    if (activity == null) {
        return;
    }


    EditSession session =
            getEditSession(
                    activity
            );


    if (
            session == null
            ||
            !session.editMode
    ) {

        dismissEditPopup(
                activity
        );

        return;
    }


    SelectionState state =
            getSelectionState(
                    activity
            );


    if (state != null) {

        restoreOriginalRecords(
                activity,
                state,
                session.originalRecords
        );


        showAllRecordRows(
                state
        );

        restoreRecordRowLayout(
                session
        );
    }


    session.editMode =
            false;


    session.deletedRows.clear();


    session.removedCells.clear();


    session.originalRecords.clear();


    session.focusedCell =
            null;


    SelectionState newState =
            getSelectionState(
                    activity
            );


    if (newState != null) {

        disableEditingOnCells(
                newState
        );


        renumberVisibleSerialNumbers(
                activity,
                newState
        );
    }


    selection(
            activity
    );


    dismissEditPopup(
            activity
    );


    clearEditActionFocus(
            activity
    );
}


    // ============================================================
    // RESTORE ORIGINAL RECORDS
    //
    // IMPORTANT FIX:
    //
    // When Close is pressed without Save, EditText cells are
    // replaced with normal TextViews.
    //
    // This removes the EditText underline/red bottom line.
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT SESSION / RESTORE

     * PURPOSE: Restores the original saved records when the existing edit workflow requires restoration.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: records area.

     * WORK FLOW: Connection: internal edit-session flow.

     * INPUT / OUTPUT: Input: Activity. Action: restores original record data using the current implementation.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void restoreOriginalRecords(
            Activity activity,
            SelectionState state,
            ArrayList<String> originalRecords
    ) {

        if (
                activity == null
                ||
                state == null
                ||
                originalRecords == null
        ) {
            return;
        }


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            if (
                    r >= originalRecords.size()
            ) {
                continue;
            }


            String original =
                    originalRecords.get(r);


            String[] parts =
                    original.split(
                            "\\s*\\|\\s*",
                            -1
                    );


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                String value = "";


                if (
                        c < parts.length
                ) {

                    value =
                            parts[c].trim();
                }


                TextView oldCell =
                        record.get(c);


                if (oldCell == null) {
                    continue;
                }


                /*
                 * If it is already a normal TextView,
                 * simply restore its text.
                 */
                if (
                        !(oldCell instanceof EditText)
                ) {

                    oldCell.setText(
                            value
                    );

                    continue;
                }


                ViewParent parentObject =
                        oldCell.getParent();


                if (!(parentObject instanceof ViewGroup)) {

                    oldCell.setText(
                            value
                    );

                    continue;
                }


                ViewGroup parent =
                        (ViewGroup)
                                parentObject;


                int position =
                        parent.indexOfChild(
                                oldCell
                        );


                if (position < 0) {

                    oldCell.setText(
                            value
                    );

                    continue;
                }


                /*
                 * Create a normal TextView with the same
                 * visual properties as the EditText.
                 */
                TextView textView =
                        new TextView(
                                activity
                        );


                textView.setText(
                        value
                );


                textView.setTextColor(
                        BLACK
                );


                textView.setTextSize(
                        android.util.TypedValue
                                .COMPLEX_UNIT_PX,
                        oldCell.getTextSize()
                );


                textView.setGravity(
                        oldCell.getGravity()
                );


                textView.setPadding(
                        oldCell.getPaddingLeft(),
                        oldCell.getPaddingTop(),
                        oldCell.getPaddingRight(),
                        oldCell.getPaddingBottom()
                );


                textView.setTypeface(
                        oldCell.getTypeface()
                );


                textView.setIncludeFontPadding(
                        oldCell.getIncludeFontPadding()
                );


                textView.setMinHeight(
                        oldCell.getMinHeight()
                );


                textView.setMinWidth(
                        oldCell.getMinWidth()

    // ============================================================
    // GET FOCUSED EDIT CELL
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT HELPERS

     * PURPOSE: Returns the currently focused edit cell from the edit session.

     * ACTIVITY / VIEW CONNECTION: Activity/View: internal.

     * WORK FLOW: Connection: clear/delete/edit confirmation helpers.

     * INPUT / OUTPUT: Input: Activity. Return: focused TextView/EditText or null.

     * DETAILS: No direct XML connection.

     * ---------------------------------------------------------------

     */

    private static TextView getFocusedEditCell(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }


        EditSession session =
                getEditSession(
                        activity
                );


        if (
                session == null
                ||
                !session.editMode
        ) {

            return null;
        }


        if (
                session.focusedCell != null
                &&
                session.focusedCell.getVisibility()
                        != View.GONE
                &&
                session.focusedCell.getParent() != null
        ) {

            return session.focusedCell;
        }


        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return null;
        }


        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            TextView cell =
                    state.cells.get(i);


            if (
                    cell instanceof EditText
                    &&
                    cell.hasFocus()
            ) {

                session.focusedCell =
                        cell;

                return cell;
            }
        }


        return null;
    }

    /*

     * ---------------------------------------------------------------

     * SECTION: CONFIRMATION UI / POSITION

     * PURPOSE: Applies the existing four-side/center confirmation window position settings.

     * ACTIVITY / VIEW CONNECTION: Activity: current edit screen. View: confirmation window.

     * WORK FLOW: Connection: internal confirmation dialog/window creation.

     * INPUT / OUTPUT: Inputs: Activity + confirmation window view. Uses CONFIRM_POSITION_MODE and CONFIRM_X_DP/CONFIRM_Y_DP.

     * DETAILS: Configuration only; no Activity button hookup.

     * ---------------------------------------------------------------

     */

    private static void setConfirmationWindowPosition(
            Activity activity,
            android.view.Window window
    ) {

        if (activity == null || window == null) {
            return;
        }

        android.view.WindowManager.LayoutParams params =
                window.getAttributes();

        switch (CONFIRM_POSITION_MODE) {

            case CONFIRM_POSITION_TOP_LEFT:
                params.gravity =
                        Gravity.TOP | Gravity.LEFT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_TOP_RIGHT:
                params.gravity =
                        Gravity.TOP | Gravity.RIGHT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_BOTTOM_LEFT:
                params.gravity =
                        Gravity.BOTTOM | Gravity.LEFT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_BOTTOM_RIGHT:
                params.gravity =
                        Gravity.BOTTOM | Gravity.RIGHT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_CENTER:
                params.gravity = Gravity.CENTER;
                params.x = 0;
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            default:
                params.gravity = Gravity.CENTER;
                params.x = 0;
                params.y = 0;
                break;
        }

        window.setAttributes(params);
    }

    // CLEAR FOCUSED CELL
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT ACTION / CLEAR CELL

     * PURPOSE: Clears the currently focused editable cell.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: focused EditText cell.

     * WORK FLOW: Connection: existing clear confirmation → clearFocusedCell() flow.

     * INPUT / OUTPUT: Input: Activity. Action: clears focused cell using current implementation.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void clearFocusedCell(
            Activity activity
    ) {

        if (activity == null) return;

        EditSession session = getEditSession(activity);

        if (session == null || !session.editMode) return;

        SelectionState state = getSelectionState(activity);

        // A selection made before entering edit mode has priority.
        if (state != null && !session.preEditSelectedCells.isEmpty()) {
            boolean changed = false;

            for (int r = 0; r < state.records.size(); r++) {
                ArrayList<TextView> record = state.records.get(r);
                if (record == null) continue;

                for (int c = 0; c < record.size(); c++) {
                    if (session.preEditSelectedCells.contains(r + ":" + c)) {
                        TextView cell = record.get(c);
                        if (cell != null) {
                            cell.setText("");
                            session.removedCells.add(cell);
                            changed = true;
                        }
                    }
                }
            }

            if (changed) {
                // Clear has completed for the selected cells. The selection must
                // now become deselected, while Edit Mode itself remains active.
                deselectState(state);
                session.preEditSelectedCells.clear();
                session.focusedCell = null;
                clearEditActionFocus(activity);
                return;
            }
        }

        TextView cell = getFocusedEditCell(activity);

        if (cell == null) {
            return;
        }

        session.focusedCell = cell;
        cell.requestFocus();
        cell.setText("");

        if (cell instanceof EditText) {
            ((EditText) cell).setSelection(0);
        }

        session.removedCells.add(cell);
        clearEditActionFocus(activity);
    }


// ============================================================
    // DELETE FOCUSED ROW
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT ACTION / DELETE ROW

     * PURPOSE: Deletes/hides the currently focused record row according to the existing edit workflow.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: focused record row/cell.

     * WORK FLOW: Connection: existing delete confirmation → deleteFocusedRow().

     * INPUT / OUTPUT: Input: Activity. Action: marks/removes row through current logic.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void deleteFocusedRow(
            final Activity activity
    ) {

        if (activity == null) return;

        final EditSession session = getEditSession(activity);

        if (session == null || !session.editMode) return;

        final SelectionState state = getSelectionState(activity);

        if (state == null) return;

        final Set<Integer> rowsToDelete = new HashSet<Integer>();

        // If cells were selected before Edit Menu was opened, delete the
        // complete records containing those selected cells.
        for (String key : session.preEditSelectedCells) {
            int separator = key.indexOf(":");
            if (separator > 0) {
                try {
                    int row = Integer.parseInt(key.substring(0, separator));
                    if (row >= 0 && row < state.records.size()) {
                        rowsToDelete.add(row);
                    }
                } catch (Exception ignored) {
                }
            }
        }

        // Otherwise delete the complete record containing the currently edited cell.
        if (rowsToDelete.isEmpty()) {
            TextView focusedCell = getFocusedEditCell(activity);
            if (focusedCell == null) {
                return;
            }

            int row = getRecordIndexForView(activity, focusedCell);
            if (row >= 0) rowsToDelete.add(row);
        }

        if (rowsToDelete.isEmpty()) return;

        final int count = rowsToDelete.size();
        final String message =
                count == 1
                ? "Delete this complete record?\n\nName, Phone, Address and Product will be removed."
                : "Delete these " + count + " complete records?\n\nName, Phone, Address and Product will be removed.";

        showDeleteConfirmation(
                activity,
                rowsToDelete,
                state,
                session,
                count
        );
    }

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT / ROW VISIBILITY

     * PURPOSE: Hides a record row during the existing delete workflow.

     * WORK FLOW: Connection: deleteFocusedRow() and related edit helpers.

     * INPUT / OUTPUT: Input: Activity + row/index data as defined by the current method. Action: hides row.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void freezeRecordRowsForEdit(
            SelectionState state,
            EditSession session
    ) {

        if (state == null || session == null || session.rowsFrozen) {
            return;
        }

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();

        for (int r = 0; r < state.records.size(); r++) {
            ArrayList<TextView> record = state.records.get(r);
            if (record == null || record.size() == 0) continue;

            TextView firstCell = record.get(0);
            if (firstCell == null) continue;

            ViewParent parentObject = firstCell.getParent();
            if (!(parentObject instanceof ViewGroup)) continue;

            View row = (View) parentObject;
            ViewGroup.LayoutParams params = row.getLayoutParams();
            if (params == null) continue;

            int measuredHeight = row.getHeight();
            if (measuredHeight <= 0) {
                measuredHeight = firstCell.getHeight();
            }

            if (measuredHeight <= 0) {
                continue;
            }

            session.originalRowHeights.put(
                    row,
                    Integer.valueOf(params.height)
            );

            if (params instanceof LinearLayout.LayoutParams) {
                LinearLayout.LayoutParams lp =
                        (LinearLayout.LayoutParams) params;
                session.originalRowWeights.put(
                        row,
                        Float.valueOf(lp.weight)
                );

                lp.height = measuredHeight;
                lp.weight = 0f;
            } else {
                params.height = measuredHeight;
            }

            row.setLayoutParams(params);
        }

        session.rowsFrozen = true;
    }


    private static void finalizeRecordRowLayoutAfterSave(
            EditSession session
    ) {

        if (session == null || !session.rowsFrozen) {
            return;
        }

        for (Map.Entry<View, Integer> entry :
                session.originalRowHeights.entrySet()) {

            View row = entry.getKey();
            if (row == null) {
                continue;
            }

            ViewGroup.LayoutParams params = row.getLayoutParams();
            if (params == null) {
                continue;
            }

            Integer originalHeight = entry.getValue();
            if (originalHeight != null && originalHeight.intValue() > 0) {
                params.height = originalHeight.intValue();
            }

            // Keep weight at zero after Save. Deleted rows are GONE, so
            // restoring the old weights would redistribute height.
            if (params instanceof LinearLayout.LayoutParams) {
                ((LinearLayout.LayoutParams) params).weight = 0f;
            }

            row.setLayoutParams(params);
        }

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();
        session.rowsFrozen = false;
    }


    private static void restoreRecordRowLayout(
            EditSession session
    ) {

        if (session == null || !session.rowsFrozen) {
            return;
        }

        for (Map.Entry<View, Integer> entry :
                session.originalRowHeights.entrySet()) {

            View row = entry.getKey();
            if (row == null) continue;

            ViewGroup.LayoutParams params = row.getLayoutParams();
            if (params == null) continue;

            Integer originalHeight = entry.getValue();
            if (originalHeight != null) {
                params.height = originalHeight.intValue();
            }

            if (params instanceof LinearLayout.LayoutParams) {
                Float originalWeight =
                        session.originalRowWeights.get(row);
                if (originalWeight != null) {
                    ((LinearLayout.LayoutParams) params).weight =
                            originalWeight.floatValue();
                }
            }

            row.setLayoutParams(params);
        }

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();
        session.rowsFrozen = false;
    }


    private static void hideRecordRow(
            ArrayList<TextView> record
    ) {

        if (
                record == null
                ||
                record.size() == 0
        ) {

            return;
        }


        TextView firstCell =
                record.get(0);


        if (firstCell == null) {
            return;
        }


        ViewParent parent =
                firstCell.getParent();


        if (parent instanceof ViewGroup) {

            ViewGroup row =
                    (ViewGroup)
                            parent;

            // Rows are frozen to their current measured height while Edit
            // Mode is active, so GONE removes the deleted row without making
            // the remaining record cells expand vertically.
            row.setVisibility(
                    View.GONE
            );
        }
    }


    // ============================================================
    // RESTORE EDITED RECORDS AS TEXTVIEWS
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT MODE / RESTORE VIEWS

     * PURPOSE: Restores edited record cells back to the normal TextView representation.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: recordsContainer.

     * WORK FLOW: Connection: existing save/close edit flow.

     * INPUT / OUTPUT: Input: Activity. Action: restores the normal non-edit UI representation.

     * DETAILS: Preserve current cell/layout behavior.

     * ---------------------------------------------------------------

     */

    private static void restoreEditedRecordsAsTextViews(
            Activity activity,
            SelectionState state
    ) {

        if (activity == null || state == null) return;

        for (int r = 0; r < state.records.size(); r++) {
            ArrayList<TextView> record = state.records.get(r);
            if (record == null) continue;

            for (int c = 0; c < record.size(); c++) {
                TextView oldCell = record.get(c);
                if (oldCell == null || !(oldCell instanceof EditText)) continue;

                ViewParent parentObject = oldCell.getParent();
                if (!(parentObject instanceof ViewGroup)) continue;

                ViewGroup parent = (ViewGroup) parentObject;
                int position = parent.indexOfChild(oldCell);
                if (position < 0) continue;

                TextView textView = new TextView(activity);
                textView.setText(oldCell.getText());
                textView.setTextColor(BLACK);
                textView.setTextSize(
                        android.util.TypedValue.COMPLEX_UNIT_PX,
                        oldCell.getTextSize()
                );
                textView.setGravity(oldCell.getGravity());
                textView.setPadding(
                        oldCell.getPaddingLeft(),
                        oldCell.getPaddingTop(),
                        oldCell.getPaddingRight(),
                        oldCell.getPaddingBottom()
                );
                textView.setTypeface(oldCell.getTypeface());
                textView.setIncludeFontPadding(oldCell.getIncludeFontPadding());
                textView.setMinHeight(oldCell.getMinHeight());
                textView.setMinWidth(oldCell.getMinWidth());

                Drawable background = oldCell.getBackground();
                if (background != null) textView.setBackground(background);

                textView.setClickable(true);
                textView.setEnabled(true);

                ViewGroup.LayoutParams params = oldCell.getLayoutParams();
                textView.setLayoutParams(params);

                parent.removeViewAt(position);
                parent.addView(textView, position);
                record.set(c, textView);
            }
        }

        LinearLayout container =
                activity.findViewById(R.id.recordsContainer);
        if (container != null) container.setTag(state.records);
    }


    // ============================================================
    // SAVE EDITED RECORDS
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT DATA / SAVE

     * PURPOSE: Persists the edited records using the existing edit-session data.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: recordsContainer/edit cells.

     * WORK FLOW: Connection: save(activity) → saveEditedRecords(activity).

     * INPUT / OUTPUT: Input: Activity. Action: writes the current edited record set using the existing implementation.

     * DETAILS: Internal controller method; no separate XML hookup.

     * ---------------------------------------------------------------

     */

    private static void saveEditedRecords(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }


        EditSession session =
                getEditSession(
                        activity
                );


        if (
                session == null
                ||
                !session.editMode
        ) {

            return;
        }


        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return;
        }


        ArrayList<String> updatedRecords =
                new ArrayList<String>();


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            if (
                    session.deletedRows.contains(
                            r
                    )
            ) {

                continue;
            }


            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            String name = "";

            String phone = "";

            String address = "";

            String product = "";


            if (
                    record.size() > 0
                    &&
                    record.get(0) != null
            ) {

                name =
                        record.get(0)
                                .getText()
                                .toString()
                                .trim();
            }


            if (
                    record.size() > 1
                    &&
                    record.get(1) != null
            ) {

                phone =
                        record.get(1)
                                .getText()
                                .toString()
                                .trim();
            }


            if (
                    record.size() > 2
                    &&
                    record.get(2) != null
            ) {

                address =
                        record.get(2)
                                .getText()
                                .toString()
                                .trim();
            }


            if (
                    record.size() > 3
                    &&
                    record.get(3) != null
            ) {

                product =
                        record.get(3)
                                .getText()
                                .toString()
                                .trim();
            }


            String finalLine =
                    name
                            + " | "
                            + phone
                            + " | "
                            + address
                            + " | "
                            + product;


            updatedRecords.add(
                    finalLine
            );
        }


        FileOutputStream output =
                null;


        try {

            output =
                    activity.openFileOutput(
                            RECORDS_FILE,
                            Context.MODE_PRIVATE
                    );


            for (
                    int i = 0;
                    i < updatedRecords.size();
                    i++
            ) {

                String line =
                        updatedRecords.get(i);


                output.write(
                        (
                                line
                                        +
                                        "\n"
                        ).getBytes("UTF-8")
                );
            }


            output.flush();

            output.close();

            output = null;


            // Replace EditTexts with normal TextViews after Save so the
            // EditText red underline cannot remain visible.
            restoreEditedRecordsAsTextViews(
                    activity,
                    state
            );

            // Keep the original compact row heights after Save.
            // Do not restore old row weights here because deleted GONE rows
            // would otherwise make remaining rows expand vertically.
            finalizeRecordRowLayoutAfterSave(
                    session
            );


            session.editMode =
                    false;

            session.deletedRows.clear();

            session.removedCells.clear();

            session.originalRecords.clear();

            session.preEditSelectedCells.clear();

            session.focusedCell =
                    null;


            disableEditingOnCells(
                    state
            );


            // Save completes the edit operation, so the previous selection
            // must now be fully deselected.
            deselectAll(
                    activity
            );


            selection(
                    activity
            );


            renumberVisibleSerialNumbers(
                    activity,
                    state
            );


            dismissEditPopup(
                    activity
            );


            clearEditActionFocus(
                    activity
            );


            Toast.makeText(
                    activity,
                    "Data Update",
                    Toast.LENGTH_SHORT
            ).show();


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to update records",
                    Toast.LENGTH_SHORT
            ).show();


        } finally {

            if (output != null) {

                try {
                    output.close();
                } catch (Exception ignored) {
                }
            }
        }
    }


    // ============================================================
    // RENUMBER VISIBLE SERIAL NUMBERS
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT / RECORD DISPLAY

     * PURPOSE: Renumbers visible record serial-number views after row changes.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: visible serial-number TextViews.

     * WORK FLOW: Connection: existing delete/restore/save flow.

     * INPUT / OUTPUT: Input: Activity. Action: updates visible numbering.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static void renumberVisibleSerialNumbers(
            Activity activity,
            SelectionState state
    ) {

        if (
                activity == null
                ||
                state == null
        ) {
            return;
        }


        int serialNumber =
                1;


        Set<TextView> recordCells =
                new HashSet<TextView>();


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);


                if (cell != null) {

                    recordCells.add(
                            cell
                    );
                }
            }
        }


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (
                    record == null
                    ||
                    record.size() == 0
            ) {
                continue;
            }


            TextView firstCell =
                    record.get(0);


            if (firstCell == null) {
                continue;
            }


            ViewParent parent =
                    firstCell.getParent();


            if (!(parent instanceof ViewGroup)) {
                continue;
            }


            ViewGroup row =
                    (ViewGroup)
                            parent;


            if (
                    row.getVisibility()
                            !=
                            View.VISIBLE
            ) {

                continue;
            }


            TextView serialView =
                    findSerialTextView(
                            row,
                            recordCells
                    );


            if (serialView != null) {

                serialView.setText(
                        String.valueOf(
                                serialNumber
                        )
                );


                serialNumber++;
            }
        }
    }


    // ============================================================
    // FIND SERIAL TEXTVIEW
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: EDIT / RECORD DISPLAY HELPER

     * PURPOSE: Finds the serial-number TextView associated with a record row.

     * ACTIVITY / VIEW CONNECTION: Activity: MyrecordsActivity. View: serial-number TextView.

     * WORK FLOW: Connection: renumberVisibleSerialNumbers() and row operations.

     * INPUT / OUTPUT: Inputs/return: current method contract is preserved.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static TextView findSerialTextView(
            ViewGroup row,
            Set<TextView> recordCells
    ) {

        if (
                row == null
                ||
                recordCells == null
        ) {
            return null;
        }


        for (
                int i = 0;
                i < row.getChildCount();
                i++
        ) {

            View child =
                    row.getChildAt(i);


            if (
                    child instanceof TextView
                    &&
                    !(child instanceof EditText)
            ) {

                TextView textView =
                        (TextView)
                                child;


                if (
                        !recordCells.contains(
                                textView
                        )
                ) {

                    return textView;
                }
            }
        }


        for (
                int i = 0;
                i < row.getChildCount();
                i++
        ) {

            View child =
                    row.getChildAt(i);


            if (child instanceof ViewGroup) {

                TextView found =
                        findSerialTextView(
                                (ViewGroup) child,
                                recordCells
                        );


                if (found != null) {
                    return found;
                }
            }
        }


        return null;
    }

            return;

                getSelectionState(

                    activity,

            ).show();

            return;

                LayoutInflater
                        .from(activity)
                        .inflate(
                                R.layout.popup_menu,

                        );


        if (popupView == null) {
            return;
        }


        final PopupWindow popupWindow =
                new PopupWindow(
                        popupView,
                        ViewGroup.LayoutParams.WRAP_CONTENT,
                        ViewGroup.LayoutParams.WRAP_CONTENT,
                        true
                );


        popupWindow.setOutsideTouchable(
                true
        );


        popupWindow.setFocusable(
                true
        );


        popupWindow.setBackgroundDrawable(
                new ColorDrawable(
                        Color.TRANSPARENT
                )
        );

                popupView.findViewById(
                        R.id.copyButton
                );

                popupView.findViewById(
                        R.id.shareButton
                );

                popupView.findViewById(
                        R.id.exportButton
                );

                popupView.findViewById(

                popupView.findViewById(
                        R.id.deselectAllButton

                POPUP_GRAVITY;

                POPUP_POSITION_MODE
                        ==
                        POSITION_TOP
        ) {

                    dp(
                            activity,
                            POPUP_X_ADJUST_DP
                    );

                    dp(
                            activity,
                            POPUP_TOP_DP
                    )
                    +

                    );

                        ==
                        POSITION_BOTTOM
        ) {

                    Gravity.BOTTOM
                            |

                    dp(
                            activity,
                            POPUP_X_ADJUST_DP
                    );

            yOffset =
                    -dp(
                            activity,
                            POPUP_BOTTOM_DP
                    )
                    +
                    dp(
                            activity,
                            POPUP_Y_ADJUST_DP
                    );
        }


        else if (
                POPUP_POSITION_MODE
                        ==
                        POSITION_LEFT
        ) {

            gravity =
                    Gravity.CENTER_VERTICAL
                            |
                            Gravity.LEFT;

            xOffset =
                    dp(
                            activity,
                            POPUP_LEFT_DP
                    )
                    +
                    dp(
                            activity,
                            POPUP_X_ADJUST_DP
                    );

            yOffset =
                    dp(
                            activity,
                            POPUP_Y_ADJUST_DP
                    );
        }

                POPUP_POSITION_MODE
                        ==
                        POSITION_RIGHT

                    dp(
                            activity,
                            POPUP_RIGHT_DP
                    )
                    +
                    dp(
                            activity,
                            POPUP_X_ADJUST_DP
                    );

            yOffset =
                    dp(
                            activity,
                            POPUP_TOP_DP
                    )
                    +
                    dp(
                            activity,
                            POPUP_Y_ADJUST_DP
                    );
        }


        else if (
                POPUP_POSITION_MODE
                        ==
                        POSITION_CENTER
        ) {

            gravity =
                    Gravity.CENTER;

            xOffset =
                    dp(
                            activity,
                            POPUP_X_ADJUST_DP
                    );

            yOffset =
                    dp(
                            activity,
                            POPUP_Y_ADJUST_DP
                    );
        }


        if (copyButton != null) {

            copyButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            copy(
                                    activity
                            );

                            popupWindow.dismiss();
                        }
                    }
            );
        }


        if (shareButton != null) {

            shareButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            share(
                                    activity
                            );

                            popupWindow.dismiss();
                        }
                    }
            );
        }


        if (exportButton != null) {

            exportButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            export(
                                    activity
                            );

                            popupWindow.dismiss();
                        }
                    }
            );
        }

            selectAllButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override

                                View v
                        ) {

                            selectAll(
                                    activity
                            );

                        }
                    }

    // ============================================================
    // DP CONVERSION
    // ============================================================

    /*

     * ---------------------------------------------------------------

     * SECTION: UI UTILITY

     * PURPOSE: Converts density-independent pixels to the device's pixel value.

     * ACTIVITY / VIEW CONNECTION: View/UI: internal sizing and positioning.

     * WORK FLOW: Connection: popup, confirmation and button-size helpers.

     * INPUT / OUTPUT: Input: Activity + dp value. Return: pixel integer.

     * DETAILS: Internal helper.

     * ---------------------------------------------------------------

     */

    private static int dp(
            Activity activity,
            int value
    ) {

        if (activity == null) {
            return value;
        }


        float density =
                activity
                        .getResources()
                        .getDisplayMetrics()
                        .density;


        return (int)(
                value
                        * density
                        + 0.5f
        );
    }
