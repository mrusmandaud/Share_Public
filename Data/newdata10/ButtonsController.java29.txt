    private static EditSession getEditSession(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }

        EditSession session =
                EDIT_SESSIONS.get(activity);

        if (session == null) {

            session =
                    new EditSession();

            EDIT_SESSIONS.put(
                    activity,
                    session
            );
        }

        return session;
    }

    //
    // GUIDANCE / INFORMATION
    // For save problems, inspect validation, spinner placeholder handling, and file write logic here.
    //
    // FEATURES
    // New-record validation, persistence, and form reset.
    //
    // IMPORTANT

    // ============================================================

    public static boolean saveRecord(
            Activity activity,
            EditText nameView,
            EditText phoneView,
            EditText addressView,
            Spinner productSpinner,
            CheckBox confirm
    ) {

        if (activity == null) {
            return false;
        }

        if (
                confirm == null
                ||
                !confirm.isChecked()
        ) {

            Toast.makeText(
                    activity,
                    "Please confirm and agree to save",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        String name = "";

        if (
                nameView != null
                &&
                nameView.getText() != null
        ) {

            name =
                    nameView
                            .getText()
                            .toString()
                            .trim();
        }


        String phone = "";

        if (
                phoneView != null
                &&
                phoneView.getText() != null
        ) {

            phone =
                    phoneView
                            .getText()
                            .toString()
                            .trim();
        }


        String address = "";

        if (
                addressView != null
                &&
                addressView.getText() != null
        ) {

            address =
                    addressView
                            .getText()
                            .toString()
                            .trim();
        }


        String product = "";

        if (
                productSpinner != null
                &&
                productSpinner.getSelectedItem() != null
        ) {

            product =
                    productSpinner
                            .getSelectedItem()
                            .toString()
                            .trim();
        }


        if (name.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Name",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (phone.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Phone",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (address.length() == 0) {

            Toast.makeText(
                    activity,
                    "Please enter Address",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        if (
                product.length() == 0
                ||
                product.equalsIgnoreCase(
                        "Select Product"
                )
                ||
                product.equalsIgnoreCase(
                        "Select Products"
                )
        ) {

            Toast.makeText(
                    activity,
                    "Please select Product",
                    Toast.LENGTH_SHORT
            ).show();

            return false;
        }


        String record =
                name
                        + " | "
                        + phone
                        + " | "
                        + address
                        + " | "
                        + product
                        + "\n";


        FileOutputStream output = null;


        try {

            output =
                    activity.openFileOutput(
                            RECORDS_FILE,
                            Context.MODE_APPEND
                    );

            output.write(
                    record.getBytes("UTF-8")
            );

            output.flush();


            Toast.makeText(
                    activity,
                    "Record saved successfully",
                    Toast.LENGTH_SHORT
            ).show();


            clearRecord(
                    nameView,
                    phoneView,
                    addressView,
                    productSpinner,
                    confirm
            );


            return true;


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to save record",
                    Toast.LENGTH_SHORT
            ).show();

            return false;


        } finally {

            if (output != null) {

                try {
                    output.close();
                } catch (Exception ignored) {
                }
            }
        }
    }

    public static ArrayList<String> getRecords(
            Activity activity
    ) {

        ArrayList<String> records =
                new ArrayList<String>();


        if (activity == null) {
            return records;
        }


        BufferedReader reader = null;


        try {

            FileInputStream input =
                    activity.openFileInput(
                            RECORDS_FILE
                    );


            reader =
                    new BufferedReader(
                            new InputStreamReader(
                                    input,
                                    "UTF-8"
                            )
                    );


            String line;


            while (
                    (line = reader.readLine()) != null
            ) {

                line =
                        line.trim();


                if (line.length() > 0) {

                    records.add(line);
                }
            }


        } catch (Exception e) {

        } finally {

            if (reader != null) {

                try {
                    reader.close();
                } catch (Exception ignored) {
                }
            }
        }


        return records;
    }

    private static class SelectionState {

        final ArrayList<ArrayList<TextView>>
                records;

        final ArrayList<TextView>
                cells;

        final ArrayList<Integer>
                cellRecordIndexes;


        int firstIndex = -1;

        int lastIndex = -1;

        int firstRecord = -1;

        int lastRecord = -1;


        SelectionState(
                ArrayList<ArrayList<TextView>>
                        records
        ) {

            this.records =
                    records;

            this.cells =
                    new ArrayList<TextView>();

            this.cellRecordIndexes =
                    new ArrayList<Integer>();


            if (records == null) {
                return;
            }


            for (
                    int r = 0;
                    r < records.size();
                    r++
            ) {

                ArrayList<TextView> record =
                        records.get(r);


                if (record == null) {
                    continue;
                }


                for (
                        int c = 0;
                        c < record.size();
                        c++
                ) {

                    TextView cell =
                            record.get(c);


                    if (cell != null) {

                        cells.add(cell);

                        cellRecordIndexes.add(r);
                    }
                }
            }
        }
    }

            int startRecord,
            int endRecord
    ) {

        if (state == null) {
            return;
        }


        if (startRecord > endRecord) {

            int temp =
                    startRecord;

            startRecord =
                    endRecord;

            endRecord =
                    temp;
        }


        for (
                int r = startRecord;
                r <= endRecord;
                r++
        ) {

            if (
                    r < 0
                    ||
                    r >= state.records.size()
            ) {

                continue;
            }


            selectRecord(
                    state.records.get(r)
            );
        }
    }

    public static void selection(
        final Activity activity
) {

    if (activity == null) {
        return;
    }


    final LinearLayout
            selectionContainer =
            activity.findViewById(
                    R.id.recordsContainer
            );


    if (selectionContainer == null) {
        return;
    }


    Object selectionObject =
            selectionContainer.getTag();


    if (!(selectionObject instanceof ArrayList)) {
        return;
    }


    final ArrayList<ArrayList<TextView>>
            selectionRecords =
            (ArrayList<ArrayList<TextView>>)
                    selectionObject;


    if (selectionRecords.size() == 0) {
        return;
    }


    final SelectionState state =
            new SelectionState(
                    selectionRecords
            );


    if (state.cells.size() == 0) {
        return;
    }


    SELECTION_STATES.put(
            activity,
            state
    );


    clearSelection(
            state
    );


    final EditSession editSession =
            getEditSession(
                    activity
            );


    for (
            int i = 0;
            i < state.cells.size();
            i++
    ) {

        final TextView cell =
                state.cells.get(i);


        if (cell == null) {
            continue;
        }


        final int index =
                i;


        cell.setClickable(
                true
        );


            // ====================================================
            // EDIT MODE
            // ====================================================

            if (
                    editSession != null
                    &&
                    editSession.editMode
            ) {

                cell.setOnClickListener(
                        new View.OnClickListener() {

                            @Override
                            public void onClick(
                                    View v
                            ) {

                                if (v instanceof EditText) {

                                    EditText editText =
                                            (EditText) v;

                                    editSession.focusedCell =
                                            editText;

                                    editText.requestFocus();

                                    editText.setSelection(
                                            editText.length()
                                    );
                                }
                            }
                        }
                );


                cell.setOnLongClickListener(
                        new View.OnLongClickListener() {

                            @Override
                            public boolean onLongClick(
                                    View v
                            ) {

                                if (v instanceof EditText) {

                                    EditText editText =
                                            (EditText) v;

                                    editSession.focusedCell =

                            }


                            if (
                                    currentState.firstIndex
                                            ==
                                            currentState.lastIndex
                                    &&
                                    index
                                            ==
                                            currentState.firstIndex
                            ) {

                                deselectState(
                                        currentState
                                );

                                return;
                            }


                            int firstRecord =

                            if (firstRecord < 0) {

                                firstRecord =
                                        getRecordIndexForCell(
                                                currentState,
                                                currentState.firstIndex
                                        );
                            }


                            if (
                                    firstRecord
                                            ==
                                            tappedRecord
                            ) {

                                ArrayList<TextView>
                                        record =
                                        currentState.records
                                                .get(
                                                        tappedRecord
                                                );


                                int firstPosition =
                                        getCellPositionInRecord(
                                                currentState,
                                                tappedRecord,
                                                currentState.cells
                                                        .get(
                                                                currentState.firstIndex
                                                        )
                                        );

                                        getCellPositionInRecord(
                                                currentState,
                                                tappedRecord,
                                                cell
                                        );


                                if (
                                        firstPosition < 0
                                        ||
                                        secondPosition < 0
                                ) {

                                    return;
                                }


                                int start =
                                        firstPosition;

                                int end =
                                        secondPosition;


                                if (start > end) {

                                    int temp =
                                            start;

                                    start =
                                            end;

                                    end =
                                            temp;
                                }


                                clearSelection(
                                        currentState
                                );


                                for (
                                        int c = start;
                                        c <= end;
                                        c++
                                ) {

                                    if (
                                            c >= 0
                                            &&
                                            c < record.size()
                                    ) {

                                        setCellSelected(
                                                record.get(c),
                                                true
                                        );
                                    }
                                }


                                currentState.firstRecord =
                                        tappedRecord;

                                currentState.lastRecord =
                                        tappedRecord;

                                currentState.firstIndex =
                                        index;

                                currentState.lastIndex =
                                        index;


                                return;
                            }


                            int startRecord =
                                    firstRecord;

                            int endRecord =
                                    tappedRecord;


                            if (
                                    startRecord
                                            >
                                            endRecord
                            ) {

                                int temp =
                                        startRecord;

                                startRecord =
                                        endRecord;

                                endRecord =
                                        temp;
                            }


                            clearSelection(
                                    currentState
                            );


                            selectRecordRange(
                                    currentState,
                                    startRecord,
                                    endRecord
                            );


                            currentState.firstRecord =
                                    startRecord;

                            currentState.lastRecord =
                                    endRecord;

                            currentState.firstIndex =
                                    index;

                            currentState.lastIndex =
                                    index;
                        }
                    }

                    SELECTED_COLOR;

    // ============================================================
    // SECTION 31 — GETSELECTEDTEXT

    //
    // CONNECTION
    // Internal controller helper; called by the existing workflow.
    //
    // PURPOSE
    // Handles the existing getSelectedText operation without changing its executable behavior.
    //
    // DETAILS

    //
    // GUIDANCE / INFORMATION

    // ============================================================

    private static String getSelectedText(
            Activity activity
    ) {

        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return "";
        }


        StringBuilder selectedText =
                new StringBuilder();


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            boolean recordSelected =
                    false;


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);


                if (
                        cell != null
                        &&
                        isSelected(cell)
                ) {

                    recordSelected =
                            true;

                    break;
                }
            }


            if (!recordSelected) {
                continue;
            }


            StringBuilder recordText =
                    new StringBuilder();


            for (
                    int c = 0;
                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);


                if (cell == null) {
                    continue;
                }


                if (
                        recordText.length() > 0
                ) {

                    recordText.append(
                            " | "
                    );
                }


                recordText.append(
                        cell.getText()
                                .toString()
                );
            }


            if (
                    recordText.length() == 0
            ) {

                continue;
            }


            if (
                    selectedText.length() > 0
            ) {

                selectedText.append(
                        "\n"
                );
            }


            selectedText.append(
                    recordText.toString()
            );
        }


        return selectedText.toString();
    }

    public static void edit(
            final Activity activity
    ) {

        if (activity == null) {
            return;
        }


        LinearLayout container =
                activity.findViewById(
                        R.id.recordsContainer
                );


        if (container == null) {
            return;
        }


        Object object =
                container.getTag();


        if (!(object instanceof ArrayList)) {

            Toast.makeText(
                    activity,
                    "No records available",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        ArrayList<ArrayList<TextView>>
                taggedRecords =
                (ArrayList<ArrayList<TextView>>)
                        object;


        if (taggedRecords.size() == 0) {

            Toast.makeText(
                    activity,
                    "No records available",
                    Toast.LENGTH_SHORT
            ).show();

            return;
        }


        EditSession session =
                getEditSession(
                        activity
                );


        if (session == null) {
            return;
        }


        if (session.editMode) {

            // Second tap on Edit Menu exits edit mode and closes the popup.
            closeEditSession(
                    activity
            );

            return;
        }


        // Remember the normal-mode cell selection before replacing
        // TextViews with EditTexts.
        session.preEditSelectedCells.clear();

        SelectionState oldState =
                getSelectionState(
                        activity
                );

        if (oldState != null) {
            for (int r = 0; r < oldState.records.size(); r++) {
                ArrayList<TextView> oldRecord = oldState.records.get(r);
                if (oldRecord == null) continue;
                for (int c = 0; c < oldRecord.size(); c++) {
                    TextView oldCell = oldRecord.get(c);
                    if (oldCell != null && isSelected(oldCell)) {
                        session.preEditSelectedCells.add(r + ":" + c);
                    }
                }
            }
        }

        session.editMode =
                true;

        session.deletedRows.clear();

        session.clearedCells.clear();

        session.originalRecords.clear();

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();
        session.rowsFrozen = false;

        session.focusedCell =
                null;


        ArrayList<String> records =
                getRecords(
                        activity
                );


        session.originalRecords.addAll(
                records
        );


        makeCellsEditable(
                activity
        );


        SelectionState currentState =
                getSelectionState(
                        activity
                );


        freezeRecordRowsForEdit(
                currentState,
                session
        );


        clearSelection(
                currentState
        );


        restorePreEditSelection(
                activity
        );


        showEditPopup(
                activity
        );
    }

    private static void restorePreEditSelection(
            Activity activity
    ) {

        if (activity == null) return;

        EditSession session = getEditSession(activity);
        SelectionState state = getSelectionState(activity);

        if (
                session == null
                ||
                state == null
                ||
                session.preEditSelectedCells.isEmpty()
        ) return;

        for (String key : session.preEditSelectedCells) {
            int separator = key.indexOf(":");
            if (separator <= 0) continue;

            try {
                int row = Integer.parseInt(key.substring(0, separator));
                int column = Integer.parseInt(key.substring(separator + 1));

                if (row >= 0 && row < state.records.size()) {
                    ArrayList<TextView> record = state.records.get(row);
                    if (
                            record != null
                            &&
                            column >= 0
                            &&
                            column < record.size()
                    ) {
                        setCellSelected(record.get(column), true);
                    }
                }
            } catch (Exception ignored) {
            }
        }
    }

// ============================================================

// Restores the original saved values when Edit Mode is closed

// ============================================================
// SECTION 36 — RESTOREORIGINALRECORDS
// ============================================================
//
// CONNECTION
// Edit Mode Close/Restore
//
// PURPOSE
// Restores the original record text/state using the existing implementation.
//
// DETAILS
// Used by the current edit-session close/restore path.
//
// GUIDANCE / INFORMATION
// Check here for Close/restore problems.
//
// FEATURES
// Edit rollback/restore.
//
// IMPORTANT
// Existing executable code below is preserved exactly.
// This section contains documentation only.
// ============================================================

private static void restoreOriginalRecords(
        Activity activity,
        SelectionState state,
        ArrayList<String> originalRecords
) {

    if (
            activity == null
            ||
            state == null
            ||
            originalRecords == null
    ) {
        return;
    }


    for (
            int r = 0;
            r < state.records.size();
            r++
    ) {

        ArrayList<TextView> record =
                state.records.get(r);


        if (record == null) {
            continue;
        }


        if (
                r >= originalRecords.size()
        ) {
            continue;
        }


        String original =
                originalRecords.get(r);


        String[] parts =
                original.split(
                        "\\s*\\|\\s*",
                        -1
                );


        /*
         * Supported saved formats:
         *
         * Normal:
         * Name | Phone | Address | Product
         *
         * Backup/Restore:
         * # | Name | Phone | Address | Product
         *
         * The # column is never part of the editable cells.
         */
        int dataOffset = 0;


        if (parts.length >= 5) {
            dataOffset = 1;
        }


        for (
                int c = 0;
                c < record.size();
                c++
        ) {

            int sourceIndex =
                    c + dataOffset;


            String value = "";


            if (
                    sourceIndex >= 0
                    &&
                    sourceIndex < parts.length
            ) {

                value =
                        parts[sourceIndex].trim();
            }


            TextView cell =
                    record.get(c);


            if (cell == null) {
                continue;
            }


            /*
             * Restore the value directly into the existing cell.
             *
             * No ViewGroup removal.
             * No View replacement.
             */
            cell.setText(
                    value
            );


            /*
             * If this cell is an EditText, keep it usable but
             * restore the normal visual appearance.
             */
            if (
                    cell instanceof EditText
            ) {

                EditText editText =
                        (EditText) cell;


                editText.clearFocus();


                editText.setCursorVisible(
                        false
                );


                editText.setFocusable(
                        false
                );


                editText.setFocusableInTouchMode(
                        false
                );


                editText.setEnabled(
                        true
                );


                editText.setBackground(
                        null
                );
            }
        }
    }


    LinearLayout container =
            activity.findViewById(
                    R.id.recordsContainer
            );


    if (container != null) {

        container.setTag(
                state.records
        );
    }
}

// ============================================================
//
// CONNECTION
// Edit Mode cell conversion
//
// PURPOSE
// Converts eligible record cells to the existing editable representation.
//
// DETAILS
// Preserves the current row/cell structure and IDs.
//
// GUIDANCE / INFORMATION
// Check this section if cells do not become editable.
//
// FEATURES
// Editable record cells.
//
// IMPORTANT
// Existing executable code below is preserved exactly.
// This section contains documentation only.
// ============================================================

private static void makeCellsEditable(
        final Activity activity
) {

    if (activity == null) {
        return;
    }


    LinearLayout container =
            activity.findViewById(
                    R.id.recordsContainer
            );


    if (container == null) {
        return;
    }


    Object object =
            container.getTag();


    if (!(object instanceof ArrayList)) {
        return;
    }


    ArrayList<ArrayList<TextView>>
            records =
            (ArrayList<ArrayList<TextView>>)
                    object;


    final EditSession session =
            getEditSession(
                    activity
            );


    for (
            int r = 0;
            r < records.size();
            r++
    ) {

        ArrayList<TextView> record =
                records.get(r);


        if (record == null) {
            continue;
        }


        for (
                int c = 0;
                c < record.size();
                c++
        ) {

            TextView cell =
                    record.get(c);


            if (cell == null) {
                continue;
            }


            /*
             * Only an existing EditText receives EditText
             * configuration.
             *
             * Normal TextViews remain untouched.
             */
            if (
                    cell instanceof EditText
            ) {

                EditText editText =
                        (EditText) cell;


                configureEditTextInput(
                        editText,
                        c
                );


                configureEditableCell(
                        editText,
                        session
                );
            }
        }
    }


    container.setTag(
            records
    );


    selection(
            activity
    );
}

private static TextView createNormalTextView(
        Activity activity,
        TextView source,
        String value
) {

    if (
            activity == null
            ||
            source == null
    ) {
        return null;
    }


    TextView textView =
            new TextView(
                    activity
            );


    textView.setText(
            value
    );


    textView.setTextColor(
            BLACK
    );


    textView.setTextSize(
            android.util.TypedValue
                    .COMPLEX_UNIT_PX,
            source.getTextSize()
    );


    textView.setGravity(
            source.getGravity()
    );


    textView.setPadding(
            source.getPaddingLeft(),
            source.getPaddingTop(),
            source.getPaddingRight(),
            source.getPaddingBottom()
    );


    textView.setTypeface(
            source.getTypeface()
    );


    textView.setIncludeFontPadding(
            source.getIncludeFontPadding()
    );


    textView.setMinHeight(
            source.getMinHeight()
    );


    textView.setMinWidth(
            source.getMinWidth()
    );


    android.graphics.drawable.Drawable
            background =
                    source.getBackground();

        textView.setBackground(
                background
        );
    }


    textView.setClickable(
            true
    );


    textView.setEnabled(
            true
    );


    ViewGroup.LayoutParams params =
            source.getLayoutParams();


    if (params != null) {

        textView.setLayoutParams(
                params
        );
    }


    return textView;

    private static void configureEditTextInput(
            EditText editText,
            int column
    ) {

        if (editText == null) {
            return;
        }


        // Keep record cells wrapped while editing instead of
        // forcing them into one long horizontal line.
        editText.setHorizontallyScrolling(
                false
        );

        editText.setMaxLines(
                2
        );


        if (column == 0) {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_WORDS
            );

            editText.setSingleLine(
                    false
            );

        }


        else if (column == 1) {

            editText.setInputType(
                    InputType.TYPE_CLASS_PHONE
            );

            editText.setSingleLine(
                    false
            );

        }


        else if (column == 2) {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_SENTENCES
                            |
                            InputType.TYPE_TEXT_FLAG_MULTI_LINE
            );

            editText.setSingleLine(
                    false
            );

        }


        else {

            editText.setInputType(
                    InputType.TYPE_CLASS_TEXT
                            |
                            InputType.TYPE_TEXT_FLAG_CAP_SENTENCES
            );

            editText.setSingleLine(
                    false
            );
        }
    }

// ============================================================
// SECTION 43 — SHOWEDITPOPUP
// ============================================================
//
// CONNECTION
// MyrecordsActivity Edit Menu
//
// PURPOSE
// Displays the existing Edit popup using the current layout/connection.
//
// DETAILS
// The popup uses edit_popup_menu.xml and existing view IDs.
//
// GUIDANCE / INFORMATION
// For “Edit Menu click but popup not visible,” check editMenu(), prepareEditPopupPlaceholder(), then this section.
//
// FEATURES
// Edit popup display.
//
// IMPORTANT
// Existing executable code below is preserved exactly.
// This section contains documentation only.
// ============================================================

private static void showEditPopup(
        final Activity activity
) {

    if (activity == null) {
        return;
    }

    final EditSession session =
            getEditSession(activity);

    if (session == null) {
        return;
    }

    // Reuse the existing XML popup if it is already attached. Keeping the
    // popup in the real page layout (but INVISIBLE when closed) prevents the
    // record area from moving when Edit Menu opens/closes.
    View reusablePopup = null;

    if (session.editPopupWindow != null) {
        ViewParent existingParent = session.editPopupWindow.getParent();
        if (existingParent instanceof ViewGroup) {
            reusablePopup = session.editPopupWindow;
        } else {
            session.editPopupWindow = null;
        }
    }

    // Main Edit Menu button.
    final View anchor =
            activity.findViewById(
                    R.id.editMenuButton
            );

    if (anchor == null) {
        return;
    }

    final ViewParent parentObject =
            anchor.getParent();

    if (!(parentObject instanceof ViewGroup)) {
        return;
    }

    final ViewGroup originalParent =
            (ViewGroup) parentObject;

    // Inflate the real on-page submenu only when no XML placeholder exists.
    // All popup size, padding, spacing, margins and visual positioning remain
    // controlled by edit_popup_menu.xml.
    final View popupView;
    final View popupContainer;

    if (reusablePopup != null) {
        popupView = reusablePopup;
        popupContainer = reusablePopup;
    } else {
        popupView =
                LayoutInflater.from(activity).inflate(
                        R.layout.edit_popup_menu,
                        originalParent,
                        false
                );

        if (popupView == null) {
            return;
        }

        popupContainer =
                popupView.findViewById(
                        R.id.editPopupContainer
                );
    }

    if (popupContainer == null) {
        return;
    }

    // ------------------------------------------------------------
    // BUTTON ACTIONS
    // ------------------------------------------------------------

    View clearButton =
            popupView.findViewById(
                    R.id.clearButton
            );

    if (clearButton != null) {
        clearButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onClear();
                        }
                    }
                }
        );
    }

    View deleteButton =
            popupView.findViewById(
                    R.id.deleteButton
            );

    if (deleteButton != null) {
        deleteButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onDelete();
                        }
                    }
                }
        );
    }

    View saveButton =
            popupView.findViewById(
                    R.id.saveButton
            );

    if (saveButton != null) {
        saveButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onSave();
                        }
                    }
                }
        );
    }

    View closeButton =
            popupView.findViewById(
                    R.id.closeButton
            );

    if (closeButton != null) {
        closeButton.setOnClickListener(
                new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        EditMenuActions actions = getEditMenuActions(activity);
                        if (actions != null) {
                            actions.onClose();
                        } else {
                            closeEditSession(activity);
                        }
                    }
                }
        );
    }

    // ------------------------------------------------------------
    // IMPORTANT ARCHITECTURE
    // ------------------------------------------------------------
    //
    // Do NOT use PopupWindow.
    // Do NOT use showAtLocation().
    // Do NOT use showAsDropDown().
    // Do NOT use screen X/Y coordinates.
    //
    // Do NOT set popup width/height here.
    // Do NOT set popup left/right/top/bottom margins here.
    // Do NOT set popup padding or spacing here.
    // Do NOT set popup translation here.
    // Do NOT set popup elevation here.
    //
    // The Edit Menu XML and edit_popup_menu.xml control their own
    // size, margins, padding, spacing and vertical/horizontal layout.
    // The existing page parent remains the real container.
    //
    // The popup is inserted immediately after Edit Menu in the same
    // parent. The current parent is the horizontal Edit Menu row, so
    // the XML controls the exact appearance and dimensions.
    // ------------------------------------------------------------

    int originalIndex =
            originalParent.indexOfChild(anchor);

    if (originalIndex < 0) {
        return;
    }

    // Add only a newly inflated popup. A prepared XML placeholder is already
    // in the parent and is simply made visible.
    if (reusablePopup == null) {
        originalParent.addView(
                popupContainer,
                originalIndex + 1
        );
    }

    popupContainer.setVisibility(View.VISIBLE);
    originalParent.requestLayout();
    originalParent.invalidate();

    // Store the actual on-page submenu root.
    session.editPopupWindow = popupContainer;
}

private static void dismissEditPopup(
        Activity activity
) {

    if (activity == null) {
        return;
    }


    EditSession session =
            getEditSession(
                    activity
            );


    if (session == null) {
        return;
    }


    if (session.editPopupWindow != null) {

        try {
            ViewParent parentObject =
                    session.editPopupWindow.getParent();

            if (parentObject instanceof ViewGroup) {
                // Keep the XML popup in the layout as an invisible placeholder.
                // This preserves the exact page layout when Edit Menu disappears.
                session.editPopupWindow.setVisibility(View.INVISIBLE);
                ((ViewGroup) parentObject).requestLayout();
                ((ViewGroup) parentObject).invalidate();
            } else {
                session.editPopupWindow = null;
            }
        } catch (Exception ignored) {
        }
    }
}

// ============================================================
//
// CONNECTION
// MyrecordsActivity Edit Menu
//
// PURPOSE
// Prepares the existing edit_popup_menu.xml placeholder/container.
//
// DETAILS
// Uses the current editMenuButton anchor and existing parent/layout connection.
//
// GUIDANCE / INFORMATION
// If the popup cannot be prepared, verify editMenuButton, edit_popup_menu.xml, and editPopupContainer.
//

//

// Existing executable code below is preserved exactly.
// This section contains documentation only.
// ============================================================

private static void prepareEditPopupPlaceholder(
        final Activity activity
) {

    if (activity == null) {
        return;
    }

    final EditSession session = getEditSession(activity);
    if (session == null || session.editPopupWindow != null) {
        return;
    }

    final View anchor = activity.findViewById(R.id.editMenuButton);
    if (anchor == null) {
        return;
    }

    final ViewParent parentObject = anchor.getParent();
    if (!(parentObject instanceof ViewGroup)) {
        return;
    }

    final ViewGroup originalParent = (ViewGroup) parentObject;

    try {
        final View popupView =
                LayoutInflater.from(activity).inflate(
                        R.layout.edit_popup_menu,
                        originalParent,
                        false
                );

        if (popupView == null) {
            return;
        }

        final View popupContainer =
                popupView.findViewById(R.id.editPopupContainer);

        if (popupContainer == null) {
            return;
        }

        int originalIndex = originalParent.indexOfChild(anchor);
        if (originalIndex < 0) {
            return;
        }

        popupContainer.setVisibility(View.INVISIBLE);
        originalParent.addView(
                popupContainer,
                originalIndex + 1
        );

        session.editPopupWindow = popupContainer;
        originalParent.requestLayout();
        originalParent.invalidate();
    } catch (Exception ignored) {
    }
}

public static void editMenu(
        final Activity activity
) {

    if (activity == null) {
        return;
    }


    final View editMenu =
            activity.findViewById(
                    R.id.editMenuButton
            );


    if (editMenu == null) {
        return;
    }


    final EditSession session =
            getEditSession(
                    activity
            );


    if (session == null) {
        return;
    }


    // ========================================================
    // PREVENT DUPLICATE LISTENER
    // ========================================================

    if (session.menuConnected) {
        return;
    }


    session.menuConnected =
            true;


    // Create the XML popup once as an INVISIBLE placeholder. This reserves
    // exactly the same XML layout space before the first Edit click, so the
    // Saved Records/data area never jumps up or down when the popup appears.
    prepareEditPopupPlaceholder(activity);


    // ========================================================
    // MAIN EDIT MENU CLICK
    // ========================================================

    editMenu.setOnClickListener(
            new View.OnClickListener() {

                @Override
                public void onClick(
                        View v
                ) {

                    edit(
                            activity
                    );
                }
            }
    );
}

        return;

// DETAILS
// Dismisses/hides the popup and clears the current edit-session state according to the source.
//

//

private static void closeEditSession(
        Activity activity
) {

    if (activity == null) {
        return;
    }


    EditSession session =
            getEditSession(
                    activity
            );


    if (
            session == null
            ||
            !session.editMode
    ) {

        dismissEditPopup(
                activity
        );

        return;
    }


    SelectionState state =
            getSelectionState(
                    activity
            );


    if (state != null) {

        restoreOriginalRecords(
                activity,
                state,
                session.originalRecords
        );


        showAllRecordRows(
                state
        );

        restoreRecordRowLayout(
                session
        );
    }


    session.editMode =
            false;


    session.deletedRows.clear();


    session.clearedCells.clear();


    session.originalRecords.clear();


    session.focusedCell =
            null;


    SelectionState newState =
            getSelectionState(
                    activity
            );


    if (newState != null) {

        disableEditingOnCells(
                newState
        );


        renumberVisibleSerialNumbers(
                activity,
                newState
        );
    }


    selection(
            activity
    );


    dismissEditPopup(
            activity
    );


    clearEditActionFocus(
            activity
    );
}

     * ACTIVITY / VIEW CONNECTION: Activity/View: internal.

     * WORK FLOW: Connection: clear/delete/edit confirmation helpers.

    // ============================================================
    // SECTION 55 — GETFOCUSEDEDITCELL
    // ============================================================
    //
    // CONNECTION
    // Edit Mode
    //
    // PURPOSE
    // Finds the current focused editable cell using the existing state.
    //
    // DETAILS
    // Internal helper used by Clear/Delete/Save edit actions.
    //
    // GUIDANCE / INFORMATION
    // Check here if actions target the wrong cell.
    //
    // FEATURES
    // Focused-cell lookup.
    //
    // IMPORTANT
    // Existing executable code below is preserved exactly.
    // This section contains documentation only.
    // ============================================================

    private static TextView getFocusedEditCell(
            Activity activity
    ) {

        if (activity == null) {
            return null;
        }


        EditSession session =
                getEditSession(
                        activity
                );


        if (
                session == null
                ||
                !session.editMode
        ) {

            return null;
        }


        if (
                session.focusedCell != null
                &&
                session.focusedCell.getVisibility()
                        != View.GONE
                &&
                session.focusedCell.getParent() != null
        ) {

            return session.focusedCell;
        }


        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return null;
        }


        for (
                int i = 0;
                i < state.cells.size();
                i++
        ) {

            TextView cell =
                    state.cells.get(i);


            if (
                    cell instanceof EditText
                    &&
                    cell.hasFocus()
            ) {

                session.focusedCell =
                        cell;

                return cell;
            }
        }


        return null;
    }

    private static void setConfirmationWindowPosition(
            Activity activity,
            android.view.Window window
    ) {

        if (activity == null || window == null) {
            return;
        }

        android.view.WindowManager.LayoutParams params =
                window.getAttributes();

        switch (CONFIRM_POSITION_MODE) {

            case CONFIRM_POSITION_TOP_LEFT:
                params.gravity =
                        Gravity.TOP | Gravity.LEFT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_TOP_RIGHT:
                params.gravity =
                        Gravity.TOP | Gravity.RIGHT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_BOTTOM_LEFT:
                params.gravity =
                        Gravity.BOTTOM | Gravity.LEFT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_BOTTOM_RIGHT:
                params.gravity =
                        Gravity.BOTTOM | Gravity.RIGHT;
                params.x = dp(activity, CONFIRM_X_DP);
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            case CONFIRM_POSITION_CENTER:
                params.gravity = Gravity.CENTER;
                params.x = 0;
                params.y = dp(activity, CONFIRM_Y_DP);
                break;

            default:
                params.gravity = Gravity.CENTER;
                params.x = 0;
                params.y = 0;
                break;
        }

        window.setAttributes(params);
    }

    // SECTION 60 — CLEARFOCUSEDCELL
    // ============================================================
    //
    // CONNECTION
    // Edit Mode Clear
    //
    // PURPOSE
    // Clears the current focused/selected edit cell according to the existing workflow.
    //
    // DETAILS
    // Handles the source’s pre-edit selection/focused-cell rules.
    //
    // GUIDANCE / INFORMATION
    // Check here when Clear targets the wrong cell or does not clear.
    //
    // FEATURES
    // Cell clearing.
    //
    // IMPORTANT
    // Existing executable code below is preserved exactly.
    // This section contains documentation only.
    // ============================================================

    private static void clearFocusedCell(
            Activity activity
    ) {

        if (activity == null) return;

        EditSession session = getEditSession(activity);

        if (session == null || !session.editMode) return;

        SelectionState state = getSelectionState(activity);

        // A selection made before entering edit mode has priority.
        if (state != null && !session.preEditSelectedCells.isEmpty()) {
            boolean changed = false;

            for (int r = 0; r < state.records.size(); r++) {
                ArrayList<TextView> record = state.records.get(r);
                if (record == null) continue;

                for (int c = 0; c < record.size(); c++) {
                    if (session.preEditSelectedCells.contains(r + ":" + c)) {
                        TextView cell = record.get(c);
                        if (cell != null) {
                            cell.setText("");
                            session.clearedCells.add(cell);
                            changed = true;
                        }
                    }
                }
            }

            if (changed) {
                // Clear has completed for the selected cells. The selection must
                // now become deselected, while Edit Mode itself remains active.
                deselectState(state);
                session.preEditSelectedCells.clear();
                session.focusedCell = null;
                clearEditActionFocus(activity);
                return;
            }
        }

        TextView cell = getFocusedEditCell(activity);

        if (cell == null) {
            return;
        }

        session.focusedCell = cell;
        cell.requestFocus();
        cell.setText("");

        if (cell instanceof EditText) {
            ((EditText) cell).setSelection(0);
        }

        session.clearedCells.add(cell);
        clearEditActionFocus(activity);
    }

    // ============================================================
    //
    // CONNECTION
    // Edit Mode Delete
    //
    // PURPOSE
    // Determines and deletes the current edit row selection using the existing confirmation flow.
    //
    // DETAILS
    // Preserves existing selected-row/focused-row behavior.
    //
    // GUIDANCE / INFORMATION

    private static void deleteFocusedRow(
            final Activity activity
    ) {

        if (activity == null) return;

        final EditSession session = getEditSession(activity);

        if (session == null || !session.editMode) return;

        final SelectionState state = getSelectionState(activity);

        if (state == null) return;

        final Set<Integer> rowsToDelete = new HashSet<Integer>();

        // If cells were selected before Edit Menu was opened, delete the
        // complete records containing those selected cells.
        for (String key : session.preEditSelectedCells) {
            int separator = key.indexOf(":");
            if (separator > 0) {
                try {
                    int row = Integer.parseInt(key.substring(0, separator));
                    if (row >= 0 && row < state.records.size()) {
                        rowsToDelete.add(row);
                    }
                } catch (Exception ignored) {
                }
            }
        }

        // Otherwise delete the complete record containing the currently edited cell.
        if (rowsToDelete.isEmpty()) {
            TextView focusedCell = getFocusedEditCell(activity);
            if (focusedCell == null) {
                return;
            }

            int row = getRecordIndexForView(activity, focusedCell);
            if (row >= 0) rowsToDelete.add(row);
        }

        if (rowsToDelete.isEmpty()) return;

        final int count = rowsToDelete.size();
        final String message =
                count == 1
                ? "Delete this complete record?\n\nName, Phone, Address and Product will be cleared."
                : "Delete these " + count + " complete records?\n\nName, Phone, Address and Product will be cleared.";

        showDeleteConfirmation(
                activity,
                rowsToDelete,
                state,
                session,
                count
        );
    }

    // ============================================================
    // SECTION 62 — FREEZERECORDROWSFOREDIT
    // ============================================================
    //
    // CONNECTION
    // Edit Mode row layout
    //
    // PURPOSE
    // Freezes row layout using the existing implementation while editing.
    //
    // DETAILS
    // Stores original row sizing for later restoration.
    //
    // GUIDANCE / INFORMATION
    // Check here for row expansion/height issues during Edit Mode.
    //
    // FEATURES
    // Row layout stability.
    //
    // IMPORTANT
    // Existing executable code below is preserved exactly.
    // This section contains documentation only.
    // ============================================================

    private static void freezeRecordRowsForEdit(
            SelectionState state,
            EditSession session
    ) {

        if (state == null || session == null || session.rowsFrozen) {
            return;
        }

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();

        for (int r = 0; r < state.records.size(); r++) {
            ArrayList<TextView> record = state.records.get(r);
            if (record == null || record.size() == 0) continue;

            TextView firstCell = record.get(0);
            if (firstCell == null) continue;

            ViewParent parentObject = firstCell.getParent();
            if (!(parentObject instanceof ViewGroup)) continue;

            View row = (View) parentObject;
            ViewGroup.LayoutParams params = row.getLayoutParams();
            if (params == null) continue;

            int measuredHeight = row.getHeight();
            if (measuredHeight <= 0) {
                measuredHeight = firstCell.getHeight();
            }

            if (measuredHeight <= 0) {
                continue;
            }

            session.originalRowHeights.put(
                    row,
                    Integer.valueOf(params.height)
            );

            if (params instanceof LinearLayout.LayoutParams) {
                LinearLayout.LayoutParams lp =
                        (LinearLayout.LayoutParams) params;
                session.originalRowWeights.put(
                        row,
                        Float.valueOf(lp.weight)
                );

                lp.height = measuredHeight;
                lp.weight = 0f;
            } else {
                params.height = measuredHeight;
            }

            row.setLayoutParams(params);
        }

        session.rowsFrozen = true;
    }

    // SECTION 63 — FINALIZERECORDROWLAYOUTAFTERSAVE
    // ============================================================
    //
    // CONNECTION
    // Edit Mode Save
    //
    // PURPOSE
    // Finalizes row layout after the existing save operation.
    //
    // DETAILS
    // Uses the stored row layout state.
    //
    // GUIDANCE / INFORMATION
    // Check here if rows look incorrect immediately after Save.

    private static void finalizeRecordRowLayoutAfterSave(
            EditSession session
    ) {

        if (session == null || !session.rowsFrozen) {
            return;
        }

        for (Map.Entry<View, Integer> entry :
                session.originalRowHeights.entrySet()) {

            View row = entry.getKey();
            if (row == null) {
                continue;
            }

            ViewGroup.LayoutParams params = row.getLayoutParams();
            if (params == null) {
                continue;
            }

            Integer originalHeight = entry.getValue();
            if (originalHeight != null && originalHeight.intValue() > 0) {
                params.height = originalHeight.intValue();
            }

            // Keep weight at zero after Save. Deleted rows are GONE, so
            // restoring the old weights would redistribute height.
            if (params instanceof LinearLayout.LayoutParams) {
                ((LinearLayout.LayoutParams) params).weight = 0f;
            }

            row.setLayoutParams(params);
        }

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();
        session.rowsFrozen = false;
    }

    private static void restoreRecordRowLayout(
            EditSession session
    ) {

        if (session == null || !session.rowsFrozen) {
            return;
        }

        for (Map.Entry<View, Integer> entry :
                session.originalRowHeights.entrySet()) {

            View row = entry.getKey();
            if (row == null) continue;

            ViewGroup.LayoutParams params = row.getLayoutParams();
            if (params == null) continue;

            Integer originalHeight = entry.getValue();
            if (originalHeight != null) {
                params.height = originalHeight.intValue();
            }

            if (params instanceof LinearLayout.LayoutParams) {
                Float originalWeight =
                        session.originalRowWeights.get(row);
                if (originalWeight != null) {
                    ((LinearLayout.LayoutParams) params).weight =
                            originalWeight.floatValue();
                }
            }

            row.setLayoutParams(params);
        }

        session.originalRowHeights.clear();
        session.originalRowWeights.clear();
        session.rowsFrozen = false;
    }

    LinearLayout container =
            activity.findViewById(

    // ============================================================
    // SECTION 67 — SAVEEDITEDRECORDS
    // ============================================================
    //
    // CONNECTION
    // Edit Menu Save
    //
    // PURPOSE
    // Persists the existing edited record state to myrecords.txt.
    //
    // DETAILS
    // Keeps current record formatting, deletion/clear handling, and file-write flow.
    //
    // GUIDANCE / INFORMATION
    // For Save corruption/loss, inspect this section and row/cell state helpers.
    //
    // FEATURES
    // Edited-record persistence.
    //
    // IMPORTANT
    // Existing executable code below is preserved exactly.
    // This section contains documentation only.
    // ============================================================

    private static void saveEditedRecords(
            Activity activity
    ) {

        if (activity == null) {
            return;
        }


        EditSession session =
                getEditSession(
                        activity
                );


        if (
                session == null
                ||
                !session.editMode
        ) {

            return;
        }


        SelectionState state =
                getSelectionState(
                        activity
                );


        if (state == null) {
            return;
        }


        ArrayList<String> updatedRecords =
                new ArrayList<String>();


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            if (
                    session.deletedRows.contains(
                            r
                    )
            ) {

                continue;
            }


            ArrayList<TextView> record =
                    state.records.get(r);


            if (record == null) {
                continue;
            }


            String name = "";

            String phone = "";

            String address = "";

            String product = "";


            if (
                    record.size() > 0
                    &&
                    record.get(0) != null
            ) {

                name =
                        record.get(0)
                                .getText()
                                .toString()
                                .trim();
            }


            if (
                    record.size() > 1
                    &&
                    record.get(1) != null
            ) {

                phone =
                        record.get(1)
                                .getText()
                                .toString()
                                .trim();
            }


            if (
                    record.size() > 2
                    &&
                    record.get(2) != null
            ) {

                address =
                        record.get(2)
                                .getText()
                                .toString()
                                .trim();
            }


            if (
                    record.size() > 3
                    &&
                    record.get(3) != null
            ) {

                product =
                        record.get(3)
                                .getText()
                                .toString()
                                .trim();
            }


            String finalLine =
                    name
                            + " | "
                            + phone
                            + " | "
                            + address
                            + " | "
                            + product;


            updatedRecords.add(
                    finalLine
            );
        }


        FileOutputStream output =
                null;


        try {

            output =
                    activity.openFileOutput(
                            RECORDS_FILE,
                            Context.MODE_PRIVATE
                    );


            for (
                    int i = 0;
                    i < updatedRecords.size();
                    i++
            ) {

                String line =
                        updatedRecords.get(i);


                output.write(
                        (
                                line
                                        +
                                        "\n"
                        ).getBytes("UTF-8")
                );
            }


            output.flush();

            output.close();

            output = null;


            // Replace EditTexts with normal TextViews after Save so the
            // EditText red underline cannot remain visible.
            restoreEditedRecordsAsTextViews(
                    activity,
                    state
            );

            // Keep the original compact row heights after Save.
            // Do not restore old row weights here because deleted GONE rows
            // would otherwise make remaining rows expand vertically.
            finalizeRecordRowLayoutAfterSave(
                    session
            );


            session.editMode =
                    false;

            session.deletedRows.clear();

            session.clearedCells.clear();

            session.originalRecords.clear();

            session.preEditSelectedCells.clear();

            session.focusedCell =
                    null;


            disableEditingOnCells(
                    state
            );


            // Save completes the edit operation, so the previous selection
            // must now be fully deselected.
            deselectAll(
                    activity
            );


            selection(
                    activity
            );


            renumberVisibleSerialNumbers(
                    activity,
                    state
            );


            dismissEditPopup(
                    activity
            );


            clearEditActionFocus(
                    activity
            );


            Toast.makeText(
                    activity,
                    "Data Update",
                    Toast.LENGTH_SHORT
            ).show();


        } catch (Exception e) {

            Toast.makeText(
                    activity,
                    "Unable to update records",
                    Toast.LENGTH_SHORT
            ).show();


        } finally {

            if (output != null) {

                try {
                    output.close();
                } catch (Exception ignored) {
                }
            }
        }
    }

                    c < record.size();
                    c++
            ) {

                TextView cell =
                        record.get(c);


                if (cell != null) {

                    recordCells.add(
                            cell
                    );
                }
            }
        }


        for (
                int r = 0;
                r < state.records.size();
                r++
        ) {

            ArrayList<TextView> record =
                    state.records.get(r);


            if (
                    record == null
                    ||
                    record.size() == 0
            ) {
                continue;
            }


            TextView firstCell =
                    record.get(0);


            if (firstCell == null) {
                continue;
            }


            ViewParent parent =
                    firstCell.getParent();


            if (!(parent instanceof ViewGroup)) {
                continue;
            }


            ViewGroup row =
                    (ViewGroup)
                            parent;


            if (
                    row.getVisibility()
                            !=
                            View.VISIBLE
            ) {

                continue;
            }


            TextView serialView =
                    findSerialTextView(
                            row,
                            recordCells
                    );


            if (serialView != null) {

                serialView.setText(
                        String.valueOf(
                                serialNumber
                        )
                );


                serialNumber++;
            }
        }
    }

    // ============================================================
    //
    // CONNECTION

    // Finds the existing serial-number TextView for a row.

    private static TextView findSerialTextView(
            ViewGroup row,
            Set<TextView> recordCells
    ) {

        if (
                row == null
                ||
                recordCells == null
        ) {
            return null;
        }


        for (
                int i = 0;
                i < row.getChildCount();
                i++
        ) {

            View child =
                    row.getChildAt(i);


            if (
                    child instanceof TextView
                    &&
                    !(child instanceof EditText)
            ) {

                TextView textView =
                        (TextView)
                                child;


                if (
                        !recordCells.contains(
                                textView
                        )
                ) {

                    return textView;
                }
            }
        }


        for (
                int i = 0;
                i < row.getChildCount();
                i++
        ) {

            View child =
                    row.getChildAt(i);


            if (child instanceof ViewGroup) {

                TextView found =
                        findSerialTextView(
                                (ViewGroup) child,
                                recordCells
                        );


                if (found != null) {
                    return found;
                }
            }
        }


        return null;
    }

                LayoutInflater

            return;
        }

                new PopupWindow(
                        popupView,
                        ViewGroup.LayoutParams.WRAP_CONTENT,
                        ViewGroup.LayoutParams.WRAP_CONTENT,
                        true

                true

                true
        );

                new ColorDrawable(
                        Color.TRANSPARENT
                )
        );

                popupView.findViewById(
                        R.id.copyButton
                );

                popupView.findViewById(
                        R.id.shareButton
                );

                popupView.findViewById(
                        R.id.exportButton
                );

                popupView.findViewById(
                        R.id.selectAllButton

                popupView.findViewById(
                        R.id.deselectAllButton
                );

        int yOffset = 0;

                POPUP_GRAVITY;


        if (
                POPUP_POSITION_MODE
                        ==
                        POSITION_TOP
        ) {

            gravity =
                    Gravity.TOP
                            |
                            Gravity.CENTER_HORIZONTAL;

            xOffset =
                    dp(
                            activity,
                            POPUP_X_ADJUST_DP
                    );

                    dp(
                            activity,
                            POPUP_TOP_DP

                    );

                    Gravity.BOTTOM

                    dp(
                            activity,
                            POPUP_X_ADJUST_DP
                    );

            yOffset =
                    -dp(
                            activity,
                            POPUP_BOTTOM_DP
                    )
                    +
                    dp(
                            activity,
                            POPUP_Y_ADJUST_DP
                    );
        }

                POPUP_POSITION_MODE
                        ==
                        POSITION_LEFT
        ) {

                    Gravity.CENTER_VERTICAL
                            |

            xOffset =
                    dp(
                            activity,
                            POPUP_LEFT_DP
                    )
                    +
                    dp(

                            POPUP_X_ADJUST_DP
                    );

            yOffset =
                    dp(
                            activity,
                            POPUP_Y_ADJUST_DP
                    );

                        ==
                        POSITION_RIGHT
        ) {

                            |

            xOffset =
                    dp(

                            POPUP_RIGHT_DP
                    )
                    +

                    );

                        }
                    }
            );
        }

            shareButton.setOnClickListener(
                    new View.OnClickListener() {

                        @Override
                        public void onClick(
                                View v
                        ) {

                            share(
                                    activity
                            );

                            popupWindow.dismiss();
                        }
                    }
            );
        }

            exportButton.setOnClickListener(
                    new View.OnClickListener() {

                        public void onClick(

    // ============================================================
    //
    // CONNECTION
    // Internal UI utility
    //
    // PURPOSE
    // Converts dp to pixels using the current device density.
    //
    // DETAILS
    // Used by existing popup/confirmation/UI sizing logic.
    //
    // GUIDANCE / INFORMATION
    // Check here only for density-related sizing issues.
    //
    // FEATURES
    // DP conversion.
    //
    // IMPORTANT
    // Existing executable code below is preserved exactly.
    // This section contains documentation only.
    // ============================================================

    private static int dp(
            Activity activity,
            int value
    ) {

        if (activity == null) {
            return value;
        }


        float density =
                activity
                        .getResources()
                        .getDisplayMetrics()
                        .density;


        return (int)(
                value
                        * density
                        + 0.5f
        );
    }
