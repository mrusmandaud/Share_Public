// ============================================================
// SECTION 06 — RECORD FILE READING
// ============================================================
//
// CONNECTION
// myrecords.txt
//
// PURPOSE
// Reads saved records using the existing file-reading logic.
//
// DETAILS
// This section documents the existing onCreate block and how it
// connects to MyrecordsActivity, myrecords.xml, and ButtonsController.
//
// GUIDANCE / INFORMATION
// For missing or shifted records, inspect file reading and record parsing before changing table UI.
//
// FEATURES
// Record loading.
//
// IMPORTANT
// Existing executable Java code in this section is preserved exactly.
// Documentation only — no IDs, listeners, values, or logic are changed.
// ============================================================

try {

    java.io.FileInputStream fileInputStream =
            openFileInput("myrecords.txt");

    java.io.BufferedReader reader =
            new java.io.BufferedReader(
                    new java.io.InputStreamReader(
                            fileInputStream,
                            "UTF-8"
                    )
            );

    String line;

    boolean firstLine = true;

while ((line = reader.readLine()) != null) {
    line = line.trim();

    if (firstLine) {
        firstLine = false;
        continue;
    }

    if (line.length() > 0) {
        allRecords.add(line);
    }

    }

    reader.close();
    fileInputStream.close();

} catch (Exception e) {

    allRecords.clear();

}

// ============================================================
// SEARCH
// ============================================================

searchInput.addTextChangedListener(
        new android.text.TextWatcher() {

            @Override
            public void beforeTextChanged(
                    CharSequence s,
                    int start,
                    int count,
                    int after
            ) {
            }


            @Override
            public void onTextChanged(
                    CharSequence s,
                    int start,
                    int before,
                    int count
            ) {

                String query = "";

                if (s != null) {

                    query =
                            s.toString()
                                    .trim()
                                    .toLowerCase(
                                            java.util.Locale
                                                    .getDefault()
                                    );

                }

                int visibleCount = 0;


                // ====================================================
                // CHECK EVERY RECORD
                // ====================================================

                for (
                        int i = 0;
                        i < recordsContainer.getChildCount();
                        i++
                ) {

                    android.view.View child =
                            recordsContainer.getChildAt(i);


                    if (child.getTag() == null) {

                        continue;

                    }


                    String text =
                            child.getTag()
                                    .toString()
                                    .toLowerCase(
                                            java.util.Locale
                                                    .getDefault()
                                    );


                    if (
                            query.length() == 0 ||
                            text.contains(query)
                    ) {

                        child.setVisibility(
                                android.view.View.VISIBLE
                        );

                        visibleCount++;

                    } else {

                        child.setVisibility(
                                android.view.View.GONE
                        );

                    }

                }


                // ====================================================
                // SEARCH EMPTY STATE
                // ====================================================

                if (visibleCount == 0) {

                    emptyState.setVisibility(
                            android.view.View.VISIBLE
                    );


                    if (query.length() == 0) {

                        emptyTitle.setText(
                                "No Records Yet"
                        );

                        emptyMessage.setText(
                                "Your saved records will appear here."
                        );

                    } else {

                        emptyTitle.setText(
