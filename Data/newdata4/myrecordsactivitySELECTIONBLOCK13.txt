// ============================================================
// SELECTION
//
// SELECTABLE DATA:
//
// Name
// Phone
// Address
// Product
//
// NEVER SELECTABLE:
//
// # / serial number
// Headers
// Lines
// Record container
//
// TAP:
// Single cell
//
// TAP A -> H:
// Inclusive range
//
// TAP selected cell again:
// Deselect current selection
//
// LONG PRESS:
// ONLY OPEN POPUP MENU
// ============================================================


// ============================================================
// GET SHARED RECORDS
// ============================================================

final android.widget.LinearLayout selectionContainer =
        (android.widget.LinearLayout)
                findViewById(
                        R.id.recordsContainer
                );

if (selectionContainer == null) {
    return;
}


Object selectionObject =
        selectionContainer.getTag();

if (!(selectionObject instanceof
        java.util.ArrayList)) {
    return;
}


final java.util.ArrayList<
        java.util.ArrayList<
                android.widget.TextView
        >
> selectionRecords =
        (java.util.ArrayList<
                java.util.ArrayList<
                        android.widget.TextView
                >
        >) selectionObject;


if (selectionRecords.size() == 0) {
    return;
}


// ============================================================
// FLAT DATA CELL LIST
//
// ONLY:
// Name
// Phone
// Address
// Product
// ============================================================

final java.util.ArrayList<
        android.widget.TextView
> cells =
        new java.util.ArrayList<
                android.widget.TextView
        >();


for (
        int r = 0;
        r < selectionRecords.size();
        r++
) {

    java.util.ArrayList<
            android.widget.TextView
    > record =
            selectionRecords.get(r);


    for (
            int c = 0;
            c < record.size();
            c++
    ) {

        android.widget.TextView cell =
                record.get(c);

        if (cell != null) {
            cells.add(cell);
        }

    }

}


if (cells.size() == 0) {
    return;
}


// ============================================================
// COLORS
// ============================================================

final int BLUE =
        android.graphics.Color.rgb(
                33,
                150,
                243
        );

final int WHITE =
        android.graphics.Color.WHITE;

final int BLACK =
        android.graphics.Color.BLACK;


// ============================================================
// CURRENT RANGE
// ============================================================

final int[] firstIndex =
        new int[] {
                -1
        };

final int[] lastIndex =
        new int[] {
                -1
        };


// ============================================================
// CLEAR ALL BLUE CELLS
// ============================================================

final Runnable clearBlue =
        new Runnable() {

            @Override
            public void run() {

                for (
                        int i = 0;
                        i < cells.size();
                        i++
                ) {

                    android.widget.TextView cell =
                            cells.get(i);

                    if (cell != null) {

                        cell.setBackgroundColor(
                                WHITE
                        );

                        cell.setTextColor(
                                BLACK
                        );

                    }

                }

            }

        };


// ============================================================
// DESELECT ALL
//
// THIS IS THE REAL DESELECT-ALL LOGIC.
// ============================================================

final Runnable deselectAll =
        new Runnable() {

            @Override
            public void run() {

                clearBlue.run();

                firstIndex[0] =
                        -1;

                lastIndex[0] =
                        -1;

            }

        };


// ============================================================
// SELECT ALL
//
// THIS IS THE REAL SELECT-ALL LOGIC.
// ============================================================

final Runnable selectAll =
        new Runnable() {

            @Override
            public void run() {

                if (cells.size() == 0) {
                    return;
                }

                for (
                        int i = 0;
                        i < cells.size();
                        i++
                ) {

                    android.widget.TextView cell =
                            cells.get(i);

                    if (cell != null) {

                        cell.setBackgroundColor(
                                BLUE
                        );

                        cell.setTextColor(
                                WHITE
                        );

                    }

                }

                firstIndex[0] =
                        0;

                lastIndex[0] =
                        cells.size() - 1;

            }

        };


// ============================================================
// CREATE SELECTION BRIDGE
//
// popup_menu does NOT contain Select All logic.
// It only calls these two Runnable objects.
//
// popup:
// ((Runnable)((HashMap)tag).get("selectAll")).run();
//
// popup:
// ((Runnable)((HashMap)tag).get("deselectAll")).run();
// ============================================================

final java.util.HashMap<String, Object>
        selectionBridge =
        new java.util.HashMap<String, Object>();


selectionBridge.put(
        "records",
        selectionRecords
);


selectionBridge.put(
        "selectAll",
        selectAll
);


selectionBridge.put(
        "deselectAll",
        deselectAll
);


selectionContainer.setTag(
        selectionBridge
);


// ============================================================
// TAP HANDLERS
// ============================================================

for (
        int i = 0;
        i < cells.size();
        i++
) {

    final android.widget.TextView cell =
            cells.get(i);

    final int index =
            i;


    cell.setClickable(true);


    // ========================================================
    // NORMAL TAP
    // ========================================================

    cell.setOnClickListener(
            new android.view.View.OnClickListener() {

                @Override
                public void onClick(
                        android.view.View v
                ) {


                    // ==========================================
                    // FIRST CELL
                    // ==========================================

                    if (
                            firstIndex[0] == -1
                    ) {

                        clearBlue.run();

                        cell.setBackgroundColor(
                                BLUE
                        );

                        cell.setTextColor(
                                WHITE
                        );


                        firstIndex[0] =
                                index;

                        lastIndex[0] =
                                index;

                        return;
                    }


                    // ==========================================
                    // SAME SINGLE CELL
                    // ==========================================

                    if (
                            firstIndex[0] ==
                                    lastIndex[0]
                            &&
                            index ==
                                    firstIndex[0]
                    ) {

                        deselectAll.run();

                        return;
                    }


                    // ==========================================
                    // CURRENT RANGE
                    // ==========================================

                    int currentStart =
                            firstIndex[0];

                    int currentEnd =
                            lastIndex[0];


                    if (
                            currentStart >
                                    currentEnd
                    ) {

                        int temp =
                                currentStart;

                        currentStart =
                                currentEnd;

                        currentEnd =
                                temp;
                    }


                    // ==========================================
                    // TAP INSIDE CURRENT RANGE
                    // ==========================================

                    if (
                            index >= currentStart
                            &&
                            index <= currentEnd
                    ) {

                        deselectAll.run();

                        return;
                    }


                    // ==========================================
                    // SECOND TAP
                    //
                    // A -> H
                    // H -> A
                    // ==========================================

                    int start =
                            firstIndex[0];

                    int end =
                            index;


                    if (start > end) {

                        int temp =
                                start;

                        start =
                                end;

                        end =
                                temp;
                    }


                    clearBlue.run();


                    // ==========================================
                    // INCLUSIVE RANGE
                    // ==========================================

                    for (
                            int x = start;
                            x <= end;
                            x++
                    ) {

                        android.widget.TextView
                                selected =
                                cells.get(x);


                        if (selected != null) {

                            selected.setBackgroundColor(
                                    BLUE
                            );

                            selected.setTextColor(
                                    WHITE
                            );

                        }

                    }


                    firstIndex[0] =
                            start;

                    lastIndex[0] =
                            end;

                }

            }
    );


    // ========================================================
    // LONG PRESS
    //
    // NO SELECTION CHANGE.
    // ONLY OPEN POPUP.
    // ========================================================

    cell.setOnLongClickListener(
            new android.view.View.OnLongClickListener() {

                @Override
                public boolean onLongClick(
                        android.view.View v
                ) {

                    _popup_menu();

                    return true;
                }

            }
    );

}
