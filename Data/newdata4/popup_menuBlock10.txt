// ============================================================
// COPY
// ============================================================

copyButton.setOnClickListener(
        new android.view.View.OnClickListener() {

            @Override
            public void onClick(
                    android.view.View v
            ) {

                StringBuilder selectedText =
                        new StringBuilder();


                for (
                        int r = 0;
                        r < popupSelectionRecords.size();
                        r++
                ) {

                    java.util.ArrayList<
                            android.widget.TextView
                    > record =
                            popupSelectionRecords.get(r);


                    if (record == null) {
                        continue;
                    }


                    StringBuilder recordText =
                            new StringBuilder();


                    for (
                            int c = 0;
                            c < record.size();
                            c++
                    ) {

                        android.widget.TextView cell =
                                record.get(c);


                        if (cell == null) {
                            continue;
                        }


                        android.graphics.drawable.Drawable
                                background =
                                cell.getBackground();


                        boolean selected =
                                false;


                        if (
                                background instanceof
                                android.graphics.drawable.ColorDrawable
                        ) {

                            android.graphics.drawable.ColorDrawable
                                    colorBackground =
                                    (android.graphics.drawable.ColorDrawable)
                                            background;


                            if (
                                    colorBackground.getColor()
                                            == BLUE
                            ) {

                                selected =
                                        true;

                            }

                        }


                        if (selected) {

                            if (
                                    recordText.length() > 0
                            ) {

                                recordText.append(
                                        " | "
                                );

                            }


                            recordText.append(
                                    cell.getText()
                                            .toString()
                            );

                        }

                    }


                    if (
                            recordText.length() > 0
                    ) {

                        if (
                                selectedText.length() > 0
                        ) {

                            selectedText.append(
                                    "\n"
                            );

                        }


                        selectedText.append(
                                recordText.toString()
                        );

                    }

                }


                if (
                        selectedText.length() == 0
                ) {

                    android.widget.Toast.makeText(
                            MyrecordsActivity.this,
                            "No data selected",
                            android.widget.Toast.LENGTH_SHORT
                    ).show();

                } else {

                    android.content.ClipboardManager
                            clipboard =
                            (android.content.ClipboardManager)
                                    getSystemService(
                                            android.content.Context
                                                    .CLIPBOARD_SERVICE
                                    );


                    clipboard.setPrimaryClip(
                            android.content.ClipData
                                    .newPlainText(
                                            "Selected Records",
                                            selectedText.toString()
                                    )
                    );


                    android.widget.Toast.makeText(
                            MyrecordsActivity.this,
                            "Copied",
                            android.widget.Toast.LENGTH_SHORT
                    ).show();

                }


                popupWindow.dismiss();

            }

        }
);
