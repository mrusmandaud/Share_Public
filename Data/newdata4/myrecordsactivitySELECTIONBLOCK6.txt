// ============================================================
// SELECTION
//
// ONLY THESE CELLS ARE SELECTABLE:
//
// Name
// Phone
// Address
// Product
//
// NEVER SELECT:
//
// # / 1 / 2 / 3 / 4 / 5 / ...
// headers
// grid lines
// record containers
// ============================================================


// ============================================================
// GET SHARED SELECTION RECORDS
// ============================================================

final android.widget.LinearLayout selectionContainer =
        (android.widget.LinearLayout)
                findViewById(
                        R.id.recordsContainer
                );

if (selectionContainer == null) {
    return;
}


Object selectionObject =
        selectionContainer.getTag();

if (
        !(selectionObject instanceof
                java.util.ArrayList)
) {
    return;
}


final java.util.ArrayList<
        java.util.ArrayList<
                android.widget.TextView
        >
> selectionRecords =
        (java.util.ArrayList<
                java.util.ArrayList<
                        android.widget.TextView
                >
        >)
                selectionObject;


if (
        selectionRecords.size() == 0
) {
    return;
}


// ============================================================
// FLAT CELL LIST
//
// Order:
//
// Record 1 Name
// Record 1 Phone
// Record 1 Address
// Record 1 Product
// Record 2 Name
// Record 2 Phone
// Record 2 Address
// Record 2 Product
// ...
//
// # IS NEVER HERE.
// ============================================================

final java.util.ArrayList<
        android.widget.TextView
> cells =
        new java.util.ArrayList<
                android.widget.TextView
        >();


for (
        int r = 0;
        r < selectionRecords.size();
        r++
) {

    java.util.ArrayList<
            android.widget.TextView
    > record =
            selectionRecords.get(r);


    for (
            int c = 0;
            c < record.size();
            c++
    ) {

        android.widget.TextView cell =
                record.get(c);

        if (cell != null) {

            cells.add(cell);

        }

    }

}


if (cells.size() == 0) {
    return;
}


// ============================================================
// COLORS
// ============================================================

final int BLUE =
        android.graphics.Color.rgb(
                33,
                150,
                243
        );

final int WHITE =
        android.graphics.Color.WHITE;

final int BLACK =
        android.graphics.Color.BLACK;


// ============================================================
// SELECTION STATE
// ============================================================

final int[] firstIndex =
        new int[] {
                -1
        };

final int[] lastIndex =
        new int[] {
                -1
        };


// ============================================================
// CLEAR BLUE
// ============================================================

final Runnable clearBlue =
        new Runnable() {

            @Override
            public void run() {

                for (
                        int i = 0;
                        i < cells.size();
                        i++
                ) {

                    android.widget.TextView cell =
                            cells.get(i);

                    cell.setBackgroundColor(
                            WHITE
                    );

                    cell.setTextColor(
                            BLACK
                    );

                }

            }

        };


// ============================================================
// DESELECT EVERYTHING
// ============================================================

final Runnable deselectAll =
        new Runnable() {

            @Override
            public void run() {

                clearBlue.run();

                firstIndex[0] =
                        -1;

                lastIndex[0] =
                        -1;

            }

        };


// ============================================================
// SELECT RANGE
// ============================================================

final Runnable applyCurrentRange =
        new Runnable() {

            @Override
            public void run() {

                if (
                        firstIndex[0] < 0 ||
                        lastIndex[0] < 0
                ) {
                    return;
                }


                int start =
                        firstIndex[0];

                int end =
                        lastIndex[0];


                if (start > end) {

                    int temp =
                            start;

                    start =
                            end;

                    end =
                            temp;

                }


                for (
                        int i = start;
                        i <= end;
                        i++
                ) {

                    if (
                            i >= 0 &&
                            i < cells.size()
                    ) {

                        android.widget.TextView cell =
                                cells.get(i);

                        cell.setBackgroundColor(
                                BLUE
                        );

                        cell.setTextColor(
                                WHITE
                        );

                    }

                }

            }

        };


// ============================================================
// INSTALL TOUCH HANDLERS
// ============================================================

for (
        int i = 0;
        i < cells.size();
        i++
) {

    final android.widget.TextView cell =
            cells.get(i);

    final int index =
            i;


    cell.setClickable(true);


    // ========================================================
    // SINGLE TAP / RANGE TAP
    // ========================================================

    cell.setOnClickListener(
            new android.view.View.OnClickListener() {

                @Override
                public void onClick(
                        android.view.View v
                ) {


                    // ==========================================
                    // FIRST TAP
                    // ==========================================

                    if (
                            firstIndex[0] == -1
                    ) {

                        clearBlue.run();

                        cell.setBackgroundColor(
                                BLUE
                        );

                        cell.setTextColor(
                                WHITE
                        );

                        firstIndex[0] =
                                index;

                        lastIndex[0] =
                                index;

                        return;

                    }


                    // ==========================================
                    // SAME SINGLE CELL
                    //
                    // Tap selected cell again:
                    // deselect it.
                    // ==========================================

                    if (
                            firstIndex[0] ==
                                    lastIndex[0]
                            &&
                            index ==
                                    firstIndex[0]
                    ) {

                        deselectAll.run();

                        return;

                    }


                    // ==========================================
                    // CURRENT RANGE
                    //
                    // Tapping any already-blue cell
                    // clears the whole selection.
                    // ==========================================

                    int currentStart =
                            firstIndex[0];

                    int currentEnd =
                            lastIndex[0];


                    if (
                            currentStart >
                                    currentEnd
                    ) {

                        int temp =
                                currentStart;

                        currentStart =
                                currentEnd;

                        currentEnd =
                                temp;

                    }


                    if (
                            index >= currentStart &&
                            index <= currentEnd
                    ) {

                        deselectAll.run();

                        return;

                    }


                    // ==========================================
                    // SECOND TAP
                    //
                    // Select inclusive range.
                    //
                    // Works:
                    //
                    // A -> H
                    // H -> A
                    // ==========================================

                    int start =
                            firstIndex[0];

                    int end =
                            index;


                    if (start > end) {

                        int temp =
                                start;

                        start =
                                end;

                        end =
                                temp;

                    }


                    clearBlue.run();


                    for (
                            int x = start;
                            x <= end;
                            x++
                    ) {

                        android.widget.TextView selected =
                                cells.get(x);

                        selected.setBackgroundColor(
                                BLUE
                        );

                        selected.setTextColor(
                                WHITE
                        );

                    }


                    firstIndex[0] =
                            start;

                    lastIndex[0] =
                            end;

                }

            }
    );


    // ========================================================
    // LONG PRESS
    //
    // SELECT COMPLETE RECORD.
    //
    // ONLY:
    //
    // Name
    // Phone
    // Address
    // Product
    //
    // NEVER #.
    // ========================================================

    cell.setOnLongClickListener(
            new android.view.View.OnLongClickListener() {

                @Override
                public boolean onLongClick(
                        android.view.View v
                ) {

                    java.util.ArrayList<
                            android.widget.TextView
                    > selectedRecord =
                            null;


                    // ==========================================
                    // FIND RECORD CONTAINING THIS CELL
                    // ==========================================

                    for (
                            int r = 0;
                            r < selectionRecords.size();
                            r++
                    ) {

                        java.util.ArrayList<
                                android.widget.TextView
                        > record =
                                selectionRecords.get(r);


                        if (
                                record.contains(cell)
                        ) {

                            selectedRecord =
                                    record;

                            break;

                        }

                    }


                    if (
                            selectedRecord == null
                    ) {

                        return true;

                    }


                    // ==========================================
                    // CLEAR OLD SELECTION
                    // ==========================================

                    clearBlue.run();


                    // ==========================================
                    // SELECT WHOLE RECORD
                    // ==========================================

                    for (
                            int x = 0;
                            x < selectedRecord.size();
                            x++
                    ) {

                        android.widget.TextView selected =
                                selectedRecord.get(x);


                        selected.setBackgroundColor(
                                BLUE
                        );

                        selected.setTextColor(
                                WHITE
                        );

                    }


                    // ==========================================
                    // STORE RANGE
                    // ==========================================

                    int start =
                            cells.indexOf(
                                    selectedRecord.get(0)
                            );

                    int end =
                            cells.indexOf(
                                    selectedRecord.get(
                                            selectedRecord.size() - 1
                                    )
                            );


                    if (start > end) {

                        int temp =
                                start;

                        start =
                                end;

                        end =
                                temp;

                    }


                    firstIndex[0] =
                            start;

                    lastIndex[0] =
                            end;


                    return true;

                }

            }
    );

}
