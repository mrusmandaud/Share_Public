package com.myrecords.onenewexperience.one;

import android.animation.ValueAnimator;
import android.content.Context;
import android.graphics.Color;
import android.graphics.drawable.GradientDrawable;
import android.os.Handler;
import android.os.Looper;
import android.view.Gravity;
import android.view.MotionEvent;
import android.view.View;
import android.view.ViewGroup;
import android.view.animation.DecelerateInterpolator;
import android.widget.FrameLayout;
import android.widget.ImageView;
import android.widget.ScrollView;

public class CustomScrollbar extends FrameLayout {

    // ============================================================
    // CONFIGURATION
    // ============================================================

    // Main scrollbar height
    private static final int BAR_HEIGHT_DP = 600;

    // Thumb height
    private static final int THUMB_HEIGHT_DP = 200;

    // Width
    private static final int BAR_WIDTH_DP = 46;
    private static final int TRACK_WIDTH_DP = 10;
    private static final int THUMB_WIDTH_DP = 10;

    // Arrow button size
    private static final int ARROW_SIZE_DP = 38;

    // Spacing
    private static final int ARROW_MARGIN_DP = 4;

    // Colors
    private static final int TRACK_COLOR = Color.rgb(232, 235, 240);
    private static final int THUMB_COLOR = Color.rgb(70, 110, 220);
    private static final int THUMB_PRESSED_COLOR = Color.rgb(45, 82, 185);
    private static final int ARROW_COLOR = Color.rgb(70, 110, 220);
    private static final int ARROW_PRESSED_COLOR = Color.rgb(45, 82, 185);

    // Corners
    private static final int TRACK_RADIUS_DP = 8;
    private static final int THUMB_RADIUS_DP = 8;
    private static final int ARROW_RADIUS_DP = 19;

    // Page movement
    private static final float PAGE_FACTOR = 0.85f;

    // Smooth scrolling duration
    private static final int SCROLL_DURATION = 220;

    // Fade
    private static final int FADE_DELAY = 1500;
    private static final int FADE_DURATION = 350;

    // ============================================================
    // VIEWS
    // ============================================================

    private ScrollView scrollView;

    private FrameLayout scrollbarContainer;
    private View track;
    private View thumb;

    private ImageView topButton;
    private ImageView bottomButton;

    // ============================================================
    // INTERNAL VALUES
    // ============================================================

    private int barHeight;
    private int thumbHeight;
    private int barWidth;

    private float downY;
    private float startThumbY;

    private boolean draggingThumb = false;
    private boolean updatingFromScroll = false;

    private final Handler fadeHandler =
            new Handler(Looper.getMainLooper());

    private Runnable fadeRunnable;

    // ============================================================
    // CONSTRUCTOR
    // ============================================================

    public CustomScrollbar(Context context) {
        super(context);

        setClipChildren(false);
        setClipToPadding(false);

        barHeight = dp(BAR_HEIGHT_DP);
        thumbHeight = dp(THUMB_HEIGHT_DP);
        barWidth = dp(BAR_WIDTH_DP);

        buildUI();
    }

    // ============================================================
    // BUILD COMPLETE UI
    // ============================================================

    private void buildUI() {

        setLayoutParams(new ViewGroup.LayoutParams(
                barWidth,
                barHeight
        ));

        setGravity(Gravity.CENTER);

        // --------------------------------------------------------
        // MAIN CONTAINER
        // --------------------------------------------------------

        scrollbarContainer = new FrameLayout(getContext());

        LayoutParams containerParams =
                new LayoutParams(
                        barWidth,
                        barHeight
                );

        containerParams.gravity = Gravity.CENTER;

        addView(scrollbarContainer, containerParams);

        // --------------------------------------------------------
        // TRACK
        // --------------------------------------------------------

        track = new View(getContext());

        GradientDrawable trackDrawable =
                new GradientDrawable();

        trackDrawable.setColor(TRACK_COLOR);
        trackDrawable.setCornerRadius(
                dp(TRACK_RADIUS_DP)
        );

        track.setBackground(trackDrawable);

        LayoutParams trackParams =
                new LayoutParams(
                        dp(TRACK_WIDTH_DP),
                        barHeight
                );

        trackParams.gravity = Gravity.CENTER_HORIZONTAL;

        scrollbarContainer.addView(
                track,
                trackParams
        );

        // --------------------------------------------------------
        // THUMB
        // --------------------------------------------------------

        thumb = new View(getContext());

        GradientDrawable thumbDrawable =
                new GradientDrawable();

        thumbDrawable.setColor(THUMB_COLOR);
        thumbDrawable.setCornerRadius(
                dp(THUMB_RADIUS_DP)
        );

        thumb.setBackground(thumbDrawable);

        thumb.setElevation(dp(4));

        LayoutParams thumbParams =
                new LayoutParams(
                        dp(THUMB_WIDTH_DP),
                        thumbHeight
                );

        thumbParams.gravity = Gravity.TOP |
                Gravity.CENTER_HORIZONTAL;

        thumbParams.topMargin = 0;

        scrollbarContainer.addView(
                thumb,
                thumbParams
        );

        // --------------------------------------------------------
        // TOP BUTTON
        // --------------------------------------------------------

        topButton = createArrowButton(true);

        LayoutParams topParams =
                new LayoutParams(
                        dp(ARROW_SIZE_DP),
                        dp(ARROW_SIZE_DP)
                );

        topParams.gravity =
                Gravity.TOP |
                Gravity.CENTER_HORIZONTAL;

        topParams.topMargin =
                -dp(ARROW_MARGIN_DP);

        scrollbarContainer.addView(
                topButton,
                topParams
        );

        // --------------------------------------------------------
        // BOTTOM BUTTON
        // --------------------------------------------------------

        bottomButton = createArrowButton(false);

        LayoutParams bottomParams =
                new LayoutParams(
                        dp(ARROW_SIZE_DP),
                        dp(ARROW_SIZE_DP)
                );

        bottomParams.gravity =
                Gravity.BOTTOM |
                Gravity.CENTER_HORIZONTAL;

        bottomParams.bottomMargin =
                -dp(ARROW_MARGIN_DP);

        scrollbarContainer.addView(
                bottomButton,
                bottomParams
        );

        // --------------------------------------------------------
        // TOUCH EVENTS
        // --------------------------------------------------------

        setupThumbTouch();

        setupArrowButtons();

        // --------------------------------------------------------
        // FADE
        // --------------------------------------------------------

        fadeRunnable = new Runnable() {
            @Override
            public void run() {
                fadeScrollbar();
            }
        };

        showScrollbar();
    }

    // ============================================================
    // ARROW BUTTON CREATION
    // ============================================================

    private ImageView createArrowButton(boolean up) {

        ImageView button =
                new ImageView(getContext());

        GradientDrawable drawable =
                new GradientDrawable();

        drawable.setColor(ARROW_COLOR);
        drawable.setCornerRadius(
                dp(ARROW_RADIUS_DP)
        );

        button.setBackground(drawable);

        button.setElevation(dp(5));

        button.setPadding(
                dp(10),
                dp(10),
                dp(10),
                dp(10)
        );

        // Draw arrow using a simple custom drawable.
        button.setImageDrawable(
                new ArrowDrawable(
                        ARROW_COLOR,
                        up
                )
        );

        button.setScaleType(
                ImageView.ScaleType.CENTER
        );

        button.setClickable(true);
        button.setFocusable(true);

        button.setOnTouchListener(
                new OnTouchListener() {

                    @Override
                    public boolean onTouch(
                            View v,
                            MotionEvent event
                    ) {

                        if (event.getAction() ==
                                MotionEvent.ACTION_DOWN) {

                            GradientDrawable pressed =
                                    new GradientDrawable();

                            pressed.setColor(
                                    ARROW_PRESSED_COLOR
                            );

                            pressed.setCornerRadius(
                                    dp(ARROW_RADIUS_DP)
                            );

                            v.setBackground(pressed);

                            v.animate()
                                    .scaleX(0.92f)
                                    .scaleY(0.92f)
                                    .setDuration(80)
                                    .start();

                        } else if (
                                event.getAction() ==
                                MotionEvent.ACTION_UP
                                ||
                                event.getAction() ==
                                MotionEvent.ACTION_CANCEL
                        ) {

                            GradientDrawable normal =
                                    new GradientDrawable();

                            normal.setColor(
                                    ARROW_COLOR
                            );

                            normal.setCornerRadius(
                                    dp(ARROW_RADIUS_DP)
                            );

                            v.setBackground(normal);

                            v.animate()
                                    .scaleX(1f)
                                    .scaleY(1f)
                                    .setDuration(80)
                                    .start();

                        }

                        return false;
                    }
                }
        );

        return button;
    }

    // ============================================================
    // CONNECT TO SCROLLVIEW
    // ============================================================

    public void attachToScrollView(
            final ScrollView scroll
    ) {

        this.scrollView = scroll;

        if (scrollView == null) {
            return;
        }

        scrollView.setVerticalScrollBarEnabled(false);

        scrollView.getViewTreeObserver()
                .addOnScrollChangedListener(
                        new android.view.ViewTreeObserver
                                .OnScrollChangedListener() {

                            @Override
                            public void onScrollChanged() {

                                if (!draggingThumb) {

                                    updateThumbFromScroll();

                                }

                                showScrollbar();
                            }
                        }
                );

        scrollView.post(
                new Runnable() {
                    @Override
                    public void run() {

                        updateThumbFromScroll();

                    }
                }
        );

        showScrollbar();
    }

    // ============================================================
    // THUMB DRAGGING
    // ============================================================

    private void setupThumbTouch() {

        thumb.setOnTouchListener(
                new OnTouchListener() {

                    @Override
                    public boolean onTouch(
                            View v,
                            MotionEvent event
                    ) {

                        if (scrollView == null) {
                            return false;
                        }

                        switch (event.getActionMasked()) {

                            case MotionEvent.ACTION_DOWN:

                                draggingThumb = true;

                                downY = event.getRawY();

                                startThumbY =
                                        thumb.getY();

                                setThumbPressed(true);

                                showScrollbar();

                                return true;

                            case MotionEvent.ACTION_MOVE:

                                float difference =
                                        event.getRawY()
                                                - downY;

                                float newY =
                                        startThumbY +
                                                difference;

                                moveThumb(newY);

                                return true;

                            case MotionEvent.ACTION_UP:
                            case MotionEvent.ACTION_CANCEL:

                                draggingThumb = false;

                                setThumbPressed(false);

                                updateThumbFromScroll();

                                showScrollbar();

                                return true;
                        }

                        return true;
                    }
                }
        );
    }

    // ============================================================
    // MOVE THUMB
    // ============================================================

    private void moveThumb(float y) {

        int availableHeight =
                barHeight - thumbHeight;

        if (availableHeight <= 0) {
            return;
        }

        if (y < 0) {
            y = 0;
        }

        if (y > availableHeight) {
            y = availableHeight;
        }

        thumb.setY(y);

        int scrollRange =
                getScrollRange();

        if (scrollRange <= 0) {
            return;
        }

        float percentage =
                y / (float) availableHeight;

        int targetScroll =
                (int)
                        (percentage * scrollRange);

        scrollView.scrollTo(
                0,
                targetScroll
        );
    }

    // ============================================================
    // UPDATE THUMB FROM SCROLL POSITION
    // ============================================================

    private void updateThumbFromScroll() {

        if (scrollView == null) {
            return;
        }

        int range =
                getScrollRange();

        if (range <= 0) {

            thumb.setY(0);

            return;
        }

        int scrollY =
                scrollView.getScrollY();

        int availableHeight =
                barHeight - thumbHeight;

        float percentage =
                scrollY / (float) range;

        float y =
                percentage *
                        availableHeight;

        if (y < 0) {
            y = 0;
        }

        if (y > availableHeight) {
            y = availableHeight;
        }

        thumb.setY(y);
    }

    // ============================================================
    // SCROLL RANGE
    // ============================================================

    private int getScrollRange() {

        if (scrollView == null) {
            return 0;
        }

        if (scrollView.getChildCount() == 0) {
            return 0;
        }

        View child =
                scrollView.getChildAt(0);

        int contentHeight =
                child.getHeight();

        int viewportHeight =
                scrollView.getHeight();

        return Math.max(
                0,
                contentHeight -
                        viewportHeight
        );
    }

    // ============================================================
    // TOP ARROW
    // ============================================================

    private void setupArrowButtons() {

        topButton.setOnClickListener(
                new OnClickListener() {

                    @Override
                    public void onClick(View v) {

                        pageUp();

                    }
                }
        );

        bottomButton.setOnClickListener(
                new OnClickListener() {

                    @Override
                    public void onClick(View v) {

                        pageDown();

                    }
                }
        );
    }

    // ============================================================
    // PAGE UP
    // ============================================================

    private void pageUp() {

        if (scrollView == null) {
            return;
        }

        int amount =
                (int)
                        (scrollView.getHeight()
                                * PAGE_FACTOR);

        int target =
                scrollView.getScrollY()
                        - amount;

        if (target < 0) {
            target = 0;
        }

        smoothScrollTo(target);

        showScrollbar();
    }

    // ============================================================
    // PAGE DOWN
    // ============================================================

    private void pageDown() {

        if (scrollView == null) {
            return;
        }

        int amount =
                (int)
                        (scrollView.getHeight()
                                * PAGE_FACTOR);

        int target =
                scrollView.getScrollY()
                        + amount;

        int range =
                getScrollRange();

        if (target > range) {
            target = range;
        }

        smoothScrollTo(target);

        showScrollbar();
    }

    // ============================================================
    // SMOOTH SCROLL
    // ============================================================

    private void smoothScrollTo(
            int target
    ) {

        if (scrollView == null) {
            return;
        }

        final int start =
                scrollView.getScrollY();

        ValueAnimator animator =
                ValueAnimator.ofInt(
                        start,
                        target
                );

        animator.setDuration(
                SCROLL_DURATION
        );

        animator.setInterpolator(
                new DecelerateInterpolator()
        );

        animator.addUpdateListener(
                new ValueAnimator.AnimatorUpdateListener() {

                    @Override
                    public void onAnimationUpdate(
                            ValueAnimator animation
                    ) {

                        int value =
                                (Integer)
                                        animation
                                                .getAnimatedValue();

                        scrollView.scrollTo(
                                0,
                                value
                        );

                        updateThumbFromScroll();
                    }
                }
        );

        animator.start();
    }

    // ============================================================
    // THUMB PRESSED UI
    // ============================================================

    private void setThumbPressed(
            boolean pressed
    ) {

        GradientDrawable drawable =
                new GradientDrawable();

        drawable.setColor(
                pressed
                        ? THUMB_PRESSED_COLOR
                        : THUMB_COLOR
        );

        drawable.setCornerRadius(
                dp(THUMB_RADIUS_DP)
        );

        thumb.setBackground(drawable);

        thumb.animate()
                .scaleX(pressed ? 1.35f : 1f)
                .setDuration(100)
                .start();
    }

    // ============================================================
    // SHOW SCROLLBAR
    // ============================================================

    private void showScrollbar() {

        scrollbarContainer.animate()
                .alpha(1f)
                .setDuration(150)
                .start();

        fadeHandler.removeCallbacks(
                fadeRunnable
        );

        fadeHandler.postDelayed(
                fadeRunnable,
                FADE_DELAY
        );
    }

    // ============================================================
    // FADE SCROLLBAR
    // ============================================================

    private void fadeScrollbar() {

        if (draggingThumb) {
            return;
        }

        scrollbarContainer.animate()
                .alpha(0f)
                .setDuration(
                        FADE_DURATION
                )
                .start();
    }

    // ============================================================
    // FORCE SHOW
    // ============================================================

    public void show() {

        showScrollbar();

    }

    // ============================================================
    // FORCE HIDE
    // ============================================================

    public void hide() {

        fadeHandler.removeCallbacks(
                fadeRunnable
        );

        scrollbarContainer.animate()
                .alpha(0f)
                .setDuration(
                        FADE_DURATION
                )
                .start();
    }

    // ============================================================
    // SET THUMB COLOR
    // ============================================================

    public void setThumbColor(
            int color
    ) {

        GradientDrawable drawable =
                new GradientDrawable();

        drawable.setColor(color);

        drawable.setCornerRadius(
                dp(THUMB_RADIUS_DP)
        );

        thumb.setBackground(drawable);
    }

    // ============================================================
    // SET BAR HEIGHT
    // ============================================================

    public void setBarHeight(
            int dpHeight
    ) {

        barHeight =
                dp(dpHeight);

        LayoutParams params =
                new LayoutParams(
                        barWidth,
                        barHeight
                );

        setLayoutParams(params);

        ViewGroup.LayoutParams
                containerParams =
                scrollbarContainer
                        .getLayoutParams();

        containerParams.height =
                barHeight;

        scrollbarContainer
                .setLayoutParams(
                        containerParams
                );

        ViewGroup.LayoutParams
                trackParams =
                track.getLayoutParams();

        trackParams.height =
                barHeight;

        track.setLayoutParams(
                trackParams
        );

        updateThumbFromScroll();
    }

    // ============================================================
    // SET THUMB HEIGHT
    // ============================================================

    public void setThumbHeight(
            int dpHeight
    ) {

        thumbHeight =
                dp(dpHeight);

        ViewGroup.LayoutParams
                params =
                thumb.getLayoutParams();

        params.height =
                thumbHeight;

        thumb.setLayoutParams(params);

        updateThumbFromScroll();
    }

    // ============================================================
    // SET FADE DELAY
    // ============================================================

    public void setFadeDelay(
            int milliseconds
    ) {

        fadeHandler.removeCallbacks(
                fadeRunnable
        );

        fadeHandler.postDelayed(
                fadeRunnable,
                milliseconds
        );
    }

    // ============================================================
    // DP
    // ============================================================

    private int dp(int value) {

        return (int)
                (value *
                        getResources()
                                .getDisplayMetrics()
                                .density
                        + 0.5f);
    }

    // ============================================================
    // SIMPLE ARROW DRAWABLE
    // ============================================================

    private static class ArrowDrawable
            extends android.graphics.drawable.Drawable {

        private final int color;
        private final boolean up;

        private final android.graphics.Paint paint =
                new android.graphics.Paint(
                        android.graphics.Paint.ANTI_ALIAS_FLAG
                );

        ArrowDrawable(
                int color,
                boolean up
        ) {

            this.color = color;
            this.up = up;

            paint.setColor(color);
            paint.setStyle(
                    android.graphics.Paint.Style.STROKE
            );

            paint.setStrokeWidth(4f);

            paint.setStrokeCap(
                    android.graphics.Paint.Cap.ROUND
            );

            paint.setStrokeJoin(
                    android.graphics.Paint.Join.ROUND
            );
        }

        @Override
        public void draw(
                android.graphics.Canvas canvas
        ) {

            float w = getBounds().width();
            float h = getBounds().height();

            float cx = w / 2f;
            float cy = h / 2f;

            android.graphics.Path path =
                    new android.graphics.Path();

            if (up) {

                path.moveTo(
                        cx - 9,
                        cy + 5
                );

                path.lineTo(
                        cx,
                        cy - 5
                );

                path.lineTo(
                        cx + 9,
                        cy + 5
                );

            } else {

                path.moveTo(
                        cx - 9,
                        cy - 5
                );

                path.lineTo(
                        cx,
                        cy + 5
                );

                path.lineTo(
                        cx + 9,
                        cy - 5
                );
            }

            canvas.drawPath(
                    path,
                    paint
            );
        }

        @Override
        public void setAlpha(int alpha) {
            paint.setAlpha(alpha);
        }

        @Override
        public void setColorFilter(
                android.graphics.ColorFilter filter
        ) {
            paint.setColorFilter(filter);
        }

        @Override
        public int getOpacity() {
            return android.graphics.PixelFormat.TRANSLUCENT;
        }
    }
}
