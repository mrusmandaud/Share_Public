package com.myrecords.onenewexperience.one;

import android.animation.ValueAnimator;
import android.app.Activity;
import android.content.Context;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;
import android.graphics.Path;
import android.graphics.PixelFormat;
import android.graphics.drawable.Drawable;
import android.graphics.drawable.GradientDrawable;
import android.os.Handler;
import android.os.Looper;
import android.view.Gravity;
import android.view.MotionEvent;
import android.view.View;
import android.view.ViewGroup;
import android.view.ViewTreeObserver;
import android.view.animation.DecelerateInterpolator;
import android.widget.FrameLayout;
import android.widget.ImageView;
import android.widget.ScrollView;

public class CustomScrollbar extends FrameLayout {

    // =========================
    // SIZE
    // =========================
    private static final int BAR_HEIGHT_DP = 600;
    private static final int THUMB_HEIGHT_DP = 200;
    private static final int BAR_WIDTH_DP = 48;
    private static final int TRACK_WIDTH_DP = 8;
    private static final int THUMB_WIDTH_DP = 12;

    // =========================
    // ARROWS
    // =========================
    private static final int ARROW_SIZE_DP = 40;
    private static final int ARROW_MARGIN_DP = 2;

    // =========================
    // COLORS
    // =========================
    private static final int TRACK_COLOR =
            Color.rgb(226, 230, 236);

    private static final int THUMB_COLOR =
            Color.rgb(66, 103, 210);

    private static final int THUMB_PRESSED_COLOR =
            Color.rgb(45, 78, 175);

    private static final int ARROW_COLOR =
            Color.rgb(66, 103, 210);

    private static final int ARROW_PRESSED_COLOR =
            Color.rgb(45, 78, 175);

    private static final int ARROW_ICON_COLOR =
            Color.WHITE;

    // =========================
    // RADIUS
    // =========================
    private static final int TRACK_RADIUS_DP = 6;
    private static final int THUMB_RADIUS_DP = 7;
    private static final int ARROW_RADIUS_DP = 20;

    // =========================
    // SCROLL
    // =========================
    private static final float PAGE_FACTOR = 0.85f;
    private static final int SCROLL_DURATION = 220;

    // =========================
    // FADE
    // =========================
    private static final int FADE_DELAY = 1500;
    private static final int FADE_IN_DURATION = 150;
    private static final int FADE_OUT_DURATION = 350;

    private ScrollView scrollView;
    private FrameLayout scrollbarContainer;

    private View track;
    private View thumb;

    private ImageView topButton;
    private ImageView bottomButton;

    private int barHeight;
    private int thumbHeight;
    private int barWidth;

    private float downY;
    private float startThumbY;

    private boolean draggingThumb;

    private ValueAnimator scrollAnimator;

    private final Handler fadeHandler =
            new Handler(Looper.getMainLooper());

    private final Runnable fadeRunnable =
            new Runnable() {
                @Override
                public void run() {
                    fadeScrollbar();
                }
            };

    private ViewTreeObserver.OnScrollChangedListener
            scrollChangedListener;

    // =========================
    // CONSTRUCTOR
    // =========================
    public CustomScrollbar(Context context) {
        super(context);

        setClipChildren(false);
        setClipToPadding(false);

        barHeight = dp(BAR_HEIGHT_DP);
        thumbHeight = dp(THUMB_HEIGHT_DP);
        barWidth = dp(BAR_WIDTH_DP);

        buildUI();
    }

    // =========================
    // BUILD UI
    // =========================
    private void buildUI() {

        FrameLayout.LayoutParams rootParams =
                new FrameLayout.LayoutParams(
                        barWidth,
                        barHeight
                );

        rootParams.gravity = Gravity.RIGHT |
                Gravity.BOTTOM;

        setLayoutParams(rootParams);

        scrollbarContainer =
                new FrameLayout(getContext());

        FrameLayout.LayoutParams containerParams =
                new FrameLayout.LayoutParams(
                        barWidth,
                        barHeight
                );

        scrollbarContainer.setClipChildren(false);
        scrollbarContainer.setClipToPadding(false);

        addView(
                scrollbarContainer,
                containerParams
        );

        // TRACK
        track = new View(getContext());

        track.setBackground(
                rounded(
                        TRACK_COLOR,
                        TRACK_RADIUS_DP
                )
        );

        FrameLayout.LayoutParams trackParams =
                new FrameLayout.LayoutParams(
                        dp(TRACK_WIDTH_DP),
                        barHeight
                );

        trackParams.gravity =
                Gravity.CENTER_HORIZONTAL;

        scrollbarContainer.addView(
                track,
                trackParams
        );

        // THUMB
        thumb = new View(getContext());

        thumb.setBackground(
                rounded(
                        THUMB_COLOR,
                        THUMB_RADIUS_DP
                )
        );

        thumb.setElevation(dp(5));

        FrameLayout.LayoutParams thumbParams =
                new FrameLayout.LayoutParams(
                        dp(THUMB_WIDTH_DP),
                        thumbHeight
                );

        thumbParams.gravity =
                Gravity.TOP |
                Gravity.CENTER_HORIZONTAL;

        scrollbarContainer.addView(
                thumb,
                thumbParams
        );

        // TOP ARROW
        topButton = createArrowButton(true);

        FrameLayout.LayoutParams topParams =
                new FrameLayout.LayoutParams(
                        dp(ARROW_SIZE_DP),
                        dp(ARROW_SIZE_DP)
                );

        topParams.gravity =
                Gravity.TOP |
                Gravity.CENTER_HORIZONTAL;

        topParams.topMargin =
                -dp(ARROW_MARGIN_DP);

        scrollbarContainer.addView(
                topButton,
                topParams
        );

        // BOTTOM ARROW
        bottomButton = createArrowButton(false);

        FrameLayout.LayoutParams bottomParams =
                new FrameLayout.LayoutParams(
                        dp(ARROW_SIZE_DP),
                        dp(ARROW_SIZE_DP)
                );

        bottomParams.gravity =
                Gravity.BOTTOM |
                Gravity.CENTER_HORIZONTAL;

        bottomParams.bottomMargin =
                -dp(ARROW_MARGIN_DP);

        scrollbarContainer.addView(
                bottomButton,
                bottomParams
        );

        setupThumbTouch();
        setupArrowButtons();

        showScrollbar();
    }

    // =========================
    // ONE-LINE CONNECTION
    // =========================
    public void attachTo(
            final ScrollView scroll
    ) {

        attachToScrollView(scroll);

        if (getContext() instanceof Activity) {

            Activity activity =
                    (Activity) getContext();

            FrameLayout.LayoutParams params =
                    new FrameLayout.LayoutParams(
                            dp(BAR_WIDTH_DP),
                            dp(BAR_HEIGHT_DP)
                    );

            params.gravity =
                    Gravity.RIGHT |
                    Gravity.BOTTOM;

            params.rightMargin = dp(2);
            params.bottomMargin = dp(8);

            activity.addContentView(
                    this,
                    params
            );
        }
    }

    // =========================
    // SCROLLVIEW CONNECTION
    // =========================
    public void attachToScrollView(
            final ScrollView scroll
    ) {

        removePreviousScrollListener();

        scrollView = scroll;

        if (scrollView == null) {
            return;
        }

        scrollView.setVerticalScrollBarEnabled(false);
        scrollView.setScrollbarFadingEnabled(false);

        scrollChangedListener =
                new ViewTreeObserver
                        .OnScrollChangedListener() {

                    @Override
                    public void onScrollChanged() {

                        if (!draggingThumb) {
                            updateThumbFromScroll();
                        }

                        showScrollbar();
                    }
                };

        scrollView.getViewTreeObserver()
                .addOnScrollChangedListener(
                        scrollChangedListener
                );

        scrollView.post(
                new Runnable() {
                    @Override
                    public void run() {
                        updateThumbFromScroll();
                    }
                }
        );

        showScrollbar();
    }

    // =========================
    // THUMB TOUCH
    // =========================
    private void setupThumbTouch() {

        thumb.setOnTouchListener(
                new OnTouchListener() {

                    @Override
                    public boolean onTouch(
                            View view,
                            MotionEvent event
                    ) {

                        if (scrollView == null) {
                            return false;
                        }

                        switch (
                                event.getActionMasked()
                        ) {

                            case MotionEvent.ACTION_DOWN:

                                stopScrollAnimation();

                                draggingThumb = true;

                                downY =
                                        event.getRawY();

                                startThumbY =
                                        thumb.getY();

                                setThumbPressed(true);
                                showScrollbar();

                                return true;

                            case MotionEvent.ACTION_MOVE:

                                float difference =
                                        event.getRawY()
                                                - downY;

                                moveThumb(
                                        startThumbY +
                                        difference
                                );

                                return true;

                            case MotionEvent.ACTION_UP:

                                draggingThumb = false;

                                setThumbPressed(false);

                                updateThumbFromScroll();
                                showScrollbar();

                                return true;

                            case MotionEvent.ACTION_CANCEL:

                                draggingThumb = false;

                                setThumbPressed(false);

                                updateThumbFromScroll();
                                showScrollbar();

                                return true;
                        }

                        return true;
                    }
                }
        );
    }

    // =========================
    // MOVE THUMB
    // =========================
    private void moveThumb(float y) {

        int available =
                getAvailableThumbTravel();

        if (available <= 0) {
            return;
        }

        y = Math.max(
                0,
                Math.min(y, available)
        );

        thumb.setY(y);

        int range = getScrollRange();

        if (range <= 0) {
            return;
        }

        float percent =
                y / (float) available;

        scrollView.scrollTo(
                0,
                Math.round(percent * range)
        );
    }

    // =========================
    // UPDATE THUMB
    // =========================
    private void updateThumbFromScroll() {

        if (scrollView == null) {
            return;
        }

        int range = getScrollRange();

        if (range <= 0) {
            thumb.setY(0);
            return;
        }

        int available =
                getAvailableThumbTravel();

        if (available <= 0) {
            return;
        }

        float percent =
                scrollView.getScrollY() /
                (float) range;

        percent = Math.max(
                0f,
                Math.min(percent, 1f)
        );

        thumb.setY(
                percent * available
        );
    }

    private int getAvailableThumbTravel() {

        return Math.max(
                0,
                barHeight - thumbHeight
        );
    }

    // =========================
    // SCROLL RANGE
    // =========================
    private int getScrollRange() {

        if (scrollView == null ||
                scrollView.getChildCount() == 0) {

            return 0;
        }

        View child =
                scrollView.getChildAt(0);

        int contentHeight =
                child.getHeight();

        int viewportHeight =
                scrollView.getHeight()
                - scrollView.getPaddingTop()
                - scrollView.getPaddingBottom();

        return Math.max(
                0,
                contentHeight - viewportHeight
        );
    }

    // =========================
    // ARROWS
    // =========================
    private void setupArrowButtons() {

        topButton.setOnClickListener(
                new OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        pageUp();
                    }
                }
        );

        bottomButton.setOnClickListener(
                new OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        pageDown();
                    }
                }
        );
    }

    private void pageUp() {

        if (scrollView == null) {
            return;
        }

        int amount =
                Math.max(
                        1,
                        Math.round(
                                scrollView.getHeight()
                                * PAGE_FACTOR
                        )
                );

        int target =
                Math.max(
                        0,
                        scrollView.getScrollY()
                        - amount
                );

        smoothScrollTo(target);
        showScrollbar();
    }

    private void pageDown() {

        if (scrollView == null) {
            return;
        }

        int amount =
                Math.max(
                        1,
                        Math.round(
                                scrollView.getHeight()
                                * PAGE_FACTOR
                        )
                );

        int target =
                Math.min(
                        getScrollRange(),
                        scrollView.getScrollY()
                        + amount
                );

        smoothScrollTo(target);
        showScrollbar();
    }

    // =========================
    // SMOOTH SCROLL
    // =========================
    private void smoothScrollTo(
            final int target
    ) {

        if (scrollView == null) {
            return;
        }

        stopScrollAnimation();

        int start =
                scrollView.getScrollY();

        if (start == target) {
            updateThumbFromScroll();
            return;
        }

        scrollAnimator =
                ValueAnimator.ofInt(
                        start,
                        target
                );

        scrollAnimator.setDuration(
                SCROLL_DURATION
        );

        scrollAnimator.setInterpolator(
                new DecelerateInterpolator()
        );

        scrollAnimator.addUpdateListener(
                new ValueAnimator.AnimatorUpdateListener() {

                    @Override
                    public void onAnimationUpdate(
                            ValueAnimator animation
                    ) {

                        if (scrollView == null) {
                            return;
                        }

                        int value =
                                (Integer)
                                animation.getAnimatedValue();

                        scrollView.scrollTo(
                                0,
                                value
                        );

                        updateThumbFromScroll();
                    }
                }
        );

        scrollAnimator.start();
    }

    private void stopScrollAnimation() {

        if (scrollAnimator != null) {

            if (scrollAnimator.isRunning()) {
                scrollAnimator.cancel();
            }

            scrollAnimator = null;
        }
    }

    // =========================
    // PRESSED
    // =========================
    private void setThumbPressed(
            boolean pressed
    ) {

        thumb.setBackground(
                rounded(
                        pressed
                                ? THUMB_PRESSED_COLOR
                                : THUMB_COLOR,
                        THUMB_RADIUS_DP
                )
        );

        thumb.animate()
                .scaleX(
                        pressed ? 1.30f : 1f
                )
                .setDuration(100)
                .start();
    }

    // =========================
    // FADE
    // =========================
    private void showScrollbar() {

        if (scrollbarContainer == null) {
            return;
        }

        scrollbarContainer
                .animate()
                .alpha(1f)
                .setDuration(
                        FADE_IN_DURATION
                )
                .start();

        fadeHandler.removeCallbacks(
                fadeRunnable
        );

        fadeHandler.postDelayed(
                fadeRunnable,
                FADE_DELAY
        );
    }

    private void fadeScrollbar() {

        if (draggingThumb ||
                scrollbarContainer == null) {

            return;
        }

        scrollbarContainer
                .animate()
                .alpha(0f)
                .setDuration(
                        FADE_OUT_DURATION
                )
                .start();
    }

    // =========================
    // ARROW BUTTON
    // =========================
    private ImageView createArrowButton(
            final boolean up
    ) {

        ImageView button =
                new ImageView(getContext());

        button.setBackground(
                rounded(
                        ARROW_COLOR,
                        ARROW_RADIUS_DP
                )
        );

        button.setImageDrawable(
                new ArrowDrawable(
                        ARROW_ICON_COLOR,
                        up
                )
        );

        button.setScaleType(
                ImageView.ScaleType.CENTER
        );

        button.setClickable(true);
        button.setFocusable(true);
        button.setElevation(dp(6));

        button.setOnTouchListener(
                new OnTouchListener() {

                    @Override
                    public boolean onTouch(
                            View v,
                            MotionEvent event
                    ) {

                        switch (
                                event.getActionMasked()
                        ) {

                            case MotionEvent.ACTION_DOWN:

                                v.setBackground(
                                        rounded(
                                                ARROW_PRESSED_COLOR,
                                                ARROW_RADIUS_DP
                                        )
                                );

                                v.animate()
                                        .scaleX(.92f)
                                        .scaleY(.92f)
                                        .setDuration(80)
                                        .start();

                                showScrollbar();

                                break;

                            case MotionEvent.ACTION_UP:
                            case MotionEvent.ACTION_CANCEL:

                                v.setBackground(
                                        rounded(
                                                ARROW_COLOR,
                                                ARROW_RADIUS_DP
                                        )
                                );

                                v.animate()
                                        .scaleX(1f)
                                        .scaleY(1f)
                                        .setDuration(80)
                                        .start();

                                break;
                        }

                        return false;
                    }
                }
        );

        return button;
    }

    // =========================
    // DRAWABLE
    // =========================
    private GradientDrawable rounded(
            int color,
            int radiusDp
    ) {

        GradientDrawable d =
                new GradientDrawable();

        d.setColor(color);

        d.setCornerRadius(
                dp(radiusDp)
        );

        return d;
    }

    // =========================
    // PUBLIC API
    // =========================
    public void show() {
        showScrollbar();
    }

    public void hide() {

        fadeHandler.removeCallbacks(
                fadeRunnable
        );

        if (scrollbarContainer != null) {

            scrollbarContainer
                    .animate()
                    .alpha(0f)
                    .setDuration(
                            FADE_OUT_DURATION
                    )
                    .start();
        }
    }

    public void setThumbColor(int color) {

        if (thumb != null) {

            thumb.setBackground(
                    rounded(
                            color,
                            THUMB_RADIUS_DP
                    )
            );
        }
    }

    public void setTrackColor(int color) {

        if (track != null) {

            track.setBackground(
                    rounded(
                            color,
                            TRACK_RADIUS_DP
                    )
            );
        }
    }

    public void setBarHeight(
            int heightDp
    ) {

        barHeight =
                Math.max(
                        1,
                        dp(heightDp)
                );

        updateLayout();
    }

    public void setThumbHeight(
            int heightDp
    ) {

        thumbHeight =
                Math.max(
                        1,
                        dp(heightDp)
                );

        if (thumb != null) {

            ViewGroup.LayoutParams p =
                    thumb.getLayoutParams();

            p.height = thumbHeight;

            thumb.setLayoutParams(p);
        }

        updateThumbFromScroll();
    }

    public void setBarWidth(
            int widthDp
    ) {

        barWidth =
                Math.max(
                        1,
                        dp(widthDp)
                );

        updateLayout();
    }

    private void updateLayout() {

        ViewGroup.LayoutParams root =
                getLayoutParams();

        if (root != null) {

            root.width = barWidth;
            root.height = barHeight;

            setLayoutParams(root);
        }

        if (scrollbarContainer != null) {

            ViewGroup.LayoutParams p =
                    scrollbarContainer
                            .getLayoutParams();

            p.width = barWidth;
            p.height = barHeight;

            scrollbarContainer
                    .setLayoutParams(p);
        }

        if (track != null) {

            ViewGroup.LayoutParams p =
                    track.getLayoutParams();

            p.height = barHeight;

            track.setLayoutParams(p);
        }

        updateThumbFromScroll();
    }

    public void setFadeDelay(
            int milliseconds
    ) {

        fadeHandler.removeCallbacks(
                fadeRunnable
        );

        fadeHandler.postDelayed(
                fadeRunnable,
                Math.max(
                        0,
                        milliseconds
                )
        );
    }

    public void setFadeDuration(
            int milliseconds
    ) {
        // Compatibility method.
    }

    // =========================
    // CLEANUP
    // =========================
    public void detach() {

        fadeHandler.removeCallbacks(
                fadeRunnable
        );

        stopScrollAnimation();

        removePreviousScrollListener();

        scrollChangedListener = null;
        scrollView = null;
    }

    private void removePreviousScrollListener() {

        if (scrollView == null ||
                scrollChangedListener == null) {

            return;
        }

        if (
                scrollView
                        .getViewTreeObserver()
                        .isAlive()
        ) {

            scrollView
                    .getViewTreeObserver()
                    .removeOnScrollChangedListener(
                            scrollChangedListener
                    );
        }
    }

    // =========================
    // DP
    // =========================
    private int dp(int value) {

        return (int)(
                value *
                getResources()
                        .getDisplayMetrics()
                        .density
                + 0.5f
        );
    }

    // =========================
    // ARROW DRAWABLE
    // =========================
    private static class ArrowDrawable
            extends Drawable {

        private final Paint paint;
        private final boolean up;

        ArrowDrawable(
                int color,
                boolean up
        ) {

            this.up = up;

            paint =
                    new Paint(
                            Paint.ANTI_ALIAS_FLAG
                    );

            paint.setColor(color);

            paint.setStyle(
                    Paint.Style.STROKE
            );

            paint.setStrokeWidth(3.2f);

            paint.setStrokeCap(
                    Paint.Cap.ROUND
            );

            paint.setStrokeJoin(
                    Paint.Join.ROUND
            );
        }

        @Override
        public void draw(Canvas canvas) {

            float width =
                    getBounds().width();

            float height =
                    getBounds().height();

            float x = width / 2f;
            float y = height / 2f;

            Path path = new Path();

            if (up) {

                path.moveTo(
                        x - 8,
                        y + 4
                );

                path.lineTo(
                        x,
                        y - 5
                );

                path.lineTo(
                        x + 8,
                        y + 4
                );

            } else {

                path.moveTo(
                        x - 8,
                        y - 4
                );

                path.lineTo(
                        x,
                        y + 5
                );

                path.lineTo(
                        x + 8,
                        y - 4
                );
            }

            canvas.drawPath(
                    path,
                    paint
            );
        }

        @Override
        public void setAlpha(int alpha) {
            paint.setAlpha(alpha);
        }

        @Override
        public void setColorFilter(
                android.graphics.ColorFilter filter
        ) {
            paint.setColorFilter(filter);
        }

        @Override
        public int getOpacity() {
            return PixelFormat.TRANSLUCENT;
        }

        @Override
        public int getIntrinsicWidth() {
            return 40;
        }

        @Override
        public int getIntrinsicHeight() {
            return 40;
        }
    }
}
