	@Override
	protected void onActivityResult(int _requestCode, int _resultCode, Intent _data) {
		super.onActivityResult(_requestCode, _resultCode, _data);
		// ========================================================
		// BACKUP / RESTORE
		// 7001 = Backup
		// 7002 = Restore
		// ========================================================
		
		if (
		_requestCode == 7001
		|| _requestCode == 7002
		) {
			
			android.content.SharedPreferences state =
			SettingsActivity.this.getSharedPreferences(
			"settings_state",
			MODE_PRIVATE
			);
			
			String uriString = "";
			
			if (
			_resultCode == RESULT_OK
			&& _data != null
			&& _data.getData() != null
			) {
				
				uriString =
				_data.getData().toString();
			}
			
			state.edit()
			.putInt(
			"backup_result_code",
			_requestCode
			)
			.putString(
			"backup_result_uri",
			uriString
			)
			.putBoolean(
			"backup_result_ready",
			true
			)
			.apply();
			
			BACKUP();
			
			return;
		}
		
		
		// ========================================================
		// EXPORT
		// 1001 = Export
		// ========================================================
		
		if (_requestCode == 1001) {
			
			android.content.SharedPreferences state =
			SettingsActivity.this.getSharedPreferences(
			"settings_state",
			MODE_PRIVATE
			);
			
			String uriString = "";
			
			if (
			_resultCode == RESULT_OK
			&& _data != null
			&& _data.getData() != null
			) {
				
				uriString =
				_data.getData().toString();
			}
			
			state.edit()
			.putString(
			"export_result_uri",
			uriString
			)
			.putBoolean(
			"export_result_ready",
			true
			)
			.apply();
			
			EXPORT();
			
			return;
		}
		switch (_requestCode) {
			
			default:
			break;
		}
	}
	
	public void _BACKUP() {
		// ========================================================
		// BACKUP / RESTORE
		// ========================================================
		
		android.content.SharedPreferences state =
		SettingsActivity.this.getSharedPreferences(
		"settings_state",
		MODE_PRIVATE
		);
		
		
		// ========================================================
		// WAIT FOR FILE PICKER RESULT
		// ========================================================
		
		if (!state.getBoolean(
		"backup_result_ready",
		false
		)) {
			
			// ----------------------------------------------------
			// No result yet.
			// Start the appropriate Android file picker.
			// ----------------------------------------------------
			
			String action =
			state.getString(
			"backup_action",
			""
			);
			
			
			if ("backup".equals(action)) {
				
				Intent intent =
				new Intent(
				Intent.ACTION_CREATE_DOCUMENT
				);
				
				intent.addCategory(
				Intent.CATEGORY_OPENABLE
				);
				
				intent.setType(
				"text/plain"
				);
				
				intent.putExtra(
				Intent.EXTRA_TITLE,
				"saverecords_backup.txt"
				);
				
				startActivityForResult(
				intent,
				7001
				);
				
				return;
			}
			
			
			if ("restore".equals(action)) {
				
				Intent intent =
				new Intent(
				Intent.ACTION_OPEN_DOCUMENT
				);
				
				intent.addCategory(
				Intent.CATEGORY_OPENABLE
				);
				
				intent.setType(
				"text/plain"
				);
				
				startActivityForResult(
				intent,
				7002
				);
				
				return;
			}
			
			return;
		}
		
		
		// ========================================================
		// RESULT IS READY
		// ========================================================
		
		int resultCode =
		state.getInt(
		"backup_result_code",
		0
		);
		
		String uriString =
		state.getString(
		"backup_result_uri",
		""
		);
		
		
		// ========================================================
		// USER CANCELLED
		// ========================================================
		
		if (
		uriString == null
		|| uriString.length() == 0
		) {
			
			state.edit()
			.putBoolean(
			"backup_result_ready",
			false
			)
			.putInt(
			"backup_result_code",
			0
			)
			.putString(
			"backup_result_uri",
			""
			)
			.apply();
			
			return;
		}
		
		
		android.net.Uri resultUri =
		android.net.Uri.parse(
		uriString
		);
		
		
		// ========================================================
		// BACKUP
		// ========================================================
		
		if (resultCode == 7001) {
			
			try {
				
				java.io.File sourceFile =
				new java.io.File(
				getFilesDir(),
				"saverecords.txt"
				);
				
				
				if (!sourceFile.exists()) {
					
					Toast.makeText(
					SettingsActivity.this,
					"No saved records found",
					Toast.LENGTH_SHORT
					).show();
					
				} else {
					
					java.io.InputStream input =
					new java.io.FileInputStream(
					sourceFile
					);
					
					
					java.io.OutputStream output =
					getContentResolver()
					.openOutputStream(
					resultUri
					);
					
					
					if (
					input != null
					&& output != null
					) {
						
						byte[] buffer =
						new byte[8192];
						
						int length;
						
						while (
						(length =
						input.read(buffer)) != -1
						) {
							
							output.write(
							buffer,
							0,
							length
							);
						}
						
						
						output.flush();
						output.close();
						input.close();
						
						
						Toast.makeText(
						SettingsActivity.this,
						"Backup completed",
						Toast.LENGTH_SHORT
						).show();
					}
				}
				
			} catch (Exception e) {
				
				Toast.makeText(
				SettingsActivity.this,
				"Backup failed: "
				+ e.getMessage(),
				Toast.LENGTH_LONG
				).show();
			}
		}
		
		
		// ========================================================
		// RESTORE
		// ========================================================
		
		if (resultCode == 7002) {
			
			try {
				
				java.io.InputStream input =
				getContentResolver()
				.openInputStream(
				resultUri
				);
				
				
				if (input == null) {
					
					throw new Exception(
					"Unable to open backup file"
					);
				}
				
				
				java.io.File destinationFile =
				new java.io.File(
				getFilesDir(),
				"saverecords.txt"
				);
				
				
				java.io.OutputStream output =
				new java.io.FileOutputStream(
				destinationFile,
				false
				);
				
				
				byte[] buffer =
				new byte[8192];
				
				int length;
				
				while (
				(length =
				input.read(buffer)) != -1
				) {
					
					output.write(
					buffer,
					0,
					length
					);
				}
				
				
				output.flush();
				output.close();
				input.close();
				
				
				Toast.makeText(
				SettingsActivity.this,
				"Restore completed",
				Toast.LENGTH_SHORT
				).show();
				
			} catch (Exception e) {
				
				Toast.makeText(
				SettingsActivity.this,
				"Restore failed: "
				+ e.getMessage(),
				Toast.LENGTH_LONG
				).show();
			}
		}
		
		
		// ========================================================
		// CLEAR RESULT STATE
		// ========================================================
		
		state.edit()
		.putBoolean(
		"backup_result_ready",
		false
		)
		.putInt(
		"backup_result_code",
		0
		)
		.putString(
		"backup_result_uri",
		""
		)
		.putString(
		"backup_action",
		""
		)
		.apply();
	}
	
	
	public void _EXPORT() {
		// =====================================================
		// EXPORT MORE BLOCK
		//
		// 1001 = EXPORT
		//
		// Supports:
		// 1. All Records
		// 2. Selected Data
		// 3. TXT
		// 4. CSV
		// 5. PDF
		// 6. XLSX
		// 7. Android Save Location Picker
		// 8. SaveRecords_YYYYMMDD_SERIAL.extension
		// 9. Serial increases ONLY after successful save
		// =====================================================
		
		
		// =====================================================
		// 1. PROCESS EXPORT PICKER RESULT
		// =====================================================
		
		if (exportResultReady) {
			
			exportResultReady = false;
			
			
			// =================================================
			// USER CANCELLED EXPORT
			// =================================================
			
			if (exportResultUri == null) {
				
				Toast.makeText(
				SettingsActivity.this,
				"Export cancelled",
				Toast.LENGTH_SHORT
				).show();
				
				pendingExportData = "";
				
				return;
			}
			
			
			// =================================================
			// CHECK DATA
			// =================================================
			
			if (
			pendingExportData == null
			|| pendingExportData.trim().length() == 0
			) {
				
				Toast.makeText(
				SettingsActivity.this,
				"No export data available",
				Toast.LENGTH_SHORT
				).show();
				
				exportResultUri = null;
				pendingExportData = "";
				
				return;
			}
			
			
			// =================================================
			// EXPORT SETTINGS
			// =================================================
			
			android.content.SharedPreferences exportPrefs =
			SettingsActivity.this.getSharedPreferences(
			"export_settings",
			MODE_PRIVATE
			);
			
			
			String format =
			exportPrefs.getString(
			"pending_export_format",
			"TXT"
			);
			
			
			int pendingSerial =
			exportPrefs.getInt(
			"pending_export_serial",
			1
			);
			
			
			boolean exportSuccess = false;
			
			
			// =================================================
			// TXT
			// =================================================
			
			if ("TXT".equals(format)) {
				
				java.io.OutputStream output = null;
				
				try {
					
					output =
					SettingsActivity.this
					.getContentResolver()
					.openOutputStream(
					exportResultUri
					);
					
					if (output == null) {
						
						throw new java.io.IOException(
						"Output stream unavailable"
						);
					}
					
					
					output.write(
					pendingExportData.getBytes("UTF-8")
					);
					
					output.flush();
					
					exportSuccess = true;
					
					
				} catch (Exception e) {
					
					exportSuccess = false;
					
					
				} finally {
					
					try {
						
						if (output != null) {
							output.close();
						}
						
					} catch (Exception ignored) {
					}
				}
			}
			
			
			// =================================================
			// CSV
			// =================================================
			
			else if ("CSV".equals(format)) {
				
				java.io.OutputStream output = null;
				
				try {
					
					StringBuilder csv =
					new StringBuilder();
					
					
					String[] lines =
					pendingExportData.split(
					"\\r?\\n"
					);
					
					
					for (String line : lines) {
						
						if (
						line == null
						|| line.trim().length() == 0
						) {
							continue;
						}
						
						
						String[] parts =
						line.split(
						"\\s*\\|\\s*",
						-1
						);
						
						
						for (
						int i = 0;
						i < parts.length;
						i++
						) {
							
							if (i > 0) {
								csv.append(",");
							}
							
							
							String value =
							parts[i]
							.replace(
							"\"",
							"\"\""
							);
							
							
							csv.append("\"");
							csv.append(value);
							csv.append("\"");
						}
						
						
						csv.append("\n");
					}
					
					
					output =
					SettingsActivity.this
					.getContentResolver()
					.openOutputStream(
					exportResultUri
					);
					
					
					if (output == null) {
						
						throw new java.io.IOException(
						"Output stream unavailable"
						);
					}
					
					
					output.write(
					csv.toString().getBytes("UTF-8")
					);
					
					output.flush();
					
					exportSuccess = true;
					
					
				} catch (Exception e) {
					
					exportSuccess = false;
					
					
				} finally {
					
					try {
						
						if (output != null) {
							output.close();
						}
						
					} catch (Exception ignored) {
					}
				}
			}
			
			
			// =================================================
			// PDF
			// =================================================
			
			else if ("PDF".equals(format)) {
				
				android.graphics.pdf.PdfDocument document =
				new android.graphics.pdf.PdfDocument();
				
				java.io.OutputStream output = null;
				
				android.graphics.pdf.PdfDocument.Page page =
				null;
				
				try {
					
					int pageNumber = 1;
					
					int y = 40;
					
					
					android.graphics.Paint paint =
					new android.graphics.Paint();
					
					paint.setTextSize(11);
					
					
					page =
					document.startPage(
					new android.graphics.pdf.PdfDocument.PageInfo.Builder(
					595,
					842,
					pageNumber
					).create()
					);
					
					
					android.graphics.Canvas canvas =
					page.getCanvas();
					
					
					String[] lines =
					pendingExportData.split(
					"\\r?\\n"
					);
					
					
					for (String line : lines) {
						
						if (
						line == null
						|| line.length() == 0
						) {
							
							y += 18;
							
							continue;
						}
						
						
						String remaining =
						line;
						
						
						while (
						remaining.length() > 0
						) {
							
							if (y > 800) {
								
								document.finishPage(page);
								
								
								pageNumber++;
								
								
								page =
								document.startPage(
								new android.graphics.pdf.PdfDocument.PageInfo.Builder(
								595,
								842,
								pageNumber
								).create()
								);
								
								
								canvas =
								page.getCanvas();
								
								
								y = 40;
							}
							
							
							int maxCharacters = 85;
							
							String part;
							
							
							if (
							remaining.length()
							> maxCharacters
							) {
								
								part =
								remaining.substring(
								0,
								maxCharacters
								);
								
								remaining =
								remaining.substring(
								maxCharacters
								);
								
							} else {
								
								part =
								remaining;
								
								remaining =
								"";
							}
							
							
							canvas.drawText(
							part,
							30,
							y,
							paint
							);
							
							
							y += 18;
						}
					}
					
					
					if (page != null) {
						
						document.finishPage(page);
						
						page = null;
					}
					
					
					output =
					SettingsActivity.this
					.getContentResolver()
					.openOutputStream(
					exportResultUri
					);
					
					
					if (output == null) {
						
						throw new java.io.IOException(
						"PDF output unavailable"
						);
					}
					
					
					document.writeTo(output);
					
					output.flush();
					
					exportSuccess = true;
					
					
				} catch (Exception e) {
					
					exportSuccess = false;
					
					
				} finally {
					
					try {
						
						if (page != null) {
							document.finishPage(page);
						}
						
					} catch (Exception ignored) {
					}
					
					
					try {
						
						if (output != null) {
							output.close();
						}
						
					} catch (Exception ignored) {
					}
					
					
					document.close();
				}
			}
			
			
			// =================================================
			// XLSX
			//
			// Real XLSX ZIP/XML workbook
			// =================================================
			
			else if ("XLSX".equals(format)) {
				
				java.io.OutputStream output = null;
				java.util.zip.ZipOutputStream zip = null;
				
				try {
					
					output =
					SettingsActivity.this
					.getContentResolver()
					.openOutputStream(
					exportResultUri
					);
					
					
					if (output == null) {
						
						throw new java.io.IOException(
						"XLSX output unavailable"
						);
					}
					
					
					zip =
					new java.util.zip.ZipOutputStream(
					output
					);
					
					
					// =================================================
					// [Content_Types].xml
					// =================================================
					
					java.util.zip.ZipEntry entry1 =
					new java.util.zip.ZipEntry(
					"[Content_Types].xml"
					);
					
					zip.putNextEntry(entry1);
					
					
					String contentTypes =
					"<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
					+ "<Types xmlns=\"http://schemas.openxmlformats.org/package/2006/content-types\">"
					+ "<Default Extension=\"rels\" ContentType=\"application/vnd.openxmlformats-package.relationships+xml\"/>"
					+ "<Default Extension=\"xml\" ContentType=\"application/xml\"/>"
					+ "<Override PartName=\"/xl/workbook.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.spreadsheetml.sheet.main+xml\"/>"
					+ "<Override PartName=\"/xl/worksheets/sheet1.xml\" ContentType=\"application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml\"/>"
					+ "</Types>";
					
					
					zip.write(
					contentTypes.getBytes("UTF-8")
					);
					
					zip.closeEntry();
					
					
					// =================================================
					// _rels/.rels
					// =================================================
					
					java.util.zip.ZipEntry entry2 =
					new java.util.zip.ZipEntry(
					"_rels/.rels"
					);
					
					zip.putNextEntry(entry2);
					
					
					String rels =
					"<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
					+ "<Relationships xmlns=\"http://schemas.openxmlformats.org/package/2006/relationships\">"
					+ "<Relationship Id=\"rId1\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument\" Target=\"xl/workbook.xml\"/>"
					+ "</Relationships>";
					
					
					zip.write(
					rels.getBytes("UTF-8")
					);
					
					zip.closeEntry();
					
					
					// =================================================
					// xl/workbook.xml
					// =================================================
					
					java.util.zip.ZipEntry entry3 =
					new java.util.zip.ZipEntry(
					"xl/workbook.xml"
					);
					
					zip.putNextEntry(entry3);
					
					
					String workbook =
					"<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
					+ "<workbook xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\" "
					+ "xmlns:r=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships\">"
					+ "<sheets>"
					+ "<sheet name=\"Records\" sheetId=\"1\" r:id=\"rId1\"/>"
					+ "</sheets>"
					+ "</workbook>";
					
					
					zip.write(
					workbook.getBytes("UTF-8")
					);
					
					zip.closeEntry();
					
					
					// =================================================
					// xl/_rels/workbook.xml.rels
					// =================================================
					
					java.util.zip.ZipEntry entry4 =
					new java.util.zip.ZipEntry(
					"xl/_rels/workbook.xml.rels"
					);
					
					zip.putNextEntry(entry4);
					
					
					String workbookRels =
					"<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
					+ "<Relationships xmlns=\"http://schemas.openxmlformats.org/package/2006/relationships\">"
					+ "<Relationship Id=\"rId1\" Type=\"http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet\" Target=\"worksheets/sheet1.xml\"/>"
					+ "</Relationships>";
					
					
					zip.write(
					workbookRels.getBytes("UTF-8")
					);
					
					zip.closeEntry();
					
					
					// =================================================
					// xl/worksheets/sheet1.xml
					// =================================================
					
					StringBuilder sheet =
					new StringBuilder();
					
					
					sheet.append(
					"<?xml version=\"1.0\" encoding=\"UTF-8\"?>"
					);
					
					
					sheet.append(
					"<worksheet xmlns=\"http://schemas.openxmlformats.org/spreadsheetml/2006/main\">"
					);
					
					
					sheet.append("<sheetData>");
					
					
					String[] lines =
					pendingExportData.split(
					"\\r?\\n"
					);
					
					
					int rowNumber = 1;
					
					
					for (String line : lines) {
						
						if (
						line == null
						|| line.trim().length() == 0
						) {
							continue;
						}
						
						
						sheet.append(
						"<row r=\""
						);
						
						sheet.append(rowNumber);
						
						sheet.append("\">");
						
						
						String[] parts =
						line.split(
						"\\s*\\|\\s*",
						-1
						);
						
						
						for (
						int i = 0;
						i < parts.length;
						i++
						) {
							
							int column =
							i + 1;
							
							
							StringBuilder columnName =
							new StringBuilder();
							
							
							while (column > 0) {
								
								int remainder =
								(column - 1) % 26;
								
								
								columnName.insert(
								0,
								(char)(
								'A'
								+ remainder
								)
								);
								
								
								column =
								(column - 1) / 26;
							}
							
							
							String value =
							parts[i] == null
							? ""
							: parts[i];
							
							
							value =
							value
							.replace(
							"&",
							"&amp;"
							)
							.replace(
							"<",
							"&lt;"
							)
							.replace(
							">",
							"&gt;"
							)
							.replace(
							"\"",
							"&quot;"
							)
							.replace(
							"'",
							"&apos;"
							);
							
							
							sheet.append(
							"<c r=\""
							);
							
							sheet.append(
							columnName.toString()
							);
							
							sheet.append(
							rowNumber
							);
							
							sheet.append(
							"\" t=\"inlineStr\"><is><t>"
							);
							
							sheet.append(value);
							
							sheet.append(
							"</t></is></c>"
							);
						}
						
						
						sheet.append("</row>");
						
						
						rowNumber++;
					}
					
					
					sheet.append("</sheetData>");
					
					sheet.append("</worksheet>");
					
					
					java.util.zip.ZipEntry entry5 =
					new java.util.zip.ZipEntry(
					"xl/worksheets/sheet1.xml"
					);
					
					zip.putNextEntry(entry5);
					
					
					zip.write(
					sheet.toString().getBytes("UTF-8")
					);
					
					zip.closeEntry();
					
					
					zip.finish();
					
					output.flush();
					
					exportSuccess = true;
					
					
				} catch (Exception e) {
					
					exportSuccess = false;
					
					
				} finally {
					
					try {
						
						if (zip != null) {
							zip.close();
						} else if (output != null) {
							output.close();
						}
						
					} catch (Exception ignored) {
					}
				}
			}
			
			
			// =================================================
			// SUCCESS
			// =================================================
			
			if (exportSuccess) {
				
				exportPrefs.edit()
				.putInt(
				"export_serial",
				pendingSerial + 1
				)
				.remove(
				"pending_export_serial"
				)
				.remove(
				"pending_export_format"
				)
				.remove(
				"pending_export_filename"
				)
				.apply();
				
				
				Toast.makeText(
				SettingsActivity.this,
				"Export saved successfully",
				Toast.LENGTH_SHORT
				).show();
				
				
			} else {
				
				Toast.makeText(
				SettingsActivity.this,
				"Export failed",
				Toast.LENGTH_SHORT
				).show();
			}
			
			
			// =================================================
			// CLEAR TEMPORARY EXPORT STATE
			// =================================================
			
			exportResultUri = null;
			
			pendingExportData = "";
			
			return;
		}
		
		
		// =====================================================
		// 2. READ SOURCE FILE
		// =====================================================
		
		java.io.File sourceFile =
		new java.io.File(
		SettingsActivity.this.getFilesDir(),
		"saverecords.txt"
		);
		
		
		if (!sourceFile.exists()) {
			
			Toast.makeText(
			SettingsActivity.this,
			"No saved records found",
			Toast.LENGTH_SHORT
			).show();
			
			return;
		}
		
		
		// =====================================================
		// 3. READ ALL RECORDS
		// =====================================================
		
		StringBuilder allRecords =
		new StringBuilder();
		
		
		try {
			
			java.io.BufferedReader reader =
			new java.io.BufferedReader(
			new java.io.FileReader(
			sourceFile
			)
			);
			
			
			String line;
			
			
			while (
			(line = reader.readLine()) != null
			) {
				
				allRecords.append(line);
				allRecords.append("\n");
			}
			
			
			reader.close();
			
			
		} catch (Exception e) {
			
			Toast.makeText(
			SettingsActivity.this,
			"Unable to read records",
			Toast.LENGTH_SHORT
			).show();
			
			return;
		}
		
		
		// =====================================================
		// 4. SELECT EXPORT DATA
		// =====================================================
		
		String dataToExport;
		
		
		if ("selected".equals(exportMode)) {
			
			android.content.SharedPreferences selectionPrefs =
			SettingsActivity.this.getSharedPreferences(
			"selection_settings",
			MODE_PRIVATE
			);
			
			
			dataToExport =
			selectionPrefs.getString(
			"selected_export_data",
			""
			);
			
			
			if (
			dataToExport == null
			|| dataToExport.trim().length() == 0
			) {
				
				Toast.makeText(
				SettingsActivity.this,
				"No selected data available",
				Toast.LENGTH_SHORT
				).show();
				
				return;
			}
			
			
		} else {
			
			dataToExport =
			allRecords.toString();
		}
		
		
		// =====================================================
		// 5. CHECK EMPTY DATA
		// =====================================================
		
		if (
		dataToExport == null
		|| dataToExport.trim().length() == 0
		) {
			
			Toast.makeText(
			SettingsActivity.this,
			"No data available to export",
			Toast.LENGTH_SHORT
			).show();
			
			return;
		}
		
		
		// =====================================================
		// 6. STORE DATA UNTIL PICKER RETURNS
		// =====================================================
		
		pendingExportData =
		dataToExport;
		
		
		// =====================================================
		// 7. FORMAT
		// =====================================================
		
		if (
		exportFormat == null
		|| exportFormat.length() == 0
		) {
			
			exportFormat =
			"TXT";
		}
		
		
		// =====================================================
		// 8. DATE
		// =====================================================
		
		String exportDate =
		new java.text.SimpleDateFormat(
		"yyyyMMdd",
		java.util.Locale.getDefault()
		).format(
		new java.util.Date()
		);
		
		
		// =====================================================
		// 9. SERIAL
		// =====================================================
		
		android.content.SharedPreferences exportPrefs =
		SettingsActivity.this.getSharedPreferences(
		"export_settings",
		MODE_PRIVATE
		);
		
		
		int serial =
		exportPrefs.getInt(
		"export_serial",
		1
		);
		
		
		// =====================================================
		// 10. EXTENSION
		// =====================================================
		
		String extension;
		
		
		if ("CSV".equals(exportFormat)) {
			
			extension = ".csv";
			
		} else if ("PDF".equals(exportFormat)) {
			
			extension = ".pdf";
			
		} else if ("XLSX".equals(exportFormat)) {
			
			extension = ".xlsx";
			
		} else {
			
			extension = ".txt";
		}
		
		
		// =====================================================
		// 11. FILENAME
		// =====================================================
		
		String exportFileName =
		"SaveRecords_"
		+ exportDate
		+ "_"
		+ serial
		+ extension;
		
		
		// =====================================================
		// 12. MIME TYPE
		// =====================================================
		
		String mimeType;
		
		
		if ("CSV".equals(exportFormat)) {
			
			mimeType =
			"text/csv";
			
		} else if ("PDF".equals(exportFormat)) {
			
			mimeType =
			"application/pdf";
			
		} else if ("XLSX".equals(exportFormat)) {
			
			mimeType =
			"application/vnd.openxmlformats-officedocument.spreadsheetml.sheet";
			
		} else {
			
			mimeType =
			"text/plain";
		}
		
		
		// =====================================================
		// 13. SAVE PENDING INFORMATION
		// =====================================================
		
		exportPrefs.edit()
		.putInt(
		"pending_export_serial",
		serial
		)
		.putString(
		"pending_export_format",
		exportFormat
		)
		.putString(
		"pending_export_filename",
		exportFileName
		)
		.apply();
		
		
		// =====================================================
		// 14. OPEN ANDROID SAVE LOCATION PICKER
		// =====================================================
		
		Intent exportIntent =
		new Intent(
		Intent.ACTION_CREATE_DOCUMENT
		);
		
		
		exportIntent.addCategory(
		Intent.CATEGORY_OPENABLE
		);
		
		
		exportIntent.setType(
		mimeType
		);
		
		
		exportIntent.putExtra(
		Intent.EXTRA_TITLE,
		exportFileName
		);
		
		
		startActivityForResult(
		exportIntent,
		1001
		);
	}
	
}
